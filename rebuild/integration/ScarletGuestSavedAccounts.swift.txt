// P66: per-Apple-ID credential slots inside the embedded SideStore LiveProcess.
// Credentials never traverse the account-list XPC response or leave the iOS Keychain.
// The existing SideStore active Keychain fields remain the native signing source of truth.
private struct ScarletSavedAccount: Codable {
    let email: String
    let password: String?
    let adsid: String
    let xcodeToken: String
    let teamID: String
    let anisetteProfile: String
}

private struct ScarletSavedAccountRow: Codable {
    let id: String
    let active: Bool
    let teamID: String
    let certificateReady: Bool
}

@MainActor
private enum ScarletAccountVault {
    private static let service = "com.kdt.scarlet.sidestore.account.v1"
    private static let maxAccounts = 8

    private static func identifier(_ email: String) -> String {
        email.trimmingCharacters(in: .whitespacesAndNewlines).lowercased()
    }

    static func save(_ slot: ScarletSavedAccount) -> Bool {
        let accountID = identifier(slot.email)
        guard accountID.utf8.count > 3, accountID.utf8.count <= 254,
              !slot.adsid.isEmpty, !slot.xcodeToken.isEmpty,
              !slot.teamID.isEmpty,
              let encoded = try? JSONEncoder().encode(slot),
              encoded.count < 16384 else { return false }
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: accountID
        ]
        let update = SecItemUpdate(query as CFDictionary,
            [kSecValueData as String: encoded] as CFDictionary)
        if update == errSecSuccess { return true }
        guard update == errSecItemNotFound,
              let existing = readAll(), existing.count < maxAccounts else { return false }
        let insert: [String: Any] = query.merging([
            kSecValueData as String: encoded,
            kSecAttrAccessible as String: kSecAttrAccessibleAfterFirstUnlockThisDeviceOnly
        ]) { _, new in new }
        return SecItemAdd(insert as CFDictionary, nil) == errSecSuccess
    }

    static func readAll() -> [ScarletSavedAccount]? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecMatchLimit as String: kSecMatchLimitAll,
            kSecReturnData as String: true,
            kSecReturnAttributes as String: true
        ]
        var value: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &value)
        if status == errSecItemNotFound { return [] }
        guard status == errSecSuccess,
              let items = value as? [[String: Any]],
              items.count <= maxAccounts else { return nil }
        var found: [ScarletSavedAccount] = []
        for item in items {
            guard let data = item[kSecValueData as String] as? Data,
                  let storedID = item[kSecAttrAccount as String] as? String,
                  let slot = try? JSONDecoder().decode(ScarletSavedAccount.self, from: data),
                  identifier(slot.email) == storedID,
                  !slot.xcodeToken.isEmpty, !slot.adsid.isEmpty else { return nil }
            found.append(slot)
        }
        return found.sorted { identifier($0.email) < identifier($1.email) }
    }

    static func captureCurrent() async -> ScarletSavedAccount? {
        let auth = AuthManager.shared
        guard ScarletGuestSignInAgent.persistedState(),
              let email = auth.currentAppleID,
              let adsid = auth.adsid,
              let token = auth.xcodeToken else { return nil }
        do {
            try await DatabaseManager.shared.start()
            guard let team = try? await auth.getAuthenticatedTeam(),
                  !team.identifier.isEmpty else { return nil }
            let profile = UserDefaults.standard.string(
                forKey: "scarlet.auth.lastSuccessfulAnisetteProfile") ?? "default"
            return ScarletSavedAccount(email: email, password: auth.password,
                adsid: adsid, xcodeToken: token, teamID: team.identifier,
                anisetteProfile: profile)
        } catch { return nil }
    }

    static func activate(_ slot: ScarletSavedAccount) async -> Bool {
        guard !slot.email.isEmpty, !slot.adsid.isEmpty, !slot.xcodeToken.isEmpty,
              !slot.teamID.isEmpty else { return false }
        let auth = AuthManager.shared
        do {
            try await DatabaseManager.shared.start()
            let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
            // A persisted slot is usable only with the exact account/team
            // rows originally verified against Apple's portal at sign-in.
            let valid = await context.perform { () -> Bool in
                let accounts = (try? context.fetch(
                    Account.fetchRequest() as NSFetchRequest<Account>)) ?? []
                let teams = (try? context.fetch(
                    Team.fetchRequest() as NSFetchRequest<Team>)) ?? []
                guard let account = accounts.first(where: {
                    $0.appleID.caseInsensitiveCompare(slot.email) == .orderedSame
                }),
                let team = teams.first(where: {
                    $0.identifier == slot.teamID &&
                    $0.account?.objectID == account.objectID
                }) else { return false }
                for item in accounts {
                    item.isActiveAccount = item.objectID == account.objectID
                }
                for item in teams {
                    item.isActiveTeam = item.objectID == team.objectID
                }
                return (try? context.save()) != nil
            }
            guard valid else { return false }
            auth.scarletInvalidateActiveCaches()
            auth.currentAppleID = slot.email
            auth.password = slot.password
            auth.adsid = slot.adsid
            auth.xcodeToken = slot.xcodeToken
            UserDefaults.standard.set(slot.anisetteProfile,
                forKey: "scarlet.auth.lastSuccessfulAnisetteProfile")
            guard ScarletGuestSignInAgent.persistedState(),
                  (try? await auth.getAuthenticatedTeam())?.identifier == slot.teamID else {
                return false
            }
            return true
        } catch { return false }
    }

    static func savedRows() async -> (String, String) {
        // Migrate the original active SideStore login without deleting it.
        if let current = await captureCurrent(), !save(current) {
            return ("[]", "keychain_write_failed")
        }
        guard let slots = readAll() else { return ("[]", "keychain_read_failed") }
        let active = identifier(AuthManager.shared.currentAppleID ?? "")
        let rows = slots.map { slot in
            ScarletSavedAccountRow(id: slot.email,
                active: identifier(slot.email) == active &&
                    ScarletGuestSignInAgent.persistedState(),
                teamID: slot.teamID,
                certificateReady: ScarletInstallTeamAuthority.matchingCertificate(
                    for: slot.teamID).map { $0.x509.expiryDate > Date() } ?? false)
        }
        guard let data = try? JSONEncoder().encode(rows),
              let json = String(data: data, encoding: .utf8),
              json.utf8.count <= 8192 else {
            return ("[]", "response_invalid")
        }
        return (json, "ready")
    }

    static func switchTo(_ id: String) async -> (Bool, String) {
        guard id.utf8.count <= 254, let slots = readAll() else {
            return (false, "keychain_unavailable")
        }
        guard let selected = slots.first(where: {
            identifier($0.email) == identifier(id)
        }) else { return (false, "account_not_found") }
        // Never overwrite the current original SideStore credentials until
        // that account has a durable recoverable slot.
        let previous = await captureCurrent()
        if ScarletGuestSignInAgent.persistedState() && previous == nil {
            return (false, "current_session_unverified")
        }
        if let previous, !save(previous) {
            return (false, "keychain_write_failed")
        }
        if await activate(selected) {
            // Rebind the locally signable key to the selected team before
            // other work can use the native active signing slot. A missing
            // cert is handled by user-triggered Apple issuance on the host.
            _ = await ScarletInstallTeamAuthority.prepare(teamID: selected.teamID)
            return (true, "active_team_verified")
        }
        if let previous, !(await activate(previous)) {
            return (false, "rollback_unverified")
        }
        return (false, "account_activation_unverified")
    }
}

extension AuthManager {
    fileprivate func scarletInvalidateActiveCaches() {
        team = nil
        session = nil
    }
    fileprivate func scarletInvalidateCachedTeam() {
        team = nil
    }
}

extension ScarletGuestSignInAgent {
    @objc(listAccounts:)
    static func listAccounts(_ reply: @escaping (String, String) -> Void) {
        Task { @MainActor in
            let (json, state) = await ScarletAccountVault.savedRows()
            reply(json, state)
        }
    }

    @objc(selectAccount:reply:)
    static func selectAccount(_ payload: NSDictionary,
                              reply: @escaping (Bool, String) -> Void) {
        guard let id = payload["accountID"] as? String,
              id.utf8.count > 3, id.utf8.count <= 254 else {
            reply(false, "invalid_request"); return
        }
        Task { @MainActor in
            guard activeRequest == nil else {
                reply(false, "authentication_in_progress"); return
            }
            guard await ScarletTeamOperationGate.shared.claim() else {
                reply(false, "team_operation_in_progress"); return
            }
            defer { Task { await ScarletTeamOperationGate.shared.release() } }
            activeRequest = UUID().uuidString
            defer { activeRequest = nil }
            let (ok, reason) = await ScarletAccountVault.switchTo(id)
            reply(ok, reason)
        }
    }
}

// Serializes all operations that mutate the one native SideStore active
// account/team/certificate slot. Never allow an install to race a login.
private actor ScarletTeamOperationGate {
    static let shared = ScarletTeamOperationGate()
    private var running = false
    func claim() -> Bool {
        guard !running else { return false }
        running = true
        return true
    }
    func release() { running = false }
}

// R6P68: separate native signing identity from the host's signed entitlement.
// Original SideStore uses ONE active certificate; retain all existing P12s
// and choose a locally saved, matching-key certificate for each Apple Team.
@MainActor
enum ScarletInstallTeamAuthority {
    // P90: The native SideStore Team -> Account relationship is authoritative.
    // Team ID alone is insufficient when two saved Apple IDs are members of
    // the same Apple Developer team. Do not alter account/team ownership.
    static func activeOwnership() async -> (appleID: String, teamID: String)? {
        guard let team = await activeTeam(),
              let account = AuthManager.shared.currentAppleID?
                .trimmingCharacters(in: .whitespacesAndNewlines).lowercased(),
              !account.isEmpty else { return nil }
        do { try await DatabaseManager.shared.start() }
        catch { return nil }
        let context = DatabaseManager.shared.viewContext
        let teams = (try? context.fetch(Team.fetchRequest())) ?? []
        guard teams.contains(where: {
            $0.identifier == team.identifier && $0.isActiveTeam &&
            $0.account?.isActiveAccount == true &&
            $0.account?.appleID.caseInsensitiveCompare(account) == .orderedSame
        }) else { return nil }
        return (account, team.identifier)
    }

    static func appBelongsToOwner(_ app: InstalledApp,
        owner: (appleID: String, teamID: String)) -> Bool {
        app.team?.identifier == owner.teamID &&
        app.team?.account?.appleID.caseInsensitiveCompare(owner.appleID) == .orderedSame
    }

    static func validTeam(_ value: String) -> Bool {
        value.utf8.count == 10 && value.allSatisfy {
            $0.isASCII && ($0.isUppercase || $0.isNumber)
        }
    }

    static func certificateBelongsToTeam(_ cert: ALTCertificate,
                                         teamID: String) -> Bool {
        guard validTeam(teamID), let der = cert.data else { return false }
        let subject = parseCertificate(derData: der).subject
        // Compare the exact ASN.1 Organizational Unit value. A substring
        // elsewhere in a certificate is NOT evidence of team ownership.
        let fields = subject.split(separator: ",")
            .map { $0.trimmingCharacters(in: .whitespacesAndNewlines) }
        return fields.contains("Organizational Unit=" + teamID)
    }

    static func matchingCertificate(for teamID: String) -> ALTCertificate? {
        let manager = CertificateManager.shared
        if let active = manager.activeCertificate?.certificate,
           certificateBelongsToTeam(active, teamID: teamID) { return active }
        return manager.getAllLocalCertificates().first {
            certificateBelongsToTeam($0, teamID: teamID)
        }
    }

    // Return a Sendable Bool across the MainActor boundary. A saved but
    // expired per-team P12 is not signing readiness.
    static func hasValidSigningCertificate(for teamID: String) -> Bool {
        matchingCertificate(for: teamID).map { $0.x509.expiryDate > Date() } ?? false
    }

    static func activeTeam() async -> ALTTeam? {
        guard ScarletGuestSignInAgent.persistedState(),
              let team = try? await AuthManager.shared.getAuthenticatedTeam(),
              validTeam(team.identifier) else { return nil }
        return team
    }

    static func prepare(teamID: String) async -> String {
        guard validTeam(teamID) else { return "invalid_team" }
        guard let team = await activeTeam(), team.identifier == teamID else {
            return "selected_team_changed"
        }
        let teamStatus = await ScarletGuestSignInAgent
            .ensureActiveTeamForRefresh(requireHostTeam: false)
        guard teamStatus == "ready" || teamStatus == "team_restored" else {
            return teamStatus
        }
        guard let cert = matchingCertificate(for: teamID),
              cert.x509.expiryDate > Date() else {
            return "selected_team_certificate_missing"
        }
        let manager = CertificateManager.shared
        if manager.activeCertificate?.serialNumber != cert.serialNumber {
            do { try manager.setActiveCertificate(cert) }
            catch { return "selected_team_certificate_activation_failed" }
        }
        guard let active = manager.activeCertificate,
              active.serialNumber == cert.serialNumber,
              certificateBelongsToTeam(active.certificate, teamID: teamID) else {
            return "selected_team_certificate_readback_failed"
        }
        return "ready"
    }

    // P89: SideInstaller's CertManager.performImport verifies the P12 against
    // the currently selected Apple account/team before changing its key.
    // SideStore uses its own CertificateManager Keychain, NEVER the signed
    // LiveContainer host's global LCCertificateData/Password storage.
    static func importArchive(_ archive: Data, password: String,
                              teamID: String) async -> String {
        guard validTeam(teamID), archive.count > 0, archive.count <= 10_000_000,
              password.utf8.count <= 1024 else { return "invalid_p12_import_request" }
        guard let team = await activeTeam(), team.identifier == teamID else {
            return "selected_team_changed"
        }
        let teamStatus = await ScarletGuestSignInAgent
            .ensureActiveTeamForRefresh(requireHostTeam: false)
        guard teamStatus == "ready" || teamStatus == "team_restored" else {
            return teamStatus
        }
        let certificate: ALTCertificate
        do {
            certificate = try CertificateManager.parse(archive, password: password)
        } catch {
            return "p12_decode_or_password_invalid"
        }
        guard certificateBelongsToTeam(certificate, teamID: teamID),
              certificate.x509.expiryDate > Date() else {
            return "p12_wrong_team_or_expired"
        }
        // SideInstaller checks the imported certificate serial against
        // certificates Apple actually registered to the chosen account.
        // Matching ASN.1 OU alone does not authorize an otherwise unknown key.
        do {
            let portalCertificates = try await DeveloperPortalProxy.shared
                .fetchCertificates(team: team)
            guard portalCertificates.contains(where: {
                $0.serialNumber.caseInsensitiveCompare(certificate.serialNumber) == .orderedSame &&
                $0.expiryDate > Date()
            }) else { return "p12_not_registered_with_selected_team" }
        } catch {
            return "selected_team_certificate_portal_failed"
        }
        guard (await activeTeam())?.identifier == teamID else {
            return "selected_team_changed"
        }
        let manager = CertificateManager.shared
        if manager.activeCertificate?.serialNumber == certificate.serialNumber {
            return await prepare(teamID: teamID) == "ready"
                ? "ready_existing" : "selected_team_certificate_readback_failed"
        }
        let original = manager.activeCertificate?.certificate
        do {
            try manager.setActiveCertificate(certificate)
        } catch {
            if let original { try? manager.setActiveCertificate(original) }
            return "selected_team_certificate_storage_failed"
        }
        guard (await activeTeam())?.identifier == teamID,
              manager.activeCertificate?.serialNumber == certificate.serialNumber,
              await prepare(teamID: teamID) == "ready" else {
            if let original { try? manager.setActiveCertificate(original) }
            return "selected_team_certificate_readback_failed"
        }
        return "imported_selected_team"
    }

    // P86: explicitly initiated login/account-switch/IPA install certificate
    // provisioning for this team's *own* Apple portal identity. Original
    // CertificateProvisioningFlow checks isCustomCert and returns the prior
    // active team's cert unchanged, so B never reaches createCertificate.
    // Do not clear an existing A team cert, silently revoke Apple certs, or
    // infer that account B owns host A's provisioning profile.
    static func issue(teamID: String) async -> String {
        guard validTeam(teamID) else { return "invalid_team" }
        guard let team = await activeTeam(), team.identifier == teamID else {
            return "selected_team_changed"
        }
        let teamStatus = await ScarletGuestSignInAgent
            .ensureActiveTeamForRefresh(requireHostTeam: false)
        guard teamStatus == "ready" || teamStatus == "team_restored" else {
            return teamStatus
        }
        // P96: an explicit request is NOT satisfied by local X509 expiry or
        // the Team OU alone. The certificate serial must remain registered
        // for the authenticated active Apple-ID's Team in Apple's portal.
        guard let owner = await activeOwnership(), owner.teamID == teamID else {
            return "selected_account_owner_unverified"
        }
        do {
            let portal = try await DeveloperPortalProxy.shared.fetchCertificates(team: team)
            guard let currentOwner = await activeOwnership(),
                  currentOwner.appleID == owner.appleID,
                  currentOwner.teamID == owner.teamID else {
                return "selected_account_owner_changed"
            }
            let manager = CertificateManager.shared
            // Existing certificate + private key is reusable ONLY when
            // Apple still recognizes that exact unexpired certificate.
            if let current = matchingCertificate(for: teamID),
               current.x509.expiryDate > Date(),
               portal.contains(where: {
                   $0.serialNumber.caseInsensitiveCompare(current.serialNumber) == .orderedSame &&
                   $0.expiryDate > Date()
               }) {
                return await prepare(teamID: teamID) == "ready"
                    ? "ready_existing_portal" : "selected_team_certificate_readback_failed"
            }
            // The original SideStore keychain may retain an alternate
            // registered signable certificate. Reuse it before issuing anew.
            for remote in portal where remote.expiryDate > Date() {
                guard let signable = manager.getSignableCertificate(
                        for: remote.serialNumber,
                        fallbackPassword: remote.machineIdentifier),
                      signable.serialNumber.lowercased() == remote.serialNumber.lowercased(),
                      certificateBelongsToTeam(signable, teamID: teamID) else { continue }
                guard let currentOwner = await activeOwnership(),
                      currentOwner.appleID == owner.appleID,
                      currentOwner.teamID == owner.teamID else {
                    return "selected_account_owner_changed"
                }
                try manager.setActiveCertificate(signable)
                guard await prepare(teamID: teamID) == "ready" else {
                    return "selected_team_certificate_readback_failed"
                }
                return "ready_existing_portal"
            }
            // No valid matching private key exists: SideInstaller's CSR
            // request and SideStore's native requestCertificate both issue
            // directly at Apple. createCertificate returns the matching key.
            // A 7460/quota failure does NOT revoke another app's certificate.
            guard let currentOwner = await activeOwnership(),
                  currentOwner.appleID == owner.appleID,
                  currentOwner.teamID == owner.teamID else {
                return "selected_account_owner_changed"
            }
            let originalActive = manager.activeCertificate?.certificate
            let newlyIssued: ALTCertificate
            do {
                newlyIssued = try await DeveloperPortalProxy.shared.createCertificate(
                    machineName: "Scarlet Crystal (" + teamID + ")", team: team)
            } catch {
                let failure = error as NSError
                return failure.code == 7460 ? "selected_team_certificate_quota_reached"
                    : "selected_team_certificate_portal_failed"
            }
            guard certificateBelongsToTeam(newlyIssued, teamID: teamID),
                  newlyIssued.x509.expiryDate > Date() else {
                return "selected_team_certificate_unverified"
            }
            do {
                try manager.setActiveCertificate(newlyIssued)
            } catch {
                // Never remove/overwrite unrelated locally saved certs.
                if let originalActive { try? manager.setActiveCertificate(originalActive) }
                return "selected_team_certificate_storage_failed"
            }
            guard await prepare(teamID: teamID) == "ready",
                  manager.activeCertificate?.serialNumber == newlyIssued.serialNumber else {
                return "selected_team_certificate_readback_failed"
            }
            return "ready_issued_apple"
        } catch {
            return "selected_team_certificate_portal_failed"
        }
    }
}

extension ScarletGuestSignInAgent {
    @objc(importInstallCertificate:reply:)
    static func importInstallCertificate(_ request: NSDictionary,
        reply: @escaping (Bool, String) -> Void) {
        guard let archive = request["archive"] as? Data,
              let password = request["password"] as? String,
              let teamID = request["teamID"] as? String else {
            reply(false, "invalid_p12_import_request"); return
        }
        Task { @MainActor in
            guard activeRequest == nil else {
                reply(false, "authentication_in_progress"); return
            }
            guard await ScarletTeamOperationGate.shared.claim() else {
                reply(false, "team_operation_in_progress"); return
            }
            let status = await ScarletInstallTeamAuthority.importArchive(
                archive, password: password, teamID: teamID)
            await ScarletTeamOperationGate.shared.release()
            reply(status == "imported_selected_team" || status == "ready_existing", status)
        }
    }
}

extension ScarletGuestSignInAgent {
    @objc(prepareInstallTeam:reply:)
    static func prepareInstallTeam(_ id: String,
                                   reply: @escaping (Bool, String) -> Void) {
        Task { @MainActor in
            guard activeRequest == nil else {
                reply(false, "authentication_in_progress"); return
            }
            guard await ScarletTeamOperationGate.shared.claim() else {
                reply(false, "team_operation_in_progress"); return
            }
            let status = await ScarletInstallTeamAuthority.prepare(teamID: id)
            // The next on-demand issue action may immediately follow this
            // reply. Release the gate synchronously BEFORE replying.
            await ScarletTeamOperationGate.shared.release()
            reply(status == "ready", status)
        }
    }

    @objc(issueInstallTeamCertificate:reply:)
    static func issueInstallTeamCertificate(_ id: String,
                                            reply: @escaping (Bool, String) -> Void) {
        Task { @MainActor in
            guard activeRequest == nil else {
                reply(false, "authentication_in_progress"); return
            }
            guard await ScarletTeamOperationGate.shared.claim() else {
                reply(false, "team_operation_in_progress"); return
            }
            let status = await ScarletInstallTeamAuthority.issue(teamID: id)
            await ScarletTeamOperationGate.shared.release()
            reply(status == "ready" || status == "ready_existing_portal" ||
                  status == "ready_issued_apple", status)
        }
    }
}
