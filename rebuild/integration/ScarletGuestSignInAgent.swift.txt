
// Scarlet guest-auth worker injected into pinned SideStore AuthManager.swift.
// It runs in SideStore's LiveProcess context, never inside the Scarlet UI.
// NotificationCenter messages stay inside the guest process; XPC is the only
// host/guest boundary. Never log Apple IDs, passwords, OTPs or auth tokens.
@objc(ScarletGuestSignInAgent)
final class ScarletGuestSignInAgent: NSObject {
    @MainActor private static var activeRequest: String?
    @MainActor private static var pendingCode: CheckedContinuation<String, Error>?
    @MainActor private static var cancelled = false

    @objc(begin:)
    static func begin(_ payload: NSDictionary) {
        guard let id = payload["requestID"] as? String,
              let email = payload["email"] as? String,
              let password = payload["password"] as? String else { return }
        let profile = (payload["profile"] as? String) ?? "default"
        Task { @MainActor in
            guard activeRequest == nil else {
                finish(id, code: -3, domain: "scarlet.auth.busy")
                return
            }
            activeRequest = id
            cancelled = false
            defer {
                activeRequest = nil
                pendingCode = nil
                cancelled = false
            }
            do {
                let (anisette, pendingAdiBlob) = try await fetchAuthAnisette(profile: profile)
                let xcodeVersion = await AnisetteConfigManager.shared.resolvedXcodeVersion()
                let (_, session) = try await AuthManager.shared.signIn(
                    appleID: email, password: password, anisetteData: anisette,
                    xcodeVersion: xcodeVersion,
                    verificationHandler: { request in
                        switch request {
                        case .selectDeliveryMethod(let mode, let phones):
                            switch mode {
                            case .sms:
                                if let phone = phones.first { return .requestSMS(phoneID: phone.id) }
                                return .requestTrustedDevice
                            case .voice:
                                if let phone = phones.first { return .requestVoice(phoneID: phone.id) }
                                return .requestTrustedDevice
                            case .trustedDevice:
                                return .requestTrustedDevice
                            }
                        case .trustedDevice, .sms, .voice:
                            let code = try await receiveUserCode(id)
                            return .verificationCode(code)
                        }
                    })
                guard !cancelled else { throw CancellationError() }
                // Only commit any freshly provisioned compatibility ADI after
                // verified Apple authentication; never clear previous ADI.
                if let pendingAdiBlob {
                    AnisetteConfigManager.shared.anisetteAdiBlob = pendingAdiBlob.base64EncodedString()
                }
                AuthManager.shared.currentAppleID = email
                AuthManager.shared.password = password
                AuthManager.shared.adsid = session.dsid
                AuthManager.shared.xcodeToken = session.authToken
                finish(id, code: 0, domain: "success")
            } catch let error as ServerError {
                // A Swift enum NSError code (3 here) is the enum case index,
                // NOT Apple's GrandSlam status. Preserve the real status
                // from .underlyingError without serializing Apple responses.
                switch error {
                case .underlyingError(let appleCode, _):
                    finish(id, code: appleCode == 0 ? -3000 : appleCode,
                           domain: "SideSign.ServerError.underlyingError")
                case .badServerResponse:
                    finish(id, code: -3001, domain: "SideSign.ServerError.badServerResponse")
                case .invalidResponseFormat:
                    finish(id, code: -3002, domain: "SideSign.ServerError.invalidResponseFormat")
                case .missingKey:
                    finish(id, code: -3003, domain: "SideSign.ServerError.missingKey")
                }
            } catch let error as DeveloperPortalError {
                // Only categorical errors traverse the guest XPC bridge.
                // No raw Apple server message, email, OTP, token or 2FA phone.
                switch error {
                case .incorrectCredentials:
                    finish(id, code: -3101, domain: "SideSign.DeveloperPortalError.incorrectCredentials")
                case .invalidAnisetteData:
                    finish(id, code: -3102, domain: "SideSign.DeveloperPortalError.invalidAnisetteData")
                case .authenticationHandshakeFailed:
                    finish(id, code: -3103, domain: "SideSign.DeveloperPortalError.authenticationHandshakeFailed")
                case .requiresTwoFactorAuthentication:
                    finish(id, code: -3104, domain: "SideSign.DeveloperPortalError.requiresTwoFactorAuthentication")
                case .incorrectVerificationCode:
                    finish(id, code: -3105, domain: "SideSign.DeveloperPortalError.incorrectVerificationCode")
                case .accountRepairRequired:
                    finish(id, code: -3106, domain: "SideSign.DeveloperPortalError.accountRepairRequired")
                case .tooManyAttempts:
                    finish(id, code: -3107, domain: "SideSign.DeveloperPortalError.tooManyAttempts")
                case .userCancelled:
                    finish(id, code: -3108, domain: "SideSign.DeveloperPortalError.userCancelled")
                default:
                    finish(id, code: -3199, domain: "SideSign.DeveloperPortalError.other")
                }
            } catch is CancellationError {
                finish(id, code: -3200, domain: "scarlet.auth.cancelled")
            } catch {
                let failure = error as NSError
                // Preserve only the error namespace and numeric code. The
                // diagnostic recorder also redacts identifiers and secrets.
                finish(id, code: failure.code == 0 ? -3300 : failure.code,
                       domain: "scarlet.auth.unclassified")
            }
        }
    }


    // Alternative profile is an explicit, single-attempt user choice after
    // Apple -80045. Do not overwrite existing Anisette config, certificate,
    // server rotation preference, device identifier or keychain on failure.
    // Use a fresh manager because upstream remoteCache ignores header variants.
    @MainActor
    private static func fetchAuthAnisette(profile: String) async throws -> (ALTAnisetteData, Data?) {
        guard profile == "windows" || profile == "mac" else {
            return (try await AnisetteProvider.fetch(), nil)
        }

        var headers = await AnisetteConfigManager.shared.makeRequestHeaders()
        if profile == "windows" {
            headers.clientInfo = "<PC> <Windows;6.2(0,0);9200> <com.apple.AuthKit/1 (com.apple.iTunes/12.12.4.1)>"
            headers.userAgent = "akd/1.0 CFNetwork/808.1.4"
        } else {
            headers.clientInfo = "<MacBookPro13,3> <Mac OS X;27.2;26B118> <com.apple.AuthKit/1 (com.apple.akd/1.0)>"
            headers.userAgent = "akd/1.0 CFNetwork/811.5.4 Darwin/16.7.0"
        }
        let urls = await AnisetteServersManager.shared.getActiveServerURLs()
        let servers = urls.compactMap(URL.init(string:)).filter { $0.scheme == "https" }
        guard !servers.isEmpty else { throw AnisetteError.noServersConfigured }
        let prior = UserDefaults.standard.menuAnisetteURL
        let start = servers.firstIndex(where: { $0.absoluteString == prior }) ?? 0
        let selected = UserDefaults.standard.disableAnisetteRotation
            ? [servers[start]] : servers
        let currentIdentifier = await AnisetteConfigManager.shared.resolveDeviceIdentifier()
        let existingBlob = AnisetteConfigManager.shared.anisetteAdiBlob
            .flatMap { Data(base64Encoded: $0) }
        let temporaryProvider = SideSign.AnisetteDataManager()
        let (result, newAdiBlob) = try await temporaryProvider.fetchAnisetteDataWithFailover(
            servers: selected,
            startIndex: selected.count == 1 ? 0 : start,
            identifier: currentIdentifier,
            existingAdiBlob: existingBlob,
            headers: headers,
            onError: { _ in false }
        )
        // The auth request must use the same Client-Info / User-Agent pair
        // supplied during Anisette generation, even if a server omits them.
        var compatible = result
        compatible.clientInfo = headers.clientInfo ?? result.clientInfo
        compatible.userAgent = headers.userAgent ?? result.userAgent
        return (compatible, newAdiBlob)
    }

    @MainActor
    private static func receiveUserCode(_ id: String) async throws -> String {
        guard activeRequest == id, !cancelled else { throw CancellationError() }
        return try await withCheckedThrowingContinuation { continuation in
            pendingCode = continuation
            NotificationCenter.default.post(
                name: Notification.Name("ScarletGuestAuthNeedsCode"),
                object: nil, userInfo: ["requestID": id])
        }
    }

    @objc(receiveCode:)
    static func receiveCode(_ payload: NSDictionary) {
        guard let id = payload["requestID"] as? String,
              let code = payload["code"] as? String else { return }
        Task { @MainActor in
            guard activeRequest == id, let continuation = pendingCode else { return }
            pendingCode = nil
            continuation.resume(returning: code)
        }
    }

    @objc(cancel:)
    static func cancel(_ payload: NSDictionary) {
        guard let id = payload["requestID"] as? String else { return }
        Task { @MainActor in
            guard activeRequest == id else { return }
            cancelled = true
            if let continuation = pendingCode {
                pendingCode = nil
                continuation.resume(throwing: CancellationError())
            }
        }
    }

    @MainActor
    private static func finish(_ id: String, code: Int, domain: String) {
        NotificationCenter.default.post(
            name: Notification.Name("ScarletGuestAuthCompleted"), object: nil,
            userInfo: ["requestID": id, "code": code, "domain": domain])
    }
}
