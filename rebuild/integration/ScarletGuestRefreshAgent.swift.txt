// R6P73: Original SideStore starts its CoreData in an app extension but
// DatabaseManager.prepareDatabase() deliberately returns for extensions.
// A Scarlet host installed externally is therefore not a managed app even
// when its selected Apple account and saved RSD pairing are both valid.
// Bootstrap ONLY the exact, on-disk, already signed Scarlet host, with its
// actual provisioning team and upstream cached app payload. Never synthesize
// a refreshedDate, treat an external IPA as installed, or touch credentials.
private actor ScarletGuestHostManagedBootstrap {
    static let shared = ScarletGuestHostManagedBootstrap()
    private var active: Task<String, Never>?
    private var registered = false

    // P106: Bundle.Info.activeBundleIdentifier describes the embedded SideStore
    // process (and can be swapped by LiveContainer), not the real installed
    // Scarlet host. Use ONLY the host .app from the existing authenticated
    // XPC call plus its original signed profile, without guessing by suffix.
    func verifiedInstalledHostID(hostBundlePath: String) -> String? {
        guard let hostURL = verifiedHostBundleURL(hostBundlePath: hostBundlePath),
              let host = ALTApplication(fileURL: hostURL),
              let profile = host.provisioningProfile else { return nil }
        // The installed old certificate/profile may be EXPIRED. It is a
        // source to re-sign, not a certificate authorizing the new install.
        // Actual authorization comes from the new selected-account P12 and
        // original SideStore native provisioning/signing transaction.
        let id = host.bundleIdentifier
        guard id == "com.kdt.livecontainer.P8EZT3U5N4" ||
              id == "com.kdt.livecontainer.P8EZT3U5N4.P8EZT3U5N4" else {
            return nil
        }
        return id
    }

    // Only the user-tapped manual renewal may reinstall a SideInstaller-signed
    // existing host when the embedded SideStore has ZERO managed app records.
    // Enforce the installed host's real provisioning Team; never re-sign A
    // with the selected B, or accept any untrusted, fabricated IPA path.
    func verifiedSameTeamManualHostURL(hostBundlePath: String,
                                       teamID: String) -> URL? {
        guard let hostURL = verifiedHostBundleURL(hostBundlePath: hostBundlePath),
              verifiedInstalledHostID(hostBundlePath: hostBundlePath) != nil,
              let host = ALTApplication(fileURL: hostURL),
              let profile = host.provisioningProfile,
              profile.teamIdentifier == teamID else { return nil }
        // The historical provisioning expiry cannot prevent renewal.
        // The selected account's NEW signing certificate must still be valid.
        return hostURL
    }

    func ensure(hostBundlePath: String) async -> String {
        // P105: SideStore's upstream reconcile can delete externally installed
        // records after startup. A cached Bool is not a persisted CoreData
        // ownership receipt. Re-check before suppressing registration.
        if registered {
            let owner = await ScarletInstallTeamAuthority.activeOwnership()
            let installedHostID = verifiedInstalledHostID(
                hostBundlePath: hostBundlePath)
            let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
            let intact = await context.perform {
                guard let owner, let installedHostID else { return false }
                let predicate = NSPredicate(format: "%K == %@",
                    #keyPath(InstalledApp.resignedBundleIdentifier), installedHostID)
                guard let existing = InstalledApp.first(
                    satisfying: predicate, in: context) else { return false }
                return existing.isActive &&
                    existing.team?.identifier == owner.teamID &&
                    existing.team?.account?.appleID.caseInsensitiveCompare(
                        owner.appleID) == .orderedSame
            }
            if intact { return "already_registered" }
            registered = false
        }
        if let active { return await active.value }
        let task = Task { await performRegistration(hostBundlePath: hostBundlePath) }
        active = task
        let result = await task.value
        active = nil
        if result == "registered" || result == "already_managed" {
            registered = true
        }
        return result
    }

    // The original iOS executable path can be virtualized or hidden from
    // ProcessInfo when LiveContainer swaps guest process/bundle context.
    // Resolve ONLY the loaded real SideStoreSupport class bundle and its
    // actual LiveProcess.appex ancestor; never trust guest Bundle.main or a
    // user-provided path. This unlocks P73's native signed-host registration
    // when argv[0] is not the .appex executable path.
    private func verifiedHostBundleURL(hostBundlePath: String) -> URL? {
        // P85: Original host process sends ITS installed Bundle.main.bundleURL
        // via the existing authenticated LiveProcess XPC channel. The guest
        // extension cannot reliably infer that host from its class bundle.
        // Reject a guest .app, user-provided IPA, or misplaced unsigned folder.
        if hostBundlePath.hasPrefix("/") {
            let candidate = URL(fileURLWithPath: hostBundlePath)
                .resolvingSymlinksInPath().standardizedFileURL
            let actual = candidate.appendingPathComponent("PlugIns/LiveProcess.appex")
            if candidate.pathExtension == "app" &&
                FileManager.default.fileExists(atPath: actual.path) &&
                FileManager.default.fileExists(atPath: candidate.appendingPathComponent(
                    "embedded.mobileprovision").path) {
                return candidate
            }
        }
        // Compatibility only: older XPC callers can still try the old
        // extension-bundle route. Never use the guest's Bundle.main for trust.

        let codeBundle = Bundle(for: ScarletGuestRefreshAgent.self)
        var sources = [codeBundle.bundleURL]
        if let executableURL = codeBundle.executableURL {
            sources.append(executableURL)
        }
        if let executable = ProcessInfo.processInfo.arguments.first,
           !executable.isEmpty, executable.hasPrefix("/") {
            sources.append(URL(fileURLWithPath: executable))
        }
        for source in sources {
            var directory = source.standardizedFileURL
            for _ in 0..<9 {
                if directory.pathExtension == "appex" &&
                   directory.lastPathComponent == "LiveProcess.appex" &&
                   directory.deletingLastPathComponent().lastPathComponent == "PlugIns" {
                    let hostURL = directory.deletingLastPathComponent()
                        .deletingLastPathComponent().standardizedFileURL
                    if hostURL.pathExtension == "app" &&
                       FileManager.default.fileExists(atPath: hostURL.path) {
                        return hostURL
                    }
                }
                let parent = directory.deletingLastPathComponent()
                if parent.path == directory.path { break }
                directory = parent
            }
        }
        return nil
    }

    private func performRegistration(hostBundlePath: String) async -> String {
        guard let hostURL = verifiedHostBundleURL(hostBundlePath: hostBundlePath) else {
            return "host_bundle_path_unavailable"
        }
        guard hostURL.pathExtension == "app",
              let signedHost = ALTApplication(fileURL: hostURL),
              let profile = signedHost.provisioningProfile,
              profile.expirationDate > Date() else {
            return "host_signed_bundle_unavailable"
        }
        let bundleID = signedHost.bundleIdentifier
        guard bundleID == "com.kdt.livecontainer.P8EZT3U5N4" ||
              bundleID == "com.kdt.livecontainer.P8EZT3U5N4.P8EZT3U5N4" else {
            return "host_identity_mismatch"
        }
        guard let owner = await ScarletInstallTeamAuthority.activeOwnership(),
              profile.teamIdentifier == owner.teamID else {
            // Selected Apple ID B must never claim a host signed by A.
            return "host_selected_team_mismatch"
        }
        let currentTeamID = owner.teamID
        guard await ScarletInstallTeamAuthority.matchingCertificate(
            for: currentTeamID) != nil else {
            return "host_selected_team_certificate_missing"
        }
        // Use upstream SideStore's own fingerprint cache; its native
        // backgroundRefresh reads InstalledApp.fileURL from this exact cache.
        let signature: String
        do {
            let cached = try CacheAppOperation.cachePayload(for: hostURL)
            signature = cached.signature
            guard let cachedHost = ALTApplication(fileURL: cached.targetFileURL),
                  cachedHost.bundleIdentifier == bundleID,
                  cachedHost.provisioningProfile?.teamIdentifier == currentTeamID else {
                return "host_cached_bundle_mismatch"
            }
        } catch {
            return "host_payload_cache_unavailable"
        }
        let certificateSerial = CertificateManager.shared
            .getSigningCertificate(at: hostURL)?.serialNumber
        let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
        do {
            return try await context.perform {
                let duplicate = NSPredicate(
                    format: "%K == %@", #keyPath(InstalledApp.resignedBundleIdentifier), bundleID)
                if let existing = InstalledApp.first(satisfying: duplicate, in: context) {
                    // Matching bundle ID alone must never cause A's record to
                    // satisfy B, or an inactive record to look refreshable.
                    guard existing.team?.identifier == currentTeamID,
                          existing.team?.account?.appleID.caseInsensitiveCompare(
                              owner.appleID) == .orderedSame else {
                        return "host_managed_owner_mismatch"
                    }
                    return existing.isActive ? "already_managed" :
                        "host_managed_record_inactive"
                }
                let ownerTeams = (try? context.fetch(Team.fetchRequest())) ?? []
                guard let team = ownerTeams.first(where: {
                    $0.identifier == currentTeamID && $0.isActiveTeam &&
                    $0.account?.isActiveAccount == true &&
                    $0.account?.appleID.caseInsensitiveCompare(owner.appleID) == .orderedSame
                }) else { return "host_team_record_missing" }
                let record = try InstalledApp(
                    resignedAppBundle: signedHost,
                    originalBundleIdentifier: bundleID,
                    certificateSerialNumber: certificateSerial,
                    storeBuildVersion: nil, context: context)
                record.team = team
                record.isActive = true
                // Do not collapse the four distinct Scarlet extensions into
                // the host's provisioning profile when signing them later.
                record.useMainProfile = false
                record.appBundleFingerprint = signature
                var extensions = Set<InstalledExtension>()
                for extensionBundle in signedHost.appExtensions {
                    let bundle = try InstalledExtension(
                        resignedAppExtensionBundle: extensionBundle,
                        originalBundleIdentifier: extensionBundle.bundleIdentifier,
                        context: context)
                    bundle.update(resignedAppExtensionBundle: extensionBundle)
                    extensions.insert(bundle)
                }
                record.appExtensions = extensions
                try context.save()
                return "registered"
            }
        } catch {
            await context.perform { context.rollback() }
            return "host_registration_failed"
        }
    }
}

// P102: fetchAppsForRefreshingAll() filters isActive and pledge eligibility.
// Zero refresh targets is NOT evidence that CoreData has no managed apps.
// Observe the selected Apple ID+Team without exporting account/app identities.
private enum ScarletSelectedTeamRefreshInventory {
    static func zeroTargetReason(in context: NSManagedObjectContext,
                                 teamID: String, appleID: String) async -> String? {
        await context.perform {
            do {
                let all = try context.fetch(InstalledApp.fetchRequest())
                let owned = all.filter {
                    $0.team?.identifier == teamID &&
                    $0.team?.account?.appleID.caseInsensitiveCompare(
                        appleID) == .orderedSame
                }
                guard !owned.isEmpty else { return nil }
                return owned.allSatisfy { !$0.isActive }
                    ? "managed_apps_inactive_for_selected_team"
                    : "managed_apps_excluded_from_refresh_for_selected_team"
            } catch {
                return "selected_team_inventory_unavailable"
            }
        }
    }
}

// Runs inside the REAL embedded SideStore LiveProcess, where SideStore's
// CoreData database and CertificateManager live. Scarlet's host cannot infer
// whether its SideInstaller-installed signed bundle is managed by this DB.
@objc(ScarletGuestRefreshAgent)
final class ScarletGuestRefreshAgent: NSObject {
    @objc static func inspect(_ reply: @escaping (Int, Bool, Bool, String) -> Void) {
        inspectHostBundle("", reply: reply)
    }

    @objc static func inspectHostBundle(_ hostBundlePath: String,
                                       reply: @escaping (Int, Bool, Bool, String) -> Void) {
        Task {
            do {
                // SideStore's original start() prepares/reconciles its own
                // installed host record; this is not a fake copy of a DB.
                try await DatabaseManager.shared.start()
                let hostBootstrap = await ScarletGuestHostManagedBootstrap.shared.ensure(hostBundlePath: hostBundlePath)
                let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
                // Move an immutable, Sendable Team ID across CoreData's
                // background closure; the ALTTeam is not a Sendable value.
                let owner = await ScarletInstallTeamAuthority.activeOwnership()
                let currentTeamID = owner?.teamID
                let currentAppleID = owner?.appleID
                let targets = await context.perform {
                    InstalledApp.fetchAppsForRefreshingAll(in: context)
                        .filter {
                            $0.team?.identifier == currentTeamID &&
                            $0.team?.account?.appleID.caseInsensitiveCompare(
                                currentAppleID ?? "") == .orderedSame
                        }
                }
                // A Personal Team signer may suffix the original bundle ID.
                // Compare the actual signed bundle, NOT the hard-coded
                // com.kdt.livecontainer StoreApp.altstoreAppID.
                let realHostID = await ScarletGuestHostManagedBootstrap.shared
                    .verifiedInstalledHostID(hostBundlePath: hostBundlePath)
                let managesHost = realHostID.map { verified in
                    targets.contains { $0.resignedBundleIdentifier == verified }
                } ?? false
                let signingReady: Bool
                if let currentTeamID {
                    // CertificateManager and its local P12 store live on the
                    // main actor, not this synchronous Optional.flatMap body.
                    signingReady = await ScarletInstallTeamAuthority
                        .hasValidSigningCertificate(for: currentTeamID)
                } else {
                    signingReady = false
                }
                var inventoryReason: String?
                if targets.isEmpty, let currentTeamID, let currentAppleID {
                    inventoryReason = await ScarletSelectedTeamRefreshInventory
                        .zeroTargetReason(in: context, teamID: currentTeamID,
                                          appleID: currentAppleID)
                }
                let reason: String
                if let inventoryReason {
                    // Inactive/pledge-excluded managed apps are NOT absent.
                    reason = inventoryReason
                } else if targets.isEmpty {
                    // A zero-target result is not sufficient to infer the
                    // reason the native host failed to register. Preserve
                    // the exact safe categorical result of the real P73
                    // bootstrap, allowing the host or selected team's
                    // login to be corrected without force-re-signing B as A.
                    let registrationFailures: Set<String> = [
                        "host_bundle_path_unavailable", "host_signed_bundle_unavailable",
                        "host_identity_mismatch", "host_selected_team_mismatch",
                        "host_selected_team_certificate_missing",
                        "host_payload_cache_unavailable", "host_cached_bundle_mismatch",
                        "host_team_record_missing", "host_registration_failed",
                        "host_managed_owner_mismatch", "host_managed_record_inactive"
                    ]
                    // SideInstaller's account switch affects future signings,
                    // not the host's already-installed signed Team. On B,
                    // absence of B-managed apps is not a host signing error.
                    // Keep the host mismatch diagnostic inside the bootstrap,
                    // but report the selected Team's actual empty app set.
                    reason = hostBootstrap == "host_selected_team_mismatch"
                        ? "no_managed_apps_for_selected_team"
                        : (registrationFailures.contains(hostBootstrap)
                            ? hostBootstrap : "no_managed_apps_for_selected_team")
                } else if !signingReady {
                    reason = "selected_team_certificate_missing"
                } else if managesHost {
                    reason = "ready"
                } else {
                    reason = "ready_selected_team"
                }
                reply(min(targets.count, 99), managesHost, signingReady, reason)
            } catch {
                // Never send account info, DB URLs or arbitrary error strings.
                reply(0, false, false, "sidestore_database_unavailable")
            }
        }
    }
}

// Refresh exactly the apps selected by the pinned SideStore's OWN installed-app
// database, using its OWN AppManager/BackgroundRefreshAppsOperation inside
// LiveProcess. This avoids wrapping the operation through PrivateIntentRunner,
// which collapses its failures into SideStore domain/code 1 at the LC host.
// No custom signing, provisioning, installation, server or bypass logic.
private actor ScarletGuestNativeRefreshGate {
    static let shared = ScarletGuestNativeRefreshGate()
    private var active = false

    func claim() -> Bool {
        guard !active else { return false }
        active = true
        return true
    }

    func release() { active = false }
}

private enum ScarletGuestRefreshDispatchError: Error {
    case noOperation
    case savedPairingUnavailable
    case remotePairingMissing
    case remotePairingInvalid
}

// The stock SideStore UI performs AppBootManager.performBootSequence()
// when launched. Scarlet runs SideStore as an embedded LiveProcess, where
// that UI lifecycle must not be assumed to have completed. Before signing,
// check the SAME original Minimuxer transport as PipelineRunner. If needed,
// recover ONLY from the pairing file SideStore already saved; never create
// or replace pairing credentials, prompt in a background extension, or
// disable the normal VPN/endpoint verification.
// V44: keep the proven upstream SideStore signing/installation engine,
// but select the *same RPPairing + RSD* transport used in the successful
// iOS 27.2 SideInstaller installation. A saved lockdown record is not a
// working LocalDevVPN transport on iOS 26.4+.
enum ScarletGuestDeviceTransport {
    static func ensureReady() async throws -> Bool {
        if #available(iOS 26.4, *) {
            let rp = try await ensureRPPairingReady()
            return rp.reconnected
        }
        do {
            try await ensureMinimuxerReady()
            return false
        } catch {
            guard let pairing = PairingFileManager.shared.fetchPairingFile() else {
                throw ScarletGuestRefreshDispatchError.savedPairingUnavailable
            }
            syncMinimuxerBackendFromUserDefaults()
            try await AppBootManager.shared.startMinimuxer(pairingFile: pairing)
            try await ensureMinimuxerReady()
            return true
        }
    }

    // Returns whether an RSD transport was actually reconnected.
    // No key, identifier, pairing contents or device name leaves this worker.
    static func ensureRPPairingReady() async throws
        -> (reconnected: Bool, route: String) {
        let manager = PairingFileManager.shared
        // SideInstaller writes the actual RP record into this original
        // SideStore path; fetchPairingFile() can silently select legacy
        // lockdown if an older preferredProtocol is persisted.
        guard let rp = manager.fetchPairingFile(for: .rppairing),
              !rp.isEmpty else {
            throw ScarletGuestRefreshDispatchError.remotePairingMissing
        }
        let parsed = try manager.parse(content: rp, preferred: .rppairing)
        guard parsed.mode == .rppairing else {
            throw ScarletGuestRefreshDispatchError.remotePairingInvalid
        }

        // A prior successful Minimuxer 'ready' state can be a stale session.
        // Force the real remote device request (not cached Keychain UDID).
        if minimuxerPairingProtocol() == .rppairing {
            do {
                try await ensureMinimuxerReady()
                _ = try await fetchUDID(forceLive: true)
                return (false, "rppairing_rsd_live")
            } catch {
                // Idle tunnels are expected after Apple developer signing.
                // Recover with the EXISTING RP identity; do not re-pair.
            }
        }

        // This is the upstream iDevice gateway, not a new protocol library.
        // Override the active in-memory backend for this connection only;
        // don't overwrite user signing data, pairing files or Apple account.
        selectedGatewayBackendCache = .idevice
        minimuxer.set(MinimuxerParams(
            backend: .idevice,
            remotePairingPort: remotePairingPortCache,
            deviceProbeTimeout: deviceProbeTimeoutCache
        ))
        try await minimuxerStart(rp, preferred: .rppairing)
        try await ensureMinimuxerReady()
        _ = try await fetchUDID(forceLive: true)
        guard minimuxerPairingProtocol() == .rppairing else {
            throw ScarletGuestRefreshDispatchError.remotePairingInvalid
        }
        return (true, "rppairing_rsd_reconnected")
    }

    // Called immediately before AFC/installation_proxy, AFTER upstream
    // signing and provisioning. Re-establish a stale RSD tunnel, as the
    // successful SideInstaller run did after finishing signing.
    static func ensurePostSignInstallTransport() async throws {
        if #available(iOS 26.4, *) {
            _ = try await ensureRPPairingReady()
        } else {
            _ = try await ensureReady()
            _ = try await fetchUDID(forceLive: true)
        }
    }
}

extension ScarletGuestRefreshAgent {
    private static func safeFailureReason(_ error: Error) -> String {
        // Use pinned SideStore's *typed* upstream errors first. The earlier
        // V27 matcher only inspected localizedDescription and collapsed
        // noDevice, invalid pairing, minimuxer-not-started and missing auth
        // into the same native_operation_failed string.
        if let failure = error as? ScarletGuestRefreshDispatchError {
            switch failure {
            case .noOperation: return "native_worker_creation_failed"
            case .savedPairingUnavailable: return "pairing_file_missing"
            case .remotePairingMissing: return "rppairing_file_missing"
            case .remotePairingInvalid: return "rppairing_record_invalid"
            }
        }
        if let original = error as? OperationError {
            switch original {
            case .noInstalledApps: return "no_managed_apps"
            case .notAuthenticated: return "sidestore_auth_missing"
            case .noVPN, .invalidVPN: return "vpn_unavailable"
            case .noConnection: return "device_network_unavailable"
            case .noDevice, .notReachable: return "device_endpoint_unavailable"
            case .invalidPairingFile, .pairingNotComplete:
                return "pairing_invalid"
            case .minimuxerNotStarted: return "minimuxer_not_started"
            case .unknownUDID: return "device_identifier_unavailable"
            case .certificateExpired, .customCertificateExpired:
                return "certificate_expired"
            case .certificateRevoked, .customCertificateRevoked:
                return "certificate_revoked"
            case .missingProvisioningProfile: return "profile_missing"
            case .missingAppGroup: return "app_group_missing"
            case .timedOut: return "device_operation_timed_out"
            case .invalidParameters: return "sidestore_invalid_configuration"
            default: break
            }
        }
        let description = (error as NSError).localizedDescription.lowercased()
        if description.contains("minimuxer") { return "minimuxer_not_started" }
        if description.contains("pairing") { return "pairing_invalid" }
        if description.contains("device endpoint") ||
           description.contains("unable to reach") {
            return "device_endpoint_unavailable"
        }
        if description.contains("not signed in") ||
           description.contains("authentication required") {
            return "sidestore_auth_missing"
        }
        if description.contains("another refresh") ||
           description.contains("in progress") {
            return "already_running"
        }
        if description.contains("anisette") { return "anisette_unavailable" }
        if description.contains("vpn") || description.contains("wireguard") {
            return "vpn_unavailable"
        }
        if description.contains("certificate") || description.contains("signing") {
            return "signing_failed"
        }
        if description.contains("provision") || description.contains("profile") {
            return "profile_refresh_failed"
        }
        if description.contains("server") || description.contains("connection") {
            return "server_unavailable"
        }
        if description.contains("install") { return "installation_failed" }
        return "sidestore_native_operation_failed"
    }

    // R6P23: SideStore's generic code 1 hides which *native* stage failed.
    // Preserve all recognized typed failure reasons verbatim; categorize only
    // the unmatched fallback, never serialize original NSError descriptions.
    private static func stagedFailureReason(_ error: Error, stage: String) -> String {
        let reason = safeFailureReason(error)
        guard reason == "sidestore_native_operation_failed" else { return reason }
        switch stage {
        case "database": return "native_unknown_database_stage"
        case "team": return "native_unknown_team_stage"
        case "transport": return "native_unknown_transport_stage"
        case "refresh": return "native_unknown_appmanager_stage"
        default: return reason
        }
    }

    private static func nativeErrorDomain(_ error: NSError) -> String {
        switch error.domain {
        case OperationError.errorDomain: return "operation"
        case NSURLErrorDomain: return "url"
        case NSCocoaErrorDomain: return "cocoa"
        case NSPOSIXErrorDomain: return "posix"
        case NSOSStatusErrorDomain: return "osstatus"
        case "ALTServerErrorDomain": return "altserver"
        case "ALTSignErrorDomain": return "altsign"
        case "ALTAppleAPIErrorDomain": return "appleapi"
        default: return "other"
        }
    }

    private static func nativeErrorKind(_ error: Error) -> String {
        guard let original = error as? OperationError else { return "other" }
        switch original {
        case .unknownResult: return "unknown_result"
        case .invalidApp: return "invalid_app"
        case .invalidResponse: return "invalid_response"
        case .missingAppBundle: return "missing_app_bundle"
        case .missingInfoPlist: return "missing_info_plist"
        case .missingUpdate: return "missing_update"
        case .forbidden: return "forbidden"
        case .pledgeInactive: return "pledge_inactive"
        case .noSources: return "no_sources"
        case .cacheClearError: return "cache_clear"
        case .openAppFailed: return "open_app_failed"
        case .unableToConnectSideJIT, .unableToRespondSideJITDevice, .SideJITIssue:
            return "sidejit"
        default: return "classified_operation"
        }
    }

    private static func failureReceipt(_ error: Error, stage: String) -> String {
        let original = error as NSError
        var deepest = original
        var depth = 0
        var visited: Set<ObjectIdentifier> = [ObjectIdentifier(original)]
        while depth < 4,
              let next = deepest.userInfo[NSUnderlyingErrorKey] as? NSError,
              visited.insert(ObjectIdentifier(next)).inserted {
            deepest = next
            depth += 1
        }
        func boundedCode(_ value: Int) -> Int {
            max(-2_147_483_648, min(2_147_483_647, value))
        }
        let reason = stagedFailureReason(error, stage: stage)
        let payload: [String: Any] = [
            "v": 1, "reason": reason, "stage": stage,
            "domain": nativeErrorDomain(original), "code": boundedCode(original.code),
            "kind": nativeErrorKind(error),
            "underlyingDomain": depth == 0 ? "none" : nativeErrorDomain(deepest),
            "underlyingCode": depth == 0 ? 0 : boundedCode(deepest.code),
            "underlyingKind": depth == 0 ? "none" : nativeErrorKind(deepest),
            "depth": depth
        ]
        guard let data = try? JSONSerialization.data(withJSONObject: payload),
              let text = String(data: data, encoding: .utf8) else { return reason }
        return "scarlet_native_failure:" + text
    }

    @objc static func perform(_ reply: @escaping (Bool, String) -> Void) {
        performHostBundle("", reply: reply)
    }

    @objc static func performHostBundle(_ hostBundlePath: String,
                                       reply: @escaping (Bool, String) -> Void) {
        performSelectedHostBundle(hostBundlePath, manual: false, reply: reply)
    }

    // P105: Explicit user renewal is NOT the upstream background operation:
    // BackgroundRefreshAppsOperation deliberately skips RUNNING apps and
    // therefore skips the foreground Scarlet host. Keep that behavior for
    // automatic BG renewal, but use upstream AppManager.refresh for the
    // user-tapped action. Both paths still require exact Apple ID + Team,
    // an owned signing key, an actual InstalledApp, and live pairing.
    @objc static func performManualHostBundle(_ hostBundlePath: String,
                                       reply: @escaping (Bool, String) -> Void) {
        performSelectedHostBundle(hostBundlePath, manual: true, reply: reply)
    }

    // P108: An explicit P12 import triggers host-only re-signing, not a
    // CoreData managed-app refresh (which might refresh unrelated apps).
    // This distinct entry uses the native SideStore install/sign pipeline
    // with the currently selected account's pinned P12 and same signed team.
    @objc static func performCertificateRotationHostBundle(_ hostBundlePath: String,
                                       reply: @escaping (Bool, String) -> Void) {
        performSelectedHostBundle(hostBundlePath, manual: true,
            rotateImportedCertificate: true, reply: reply)
    }

    private static func performSelectedHostBundle(_ hostBundlePath: String,
                       manual: Bool, rotateImportedCertificate: Bool = false,
                       reply: @escaping (Bool, String) -> Void) {
        Task {
            let gate = ScarletGuestNativeRefreshGate.shared
            guard await gate.claim() else {
                reply(false, "already_running")
                return
            }
            defer { Task { await gate.release() } }
            guard await ScarletTeamOperationGate.shared.claim() else {
                reply(false, "team_operation_in_progress")
                return
            }
            defer { Task { await ScarletTeamOperationGate.shared.release() } }
            var stage = "database"
            do {
                try await DatabaseManager.shared.start()
                let hostBootstrap = await ScarletGuestHostManagedBootstrap.shared.ensure(hostBundlePath: hostBundlePath)
                let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
                stage = "team"
                guard let owner = await ScarletInstallTeamAuthority.activeOwnership() else {
                    reply(false, "selected_account_owner_unverified"); return
                }
                let selectedTeamID = owner.teamID
                let selectedAppleID = owner.appleID
                if rotateImportedCertificate {
                    // Force a host-only native re-sign even when OTHER
                    // apps exist in this selected account's CoreData.
                    // Installed host must originate from authenticated XPC
                    // path and carry the SAME Team as the selected P12.
                    guard let installedHost = await ScarletGuestHostManagedBootstrap.shared
                        .verifiedSameTeamManualHostURL(
                            hostBundlePath: hostBundlePath, teamID: selectedTeamID) else {
                        reply(false, hostBootstrap == "host_selected_team_mismatch"
                              ? "host_selected_team_mismatch" : "host_signed_bundle_unavailable")
                        return
                    }
                    let prepared = await ScarletInstallTeamAuthority.prepare(
                        teamID: selectedTeamID)
                    guard prepared == "ready" else {
                        reply(false, prepared); return
                    }
                    stage = "transport"
                    _ = try await ScarletGuestDeviceTransport.ensureReady()
                    stage = "refresh"
                    let installed = try await withCheckedThrowingContinuation {
                        (continuation: CheckedContinuation<InstalledApp, Error>) in
                        _ = AppManager.shared.install(.url(installedHost),
                            presentingViewController: nil) { result in
                            continuation.resume(with: result)
                        }
                    }
                    let currentHostID = await ScarletGuestHostManagedBootstrap.shared
                        .verifiedInstalledHostID(hostBundlePath: hostBundlePath)
                    guard let currentHostID,
                          installed.resignedBundleIdentifier == currentHostID,
                          await ScarletInstallTeamAuthority.appBelongsToOwner(
                              installed, owner: owner) else {
                        reply(false, "native_host_reinstall_owner_unverified")
                        return
                    }
                    // AppManager callback is NOT SpringBoard installed
                    // profile readback. The host verifies on next launch.
                    reply(true, "native_host_reinstall_returned_install_unverified")
                    return
                }
                // Native SideStore keeps each InstalledApp associated with the
                // Apple team that signed it. Refresh ONLY the selected team;
                // never re-sign the already installed host using another team.
                let targets = await context.perform {
                    InstalledApp.fetchAppsForRefreshingAll(in: context)
                        .filter {
                            $0.team?.identifier == selectedTeamID &&
                            $0.team?.account?.appleID.caseInsensitiveCompare(
                                selectedAppleID) == .orderedSame
                        }
                }
                guard !targets.isEmpty else {
                    // P106: an externally signed *installed* host has no
                    // SideStore InstalledApp row. The original managed-app
                    // refresh cannot handle an empty list. On the user-tapped
                    // manual action ONLY, sign/install the verified on-device
                    // host through original AppManager.install(.url(.app)).
                    // Never use this for background refresh, for inactive
                    // managed records, or with a foreign host signing Team.
                    let existingReason = await ScarletSelectedTeamRefreshInventory
                        .zeroTargetReason(in: context, teamID: selectedTeamID,
                                          appleID: selectedAppleID)
                    if manual, existingReason == nil,
                       let installedHost = await ScarletGuestHostManagedBootstrap.shared
                           .verifiedSameTeamManualHostURL(
                               hostBundlePath: hostBundlePath,
                               teamID: selectedTeamID),
                       await ScarletInstallTeamAuthority.hasValidSigningCertificate(
                           for: selectedTeamID) {
                        let teamState = await ScarletInstallTeamAuthority.prepare(
                            teamID: selectedTeamID)
                        guard teamState == "ready" else {
                            reply(false, teamState); return
                        }
                        stage = "transport"
                        _ = try await ScarletGuestDeviceTransport.ensureReady()
                        stage = "refresh"
                        let installed = try await withCheckedThrowingContinuation {
                            (continuation: CheckedContinuation<InstalledApp, Error>) in
                            _ = AppManager.shared.install(.url(installedHost),
                                presentingViewController: nil) { result in
                                continuation.resume(with: result)
                            }
                        }
                        let verifiedHostID = await ScarletGuestHostManagedBootstrap.shared
                            .verifiedInstalledHostID(hostBundlePath: hostBundlePath)
                        guard let verifiedHostID,
                              installed.resignedBundleIdentifier == verifiedHostID,
                              await ScarletInstallTeamAuthority.appBelongsToOwner(
                                  installed, owner: owner) else {
                            reply(false, "native_host_reinstall_owner_unverified")
                            return
                        }
                        // Callback does not prove SpringBoard committed the
                        // updated host profile. The app may be relaunched.
                        reply(true, "native_host_reinstall_returned_install_unverified")
                        return
                    }
                    reply(false, existingReason ?? (
                        hostBootstrap == "host_selected_team_mismatch"
                            ? "host_selected_team_mismatch"
                            : "no_managed_apps_for_selected_team"))
                    return
                }
                let signedHostID = await ScarletGuestHostManagedBootstrap.shared
                    .verifiedInstalledHostID(hostBundlePath: hostBundlePath)
                let managedHost = signedHostID.flatMap { verified in
                    targets.first { $0.resignedBundleIdentifier == verified }
                }
                let before = await context.perform {
                    (managedHost?.refreshedDate, managedHost?.expirationDate)
                }
                let teamState = await ScarletInstallTeamAuthority.prepare(
                    teamID: selectedTeamID)
                guard teamState == "ready" else {
                    reply(false, teamState); return
                }
                // SideStore's original AppBootManager starts Minimuxer using
                // its already-imported pairing. Embedded LiveProcess may
                // have skipped the SideStore UIApplication launch lifecycle.
                // Restore only the original transport, then re-check its
                // own preconditions. A failed check aborts before signing.
                stage = "transport"
                let recoveredTransport = try await
                    ScarletGuestDeviceTransport.ensureReady()
                // Original SideStore refresh operation, not an AppIntent
                // proxy and not a fake successful "request accepted" flag.
                stage = "refresh"
                let results: [String: Result<InstalledApp, Error>]
                if manual {
                    // Original SideStore interactive pipeline; never fake
                    // successful refresh by bypassing its signing/installer.
                    // The BACKGROUND operation suppresses the foreground host.
                    results = await withCheckedContinuation { continuation in
                        let group = AppManager.shared.refresh(
                            targets, presentingViewController: nil)
                        group.completionHandler = { result in
                            continuation.resume(returning: result)
                        }
                    }
                } else {
                    results = try await withCheckedThrowingContinuation { continuation in
                        do {
                            let operation = try AppManager.shared.backgroundRefresh(
                                targets, presentsNotifications: false
                            ) { result in
                                continuation.resume(with: result)
                            }
                            operation.ignoresServerNotFoundError = false
                        } catch {
                            continuation.resume(throwing: error)
                        }
                    }
                }
                for (_, result) in results {
                    if case let .failure(error) = result {
                        reply(false, failureReceipt(error, stage: stage))
                        return
                    }
                }
                // Original SideStore skips apps it sees as running. A
                // completed operation may contain results for other apps,
                // or no results at all. Never call that a Scarlet refresh.
                // A selected team B must not attempt to re-sign Scarlet's
                // host A. For the host team, preserve the stricter readback.
                if let managedHost {
                    guard let hostResult = results[managedHost.bundleIdentifier] else {
                        reply(false, "host_not_in_refresh_results")
                        return
                    }
                    if case let .success(refreshedHost) = hostResult,
                       let resultContext = refreshedHost.managedObjectContext {
                        let after = await resultContext.perform {
                            (refreshedHost.refreshedDate, refreshedHost.expirationDate)
                        }
                        if let beforeDate = before.0, let beforeExpiry = before.1,
                           after.0 > beforeDate {
                            reply(true, after.1 > beforeExpiry
                                ? "native_profile_date_advanced_install_unverified"
                                : "native_refresh_record_updated_install_unverified")
                            return
                        }
                    }
                    reply(true, recoveredTransport
                        ? "native_operation_returned_unverified_transport_recovered"
                        : "native_operation_returned_unverified")
                } else {
                    guard !results.isEmpty else {
                        reply(false, "selected_team_not_in_refresh_results"); return
                    }
                    reply(true, "native_selected_team_refresh_returned_install_unverified")
                }
            } catch {
                reply(false, failureReceipt(error, stage: stage))
            }
        }
    }
}

// V31: read back the saved *original SideStore* pairing record on every
// fresh Scarlet session. Do not treat a previously displayed checkmark or
// successful import receipt as proof that Minimuxer is still connected.
// The record and the underlying error never leave the SideStore process.
extension ScarletGuestRefreshAgent {
    @objc static func inspectDevicePairing(
        _ reply: @escaping (Bool, String) -> Void
    ) {
        Task {
            let pairing = PairingFileManager.shared
            let saved: String?
            if #available(iOS 26.4, *) {
                saved = pairing.fetchPairingFile(for: .rppairing)
            } else {
                saved = pairing.fetchPairingFile()
            }
            guard let saved, !saved.isEmpty else {
                reply(false, "pairing_file_missing")
                return
            }
            do {
                let parsed = try pairing.parse(content: saved, preferred: .rppairing)
                guard parsed.mode == .rppairing else {
                    reply(false, "pairing_protocol_mismatch")
                    return
                }
            } catch {
                reply(false, "pairing_invalid")
                return
            }
            // R6P15: a guest/host foreground handoff may temporarily close
            // the RSD tunnel after a prior successful inspection. The pinned
            // ensureReady() reconnects from the SAME saved RP record and
            // performs a fresh UDID transport probe, so retry one bounded
            // time without clearing credentials or implying a false success.
            do {
                _ = try await ScarletGuestDeviceTransport.ensureReady()
                reply(true, "pairing_saved_and_transport_ready")
            } catch {
                // Wait only for a transient launch/foreground handoff; do not
                // spin when VPN is disabled or a device is unreachable.
                try? await Task.sleep(nanoseconds: 1_100_000_000)
                do {
                    _ = try await ScarletGuestDeviceTransport.ensureReady()
                    reply(true, "pairing_saved_and_transport_ready")
                } catch {
                    // A valid saved file is not live connectivity. Keep it,
                    // require real RSD transport before asserting verified.
                    reply(false, "pairing_saved_transport_unavailable")
                }
            }
        }
    }
}

// V29: in-app iOS 27 RPPairing passes only an in-memory on-device record.
// The record is sensitive: never copy it into the JSON diagnostic journal,
// source control, pasteboard or remote server. Check the *original* pinned
// SideStore parser and live Minimuxer before atomically changing saved state.
extension ScarletGuestRefreshAgent {
    @objc static func importDevicePairing(_ payload: Data,
        reply: @escaping (Bool, String) -> Void) {
        Task {
            guard payload.count > 128 && payload.count <= 1024 * 1024,
                  let candidate = String(data: payload, encoding: .utf8) else {
                reply(false, "pairing_file_invalid")
                return
            }
            let pairingManager = PairingFileManager.shared
            do {
                let parsed = try pairingManager.parse(
                    content: candidate, preferred: .rppairing)
                guard parsed.mode == .rppairing else {
                    reply(false, "pairing_protocol_mismatch")
                    return
                }
                // Test the new identity while it is IN MEMORY. The saved
                // pairing file, settings, user auth and old cert are intact.
                try await AppBootManager.shared.startMinimuxer(
                    pairingFile: candidate)
                try await ensureMinimuxerReady()
                // Only a confirmed usable identity is committed.
                try pairingManager.savePairingFile(
                    contents: candidate, preferred: .rppairing)
                pairingManager.persistedActiveProtocol = .rppairing
                reply(true, "pairing_saved_and_transport_ready")
            } catch {
                // A failed candidate MUST NOT overwrite a working old file.
                // Recover previous transport when an old record exists.
                if let previous = pairingManager.fetchPairingFile(),
                   previous != candidate {
                    try? await AppBootManager.shared.startMinimuxer(
                        pairingFile: previous)
                }
                reply(false, safeFailureReason(error))
            }
        }
    }
}
