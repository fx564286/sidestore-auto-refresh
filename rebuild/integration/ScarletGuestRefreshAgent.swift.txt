// Runs inside the REAL embedded SideStore LiveProcess, where SideStore's
// CoreData database and CertificateManager live. Scarlet's host cannot infer
// whether its SideInstaller-installed signed bundle is managed by this DB.
@objc(ScarletGuestRefreshAgent)
final class ScarletGuestRefreshAgent: NSObject {
    @objc static func inspect(_ reply: @escaping (Int, Bool, Bool, String) -> Void) {
        Task {
            do {
                // SideStore's original start() prepares/reconciles its own
                // installed host record; this is not a fake copy of a DB.
                try await DatabaseManager.shared.start()
                let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
                let targets = await context.perform {
                    InstalledApp.fetchAppsForRefreshingAll(in: context)
                }
                // A Personal Team signer may suffix the original bundle ID.
                // Compare the actual signed bundle, NOT the hard-coded
                // com.kdt.livecontainer StoreApp.altstoreAppID.
                let realHostID = Bundle.Info.activeBundleIdentifier
                let managesHost = targets.contains {
                    $0.resignedBundleIdentifier == realHostID
                }
                let signingReady = CertificateManager.shared.activeCertificate != nil
                let reason: String
                if !managesHost {
                    reason = targets.isEmpty ? "no_managed_apps" : "host_not_managed"
                } else if !signingReady {
                    reason = "sidestore_certificate_missing"
                } else {
                    reason = "ready"
                }
                reply(min(targets.count, 99), managesHost, signingReady, reason)
            } catch {
                // Never send account info, DB URLs or arbitrary error strings.
                reply(0, false, false, "sidestore_database_unavailable")
            }
        }
    }
}

// Refresh exactly the apps selected by the pinned SideStore's OWN installed-app
// database, using its OWN AppManager/BackgroundRefreshAppsOperation inside
// LiveProcess. This avoids wrapping the operation through PrivateIntentRunner,
// which collapses its failures into SideStore domain/code 1 at the LC host.
// No custom signing, provisioning, installation, server or bypass logic.
private actor ScarletGuestNativeRefreshGate {
    static let shared = ScarletGuestNativeRefreshGate()
    private var active = false

    func claim() -> Bool {
        guard !active else { return false }
        active = true
        return true
    }

    func release() { active = false }
}

private enum ScarletGuestRefreshDispatchError: Error {
    case noOperation
    case savedPairingUnavailable
    case remotePairingMissing
    case remotePairingInvalid
}

// The stock SideStore UI performs AppBootManager.performBootSequence()
// when launched. Scarlet runs SideStore as an embedded LiveProcess, where
// that UI lifecycle must not be assumed to have completed. Before signing,
// check the SAME original Minimuxer transport as PipelineRunner. If needed,
// recover ONLY from the pairing file SideStore already saved; never create
// or replace pairing credentials, prompt in a background extension, or
// disable the normal VPN/endpoint verification.
// V44: keep the proven upstream SideStore signing/installation engine,
// but select the *same RPPairing + RSD* transport used in the successful
// iOS 27.2 SideInstaller installation. A saved lockdown record is not a
// working LocalDevVPN transport on iOS 26.4+.
enum ScarletGuestDeviceTransport {
    static func ensureReady() async throws -> Bool {
        if #available(iOS 26.4, *) {
            let rp = try await ensureRPPairingReady()
            return rp.reconnected
        }
        do {
            try await ensureMinimuxerReady()
            return false
        } catch {
            guard let pairing = PairingFileManager.shared.fetchPairingFile() else {
                throw ScarletGuestRefreshDispatchError.savedPairingUnavailable
            }
            syncMinimuxerBackendFromUserDefaults()
            try await AppBootManager.shared.startMinimuxer(pairingFile: pairing)
            try await ensureMinimuxerReady()
            return true
        }
    }

    // Returns whether an RSD transport was actually reconnected.
    // No key, identifier, pairing contents or device name leaves this worker.
    static func ensureRPPairingReady() async throws
        -> (reconnected: Bool, route: String) {
        let manager = PairingFileManager.shared
        // SideInstaller writes the actual RP record into this original
        // SideStore path; fetchPairingFile() can silently select legacy
        // lockdown if an older preferredProtocol is persisted.
        guard let rp = manager.fetchPairingFile(for: .rppairing),
              !rp.isEmpty else {
            throw ScarletGuestRefreshDispatchError.remotePairingMissing
        }
        let parsed = try manager.parse(content: rp, preferred: .rppairing)
        guard parsed.mode == .rppairing else {
            throw ScarletGuestRefreshDispatchError.remotePairingInvalid
        }

        // A prior successful Minimuxer 'ready' state can be a stale session.
        // Force the real remote device request (not cached Keychain UDID).
        if minimuxerPairingProtocol() == .rppairing {
            do {
                try await ensureMinimuxerReady()
                _ = try await fetchUDID(forceLive: true)
                return (false, "rppairing_rsd_live")
            } catch {
                // Idle tunnels are expected after Apple developer signing.
                // Recover with the EXISTING RP identity; do not re-pair.
            }
        }

        // This is the upstream iDevice gateway, not a new protocol library.
        // Override the active in-memory backend for this connection only;
        // don't overwrite user signing data, pairing files or Apple account.
        selectedGatewayBackendCache = .idevice
        minimuxer.set(MinimuxerParams(
            backend: .idevice,
            remotePairingPort: remotePairingPortCache,
            deviceProbeTimeout: deviceProbeTimeoutCache
        ))
        try await minimuxerStart(rp, preferred: .rppairing)
        try await ensureMinimuxerReady()
        _ = try await fetchUDID(forceLive: true)
        guard minimuxerPairingProtocol() == .rppairing else {
            throw ScarletGuestRefreshDispatchError.remotePairingInvalid
        }
        return (true, "rppairing_rsd_reconnected")
    }

    // Called immediately before AFC/installation_proxy, AFTER upstream
    // signing and provisioning. Re-establish a stale RSD tunnel, as the
    // successful SideInstaller run did after finishing signing.
    static func ensurePostSignInstallTransport() async throws {
        if #available(iOS 26.4, *) {
            _ = try await ensureRPPairingReady()
        } else {
            _ = try await ensureReady()
            _ = try await fetchUDID(forceLive: true)
        }
    }
}

extension ScarletGuestRefreshAgent {
    private static func safeFailureReason(_ error: Error) -> String {
        // Use pinned SideStore's *typed* upstream errors first. The earlier
        // V27 matcher only inspected localizedDescription and collapsed
        // noDevice, invalid pairing, minimuxer-not-started and missing auth
        // into the same native_operation_failed string.
        if let failure = error as? ScarletGuestRefreshDispatchError {
            switch failure {
            case .noOperation: return "native_worker_creation_failed"
            case .savedPairingUnavailable: return "pairing_file_missing"
            case .remotePairingMissing: return "rppairing_file_missing"
            case .remotePairingInvalid: return "rppairing_record_invalid"
            }
        }
        if let original = error as? OperationError {
            switch original {
            case .noInstalledApps: return "no_managed_apps"
            case .notAuthenticated: return "sidestore_auth_missing"
            case .noVPN, .invalidVPN: return "vpn_unavailable"
            case .noConnection: return "device_network_unavailable"
            case .noDevice, .notReachable: return "device_endpoint_unavailable"
            case .invalidPairingFile, .pairingNotComplete:
                return "pairing_invalid"
            case .minimuxerNotStarted: return "minimuxer_not_started"
            case .unknownUDID: return "device_identifier_unavailable"
            case .certificateExpired, .customCertificateExpired:
                return "certificate_expired"
            case .certificateRevoked, .customCertificateRevoked:
                return "certificate_revoked"
            case .missingProvisioningProfile: return "profile_missing"
            case .missingAppGroup: return "app_group_missing"
            case .timedOut: return "device_operation_timed_out"
            case .invalidParameters: return "sidestore_invalid_configuration"
            default: break
            }
        }
        let description = (error as NSError).localizedDescription.lowercased()
        if description.contains("minimuxer") { return "minimuxer_not_started" }
        if description.contains("pairing") { return "pairing_invalid" }
        if description.contains("device endpoint") ||
           description.contains("unable to reach") {
            return "device_endpoint_unavailable"
        }
        if description.contains("not signed in") ||
           description.contains("authentication required") {
            return "sidestore_auth_missing"
        }
        if description.contains("another refresh") ||
           description.contains("in progress") {
            return "already_running"
        }
        if description.contains("anisette") { return "anisette_unavailable" }
        if description.contains("vpn") || description.contains("wireguard") {
            return "vpn_unavailable"
        }
        if description.contains("certificate") || description.contains("signing") {
            return "signing_failed"
        }
        if description.contains("provision") || description.contains("profile") {
            return "profile_refresh_failed"
        }
        if description.contains("server") || description.contains("connection") {
            return "server_unavailable"
        }
        if description.contains("install") { return "installation_failed" }
        return "sidestore_native_operation_failed"
    }

    // R6P23: SideStore's generic code 1 hides which *native* stage failed.
    // Preserve all recognized typed failure reasons verbatim; categorize only
    // the unmatched fallback, never serialize original NSError descriptions.
    private static func stagedFailureReason(_ error: Error, stage: String) -> String {
        let reason = safeFailureReason(error)
        guard reason == "sidestore_native_operation_failed" else { return reason }
        switch stage {
        case "database": return "native_unknown_database_stage"
        case "team": return "native_unknown_team_stage"
        case "transport": return "native_unknown_transport_stage"
        case "refresh": return "native_unknown_appmanager_stage"
        default: return reason
        }
    }

    private static func nativeErrorDomain(_ error: NSError) -> String {
        switch error.domain {
        case OperationError.errorDomain: return "operation"
        case NSURLErrorDomain: return "url"
        case NSCocoaErrorDomain: return "cocoa"
        case NSPOSIXErrorDomain: return "posix"
        case NSOSStatusErrorDomain: return "osstatus"
        case "ALTServerErrorDomain": return "altserver"
        case "ALTSignErrorDomain": return "altsign"
        case "ALTAppleAPIErrorDomain": return "appleapi"
        default: return "other"
        }
    }

    private static func nativeErrorKind(_ error: Error) -> String {
        guard let original = error as? OperationError else { return "other" }
        switch original {
        case .unknownResult: return "unknown_result"
        case .invalidApp: return "invalid_app"
        case .invalidResponse: return "invalid_response"
        case .missingAppBundle: return "missing_app_bundle"
        case .missingInfoPlist: return "missing_info_plist"
        case .missingUpdate: return "missing_update"
        case .forbidden: return "forbidden"
        case .pledgeInactive: return "pledge_inactive"
        case .noSources: return "no_sources"
        case .cacheClearError: return "cache_clear"
        case .openAppFailed: return "open_app_failed"
        case .unableToConnectSideJIT, .unableToRespondSideJITDevice, .SideJITIssue:
            return "sidejit"
        default: return "classified_operation"
        }
    }

    private static func failureReceipt(_ error: Error, stage: String) -> String {
        let original = error as NSError
        var deepest = original
        var depth = 0
        var visited: Set<ObjectIdentifier> = [ObjectIdentifier(original)]
        while depth < 4,
              let next = deepest.userInfo[NSUnderlyingErrorKey] as? NSError,
              visited.insert(ObjectIdentifier(next)).inserted {
            deepest = next
            depth += 1
        }
        func boundedCode(_ value: Int) -> Int {
            max(-2_147_483_648, min(2_147_483_647, value))
        }
        let reason = stagedFailureReason(error, stage: stage)
        let payload: [String: Any] = [
            "v": 1, "reason": reason, "stage": stage,
            "domain": nativeErrorDomain(original), "code": boundedCode(original.code),
            "kind": nativeErrorKind(error),
            "underlyingDomain": depth == 0 ? "none" : nativeErrorDomain(deepest),
            "underlyingCode": depth == 0 ? 0 : boundedCode(deepest.code),
            "underlyingKind": depth == 0 ? "none" : nativeErrorKind(deepest),
            "depth": depth
        ]
        guard let data = try? JSONSerialization.data(withJSONObject: payload),
              let text = String(data: data, encoding: .utf8) else { return reason }
        return "scarlet_native_failure:" + text
    }

    @objc static func perform(_ reply: @escaping (Bool, String) -> Void) {
        Task {
            let gate = ScarletGuestNativeRefreshGate.shared
            guard await gate.claim() else {
                reply(false, "already_running")
                return
            }
            defer { Task { await gate.release() } }
            var stage = "database"
            do {
                try await DatabaseManager.shared.start()
                let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
                let targets = await context.perform {
                    InstalledApp.fetchAppsForRefreshingAll(in: context)
                }
                let signedHostID = Bundle.Info.activeBundleIdentifier
                guard let managedHost = targets.first(where: {
                    $0.resignedBundleIdentifier == signedHostID
                }) else {
                    reply(false, targets.isEmpty ? "no_managed_apps" : "host_not_managed")
                    return
                }
                guard CertificateManager.shared.activeCertificate != nil else {
                    reply(false, "sidestore_certificate_missing")
                    return
                }
                // Save pre-refresh metadata before the original background
                // operation mutates its own CoreData record.
                let before = await context.perform {
                    (managedHost.refreshedDate, managedHost.expirationDate)
                }
                // V30: Authentic Apple token restoration is not equivalent
                // to SideStore's active Team record. The stock login pipeline
                // writes that record, but Scarlet's headless login previously
                // did not. Recover only the exact Apple-verified existing
                // signed team before running original PreflightChecksOperation.
                stage = "team"
                let teamState = await ScarletGuestSignInAgent
                    .ensureActiveTeamForRefresh()
                guard teamState == "ready" || teamState == "team_restored" else {
                    reply(false, teamState)
                    return
                }
                // SideStore's original AppBootManager starts Minimuxer using
                // its already-imported pairing. Embedded LiveProcess may
                // have skipped the SideStore UIApplication launch lifecycle.
                // Restore only the original transport, then re-check its
                // own preconditions. A failed check aborts before signing.
                stage = "transport"
                let recoveredTransport = try await
                    ScarletGuestDeviceTransport.ensureReady()
                // Original SideStore refresh operation, not an AppIntent
                // proxy and not a fake successful "request accepted" flag.
                stage = "refresh"
                let results: [String: Result<InstalledApp, Error>] =
                    try await withCheckedThrowingContinuation { continuation in
                    do {
                        let operation = try AppManager.shared.backgroundRefresh(
                            targets, presentsNotifications: false
                        ) { result in
                            continuation.resume(with: result)
                        }
                        operation.ignoresServerNotFoundError = false
                    } catch {
                        // Preserve the actual synchronous dispatch error.
                        continuation.resume(throwing: error)
                    }
                }
                for (_, result) in results {
                    if case let .failure(error) = result {
                        reply(false, failureReceipt(error, stage: stage))
                        return
                    }
                }
                // Original SideStore skips apps it sees as running. A
                // completed operation may contain results for other apps,
                // or no results at all. Never call that a Scarlet refresh.
                guard let hostResult = results[managedHost.bundleIdentifier] else {
                    reply(false, "host_not_in_refresh_results")
                    return
                }
                // SideStore returns success for the host, but that does not
                // prove iOS SpringBoard switched to a renewed installation.
                // Read its own post-operation CoreData record in the owning
                // context. Never return a false "installed" receipt.
                if case let .success(refreshedHost) = hostResult,
                   let resultContext = refreshedHost.managedObjectContext {
                    let after = await resultContext.perform {
                        (refreshedHost.refreshedDate, refreshedHost.expirationDate)
                    }
                    if after.0 > before.0 {
                        reply(true, after.1 > before.1
                            ? "native_profile_date_advanced_install_unverified"
                            : "native_refresh_record_updated_install_unverified")
                        return
                    }
                }
                reply(true, recoveredTransport
                    ? "native_operation_returned_unverified_transport_recovered"
                    : "native_operation_returned_unverified")
            } catch {
                reply(false, failureReceipt(error, stage: stage))
            }
        }
    }
}

// V31: read back the saved *original SideStore* pairing record on every
// fresh Scarlet session. Do not treat a previously displayed checkmark or
// successful import receipt as proof that Minimuxer is still connected.
// The record and the underlying error never leave the SideStore process.
extension ScarletGuestRefreshAgent {
    @objc static func inspectDevicePairing(
        _ reply: @escaping (Bool, String) -> Void
    ) {
        Task {
            let pairing = PairingFileManager.shared
            let saved: String?
            if #available(iOS 26.4, *) {
                saved = pairing.fetchPairingFile(for: .rppairing)
            } else {
                saved = pairing.fetchPairingFile()
            }
            guard let saved, !saved.isEmpty else {
                reply(false, "pairing_file_missing")
                return
            }
            do {
                let parsed = try pairing.parse(content: saved, preferred: .rppairing)
                guard parsed.mode == .rppairing else {
                    reply(false, "pairing_protocol_mismatch")
                    return
                }
            } catch {
                reply(false, "pairing_invalid")
                return
            }
            // R6P15: a guest/host foreground handoff may temporarily close
            // the RSD tunnel after a prior successful inspection. The pinned
            // ensureReady() reconnects from the SAME saved RP record and
            // performs a fresh UDID transport probe, so retry one bounded
            // time without clearing credentials or implying a false success.
            do {
                _ = try await ScarletGuestDeviceTransport.ensureReady()
                reply(true, "pairing_saved_and_transport_ready")
            } catch {
                // Wait only for a transient launch/foreground handoff; do not
                // spin when VPN is disabled or a device is unreachable.
                try? await Task.sleep(nanoseconds: 1_100_000_000)
                do {
                    _ = try await ScarletGuestDeviceTransport.ensureReady()
                    reply(true, "pairing_saved_and_transport_ready")
                } catch {
                    // A valid saved file is not live connectivity. Keep it,
                    // require real RSD transport before asserting verified.
                    reply(false, "pairing_saved_transport_unavailable")
                }
            }
        }
    }
}

// V29: in-app iOS 27 RPPairing passes only an in-memory on-device record.
// The record is sensitive: never copy it into the JSON diagnostic journal,
// source control, pasteboard or remote server. Check the *original* pinned
// SideStore parser and live Minimuxer before atomically changing saved state.
extension ScarletGuestRefreshAgent {
    @objc static func importDevicePairing(_ payload: Data,
        reply: @escaping (Bool, String) -> Void) {
        Task {
            guard payload.count > 128 && payload.count <= 1024 * 1024,
                  let candidate = String(data: payload, encoding: .utf8) else {
                reply(false, "pairing_file_invalid")
                return
            }
            let pairingManager = PairingFileManager.shared
            do {
                let parsed = try pairingManager.parse(
                    content: candidate, preferred: .rppairing)
                guard parsed.mode == .rppairing else {
                    reply(false, "pairing_protocol_mismatch")
                    return
                }
                // Test the new identity while it is IN MEMORY. The saved
                // pairing file, settings, user auth and old cert are intact.
                try await AppBootManager.shared.startMinimuxer(
                    pairingFile: candidate)
                try await ensureMinimuxerReady()
                // Only a confirmed usable identity is committed.
                try pairingManager.savePairingFile(
                    contents: candidate, preferred: .rppairing)
                pairingManager.persistedActiveProtocol = .rppairing
                reply(true, "pairing_saved_and_transport_ready")
            } catch {
                // A failed candidate MUST NOT overwrite a working old file.
                // Recover previous transport when an old record exists.
                if let previous = pairingManager.fetchPairingFile(),
                   previous != candidate {
                    try? await AppBootManager.shared.startMinimuxer(
                        pairingFile: previous)
                }
                reply(false, safeFailureReason(error))
            }
        }
    }
}
