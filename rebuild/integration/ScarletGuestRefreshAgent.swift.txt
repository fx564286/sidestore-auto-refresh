// Runs inside the REAL embedded SideStore LiveProcess, where SideStore's
// CoreData database and CertificateManager live. Scarlet's host cannot infer
// whether its SideInstaller-installed signed bundle is managed by this DB.
@objc(ScarletGuestRefreshAgent)
final class ScarletGuestRefreshAgent: NSObject {
    @objc static func inspect(_ reply: @escaping (Int, Bool, Bool, String) -> Void) {
        Task {
            do {
                // SideStore's original start() prepares/reconciles its own
                // installed host record; this is not a fake copy of a DB.
                try await DatabaseManager.shared.start()
                let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
                let targets = await context.perform {
                    InstalledApp.fetchAppsForRefreshingAll(in: context)
                }
                // A Personal Team signer may suffix the original bundle ID.
                // Compare the actual signed bundle, NOT the hard-coded
                // com.kdt.livecontainer StoreApp.altstoreAppID.
                let realHostID = Bundle.Info.activeBundleIdentifier
                let managesHost = targets.contains {
                    $0.resignedBundleIdentifier == realHostID
                }
                let signingReady = CertificateManager.shared.activeCertificate != nil
                let reason: String
                if !managesHost {
                    reason = targets.isEmpty ? "no_managed_apps" : "host_not_managed"
                } else if !signingReady {
                    reason = "sidestore_certificate_missing"
                } else {
                    // V29 reported 'ready' using only app/certificate
                    // checks while native refresh immediately threw
                    // OperationError.notAuthenticated. The pinned original
                    // SideStore preflight ALSO needs active Account/Team.
                    let state = await ScarletGuestAuthTeamRepair.ensure()
                    reason = state == "auth_team_ready" ? "ready"
                        : state == "auth_team_restored"
                        ? "ready_auth_team_restored" : state
                }
                reply(min(targets.count, 99), managesHost, signingReady, reason)
            } catch {
                // Never send account info, DB URLs or arbitrary error strings.
                reply(0, false, false, "sidestore_database_unavailable")
            }
        }
    }
}

// Refresh exactly the apps selected by the pinned SideStore's OWN installed-app
// database, using its OWN AppManager/BackgroundRefreshAppsOperation inside
// LiveProcess. This avoids wrapping the operation through PrivateIntentRunner,
// which collapses its failures into SideStore domain/code 1 at the LC host.
// No custom signing, provisioning, installation, server or bypass logic.
private actor ScarletGuestNativeRefreshGate {
    static let shared = ScarletGuestNativeRefreshGate()
    private var active = false

    func claim() -> Bool {
        guard !active else { return false }
        active = true
        return true
    }

    func release() { active = false }
}

private enum ScarletGuestRefreshDispatchError: Error {
    case noOperation
    case savedPairingUnavailable
}

// The stock SideStore UI performs AppBootManager.performBootSequence()
// when launched. Scarlet runs SideStore as an embedded LiveProcess, where
// that UI lifecycle must not be assumed to have completed. Before signing,
// check the SAME original Minimuxer transport as PipelineRunner. If needed,
// recover ONLY from the pairing file SideStore already saved; never create
// or replace pairing credentials, prompt in a background extension, or
// disable the normal VPN/endpoint verification.
private enum ScarletGuestDeviceTransport {
    static func ensureReady() async throws -> Bool {
        do {
            try await ensureMinimuxerReady()
            return false
        } catch {
            guard let pairing = PairingFileManager.shared.fetchPairingFile() else {
                throw ScarletGuestRefreshDispatchError.savedPairingUnavailable
            }
            // Original SideStore backend and boot path, not a parallel
            // custom USB/provisioning implementation.
            syncMinimuxerBackendFromUserDefaults()
            try await AppBootManager.shared.startMinimuxer(pairingFile: pairing)
            try await ensureMinimuxerReady()
            return true
        }
    }
}

extension ScarletGuestRefreshAgent {
    private static func safeFailureReason(_ error: Error) -> String {
        // Use pinned SideStore's *typed* upstream errors first. The earlier
        // V27 matcher only inspected localizedDescription and collapsed
        // noDevice, invalid pairing, minimuxer-not-started and missing auth
        // into the same native_operation_failed string.
        if let failure = error as? ScarletGuestRefreshDispatchError {
            switch failure {
            case .noOperation: return "native_worker_creation_failed"
            case .savedPairingUnavailable: return "pairing_file_missing"
            }
        }
        if let original = error as? OperationError {
            switch original {
            case .noInstalledApps: return "no_managed_apps"
            case .notAuthenticated: return "sidestore_auth_missing"
            case .noVPN, .invalidVPN: return "vpn_unavailable"
            case .noConnection: return "device_network_unavailable"
            case .noDevice, .notReachable: return "device_endpoint_unavailable"
            case .invalidPairingFile, .pairingNotComplete:
                return "pairing_invalid"
            case .minimuxerNotStarted: return "minimuxer_not_started"
            case .unknownUDID: return "device_identifier_unavailable"
            case .certificateExpired, .customCertificateExpired:
                return "certificate_expired"
            case .certificateRevoked, .customCertificateRevoked:
                return "certificate_revoked"
            case .missingProvisioningProfile: return "profile_missing"
            case .missingAppGroup: return "app_group_missing"
            case .timedOut: return "device_operation_timed_out"
            case .invalidParameters: return "sidestore_invalid_configuration"
            default: break
            }
        }
        let description = (error as NSError).localizedDescription.lowercased()
        if description.contains("minimuxer") { return "minimuxer_not_started" }
        if description.contains("pairing") { return "pairing_invalid" }
        if description.contains("device endpoint") ||
           description.contains("unable to reach") {
            return "device_endpoint_unavailable"
        }
        if description.contains("not signed in") ||
           description.contains("authentication required") {
            return "sidestore_auth_missing"
        }
        if description.contains("another refresh") ||
           description.contains("in progress") {
            return "already_running"
        }
        if description.contains("anisette") { return "anisette_unavailable" }
        if description.contains("vpn") || description.contains("wireguard") {
            return "vpn_unavailable"
        }
        if description.contains("certificate") || description.contains("signing") {
            return "signing_failed"
        }
        if description.contains("provision") || description.contains("profile") {
            return "profile_refresh_failed"
        }
        if description.contains("server") || description.contains("connection") {
            return "server_unavailable"
        }
        if description.contains("install") { return "installation_failed" }
        return "sidestore_native_operation_failed"
    }

    @objc static func perform(_ reply: @escaping (Bool, String) -> Void) {
        Task {
            let gate = ScarletGuestNativeRefreshGate.shared
            guard await gate.claim() else {
                reply(false, "already_running")
                return
            }
            defer { Task { await gate.release() } }
            do {
                try await DatabaseManager.shared.start()
                let context = DatabaseManager.shared.persistentContainer.newBackgroundContext()
                let targets = await context.perform {
                    InstalledApp.fetchAppsForRefreshingAll(in: context)
                }
                let signedHostID = Bundle.Info.activeBundleIdentifier
                guard let managedHost = targets.first(where: {
                    $0.resignedBundleIdentifier == signedHostID
                }) else {
                    reply(false, targets.isEmpty ? "no_managed_apps" : "host_not_managed")
                    return
                }
                guard CertificateManager.shared.activeCertificate != nil else {
                    reply(false, "sidestore_certificate_missing")
                    return
                }
                let authState = await ScarletGuestAuthTeamRepair.ensure()
                guard authState == "auth_team_ready" ||
                      authState == "auth_team_restored" else {
                    reply(false, authState)
                    return
                }
                // SideStore's original AppBootManager starts Minimuxer using
                // its already-imported pairing. Embedded LiveProcess may
                // have skipped the SideStore UIApplication launch lifecycle.
                // Restore only the original transport, then re-check its
                // own preconditions. A failed check aborts before signing.
                let recoveredTransport = try await
                    ScarletGuestDeviceTransport.ensureReady()
                // Original SideStore refresh operation, not an AppIntent
                // proxy and not a fake successful "request accepted" flag.
                let results: [String: Result<InstalledApp, Error>] =
                    try await withCheckedThrowingContinuation { continuation in
                    let operation = try? AppManager.shared.backgroundRefresh(
                        targets, presentsNotifications: false
                    ) { result in
                        continuation.resume(with: result)
                    }
                    guard let operation else {
                        continuation.resume(throwing: ScarletGuestRefreshDispatchError.noOperation)
                        return
                    }
                    operation.ignoresServerNotFoundError = false
                }
                for (_, result) in results {
                    if case let .failure(error) = result {
                        reply(false, safeFailureReason(error))
                        return
                    }
                }
                // Original SideStore skips apps it sees as running. A
                // completed operation may contain results for other apps,
                // or no results at all. Never call that a Scarlet refresh.
                guard results[managedHost.bundleIdentifier] != nil else {
                    reply(false, "host_not_in_refresh_results")
                    return
                }
                // An operation returning successfully does not attest to
                // an iOS SpringBoard replacement or a newer mobileprovision.
                reply(true, recoveredTransport
                    ? "native_operation_returned_unverified_transport_recovered"
                    : "native_operation_returned_unverified")
            } catch {
                reply(false, safeFailureReason(error))
            }
        }
    }
}

// V29: in-app iOS 27 RPPairing passes only an in-memory on-device record.
// The record is sensitive: never copy it into the JSON diagnostic journal,
// source control, pasteboard or remote server. Check the *original* pinned
// SideStore parser and live Minimuxer before atomically changing saved state.
extension ScarletGuestRefreshAgent {
    @objc static func importDevicePairing(_ payload: Data,
        reply: @escaping (Bool, String) -> Void) {
        Task {
            guard payload.count > 128 && payload.count <= 1024 * 1024,
                  let candidate = String(data: payload, encoding: .utf8) else {
                reply(false, "pairing_file_invalid")
                return
            }
            let pairingManager = PairingFileManager.shared
            do {
                let parsed = try pairingManager.parse(
                    content: candidate, preferred: .rppairing)
                guard parsed.mode == .rppairing else {
                    reply(false, "pairing_protocol_mismatch")
                    return
                }
                // Test the new identity while it is IN MEMORY. The saved
                // pairing file, settings, user auth and old cert are intact.
                try await AppBootManager.shared.startMinimuxer(
                    pairingFile: candidate)
                try await ensureMinimuxerReady()
                // Only a confirmed usable identity is committed.
                try pairingManager.savePairingFile(
                    contents: candidate, preferred: .rppairing)
                pairingManager.persistedActiveProtocol = .rppairing
                reply(true, "pairing_saved_and_transport_ready")
            } catch {
                // A failed candidate MUST NOT overwrite a working old file.
                // Recover previous transport when an old record exists.
                if let previous = pairingManager.fetchPairingFile(),
                   previous != candidate {
                    try? await AppBootManager.shared.startMinimuxer(
                        pairingFile: previous)
                }
                reply(false, safeFailureReason(error))
            }
        }
    }
}
