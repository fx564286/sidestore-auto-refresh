
// Explicit user-requested certificate resolution in the pinned SideStore
// LiveProcess. Uses the selected Apple ID's verified Team and native
// account-pinned signing authority. No revocation, new account, or deep link.
// The private key stays in memory until the existing authenticated XPC
// connection returns it to the host; the host verifies the Apple team before
// changing its existing LC signing certificate.
@objc(ScarletGuestCertificateAgent)
final class ScarletGuestCertificateAgent: NSObject {
    @MainActor private static var busy = false

    @objc(issue:)
    static func issue(_ payload: NSDictionary) {
        guard let requestID = payload["requestID"] as? String,
              !requestID.isEmpty else { return }
        Task { @MainActor in
            guard !busy else {
                report(requestID, code: -5201, domain: "scarlet.certificate.busy")
                return
            }
            busy = true
            defer { busy = false }
            // The original V6 certificate worker was pinned to the host's
            // first Team. Route this legacy endpoint through the SAME
            // Apple-ID+Team vault as the current native P98 issuer.
            // The chosen IPA signing Team is independent of host entitlements.
            guard ScarletGuestSignInAgent.persistedState(),
                  let selected = await ScarletInstallTeamAuthority.activeOwnership()
            else {
                report(requestID, code: -5205,
                       domain: "scarlet.certificate.selected_owner_unavailable")
                return
            }
            let status = await ScarletInstallTeamAuthority.issue(
                teamID: selected.teamID, forceNew: false)
            guard status == "ready_existing_portal" ||
                  status == "ready_issued_apple" else {
                report(requestID, code: -5204,
                       domain: "scarlet.certificate.selected_team_unavailable")
                return
            }
            guard let ownerNow = await ScarletInstallTeamAuthority.activeOwnership(),
                  ownerNow.appleID == selected.appleID,
                  ownerNow.teamID == selected.teamID,
                  await ScarletInstallTeamAuthority.prepare(
                      teamID: selected.teamID) == "ready",
                  let pinnedSerial = ScarletInstallTeamAuthority
                      .activeSelectedSignerSerial(
                          appleID: selected.appleID, teamID: selected.teamID),
                  let active = CertificateManager.shared.activeCertificate,
                  active.serialNumber.lowercased() == pinnedSerial,
                  !active.p12Data.isEmpty,
                  active.p12Data.count <= 64 * 1024 else {
                report(requestID, code: -5204,
                       domain: "scarlet.certificate.selected_signer_unverified")
                return
            }
            let origin = status == "ready_issued_apple"
                ? "scarlet.cert.new_apple_portal"
                : "scarlet.cert.existing_portal"
            report(requestID, code: 0, domain: origin,
                   archive: active.p12Data, password: active.password ?? "")
        }
    }

    // SideInstaller's "Use SideStore certificate" action is a *retrieval*
    // path, not another issuance flow. In LiveContainer it reads an existing
    // signable P12; outside it asks SideStore to hand off its existing P12.
    // This agent is already inside the authenticated SideStore LiveProcess,
    // so it can return ONLY the stored certificate without touching Apple's
    // portal, generating a new key, or revoking another certificate.
    @objc(retrieve:)
    static func retrieve(_ payload: NSDictionary) {
        guard let requestID = payload["requestID"] as? String,
              !requestID.isEmpty else { return }
        Task { @MainActor in
            guard let owner = await ScarletInstallTeamAuthority.activeOwnership(),
                  await ScarletInstallTeamAuthority.prepare(
                      teamID: owner.teamID) == "ready",
                  let pinned = ScarletInstallTeamAuthority.activeSelectedSignerSerial(
                      appleID: owner.appleID, teamID: owner.teamID),
                  let active = CertificateManager.shared.activeCertificate,
                  active.serialNumber.lowercased() == pinned,
                  !active.p12Data.isEmpty,
                  active.p12Data.count <= 64 * 1024 else {
                report(requestID, code: -5302,
                       domain: "scarlet.certificate.selected_p12_missing")
                return
            }
            report(requestID, code: 0, domain: "scarlet.cert.existing_stored",
                   archive: active.p12Data, password: active.password ?? "")
        }
    }

    @MainActor
    private static func report(_ requestID: String, code: Int,
                               domain: String, archive: Data = Data(),
                               password: String = "") {
        NotificationCenter.default.post(
            name: Notification.Name("ScarletGuestCertificateCompleted"),
            object: nil,
            userInfo: ["requestID": requestID,
                       "code": code, "domain": domain,
                       "archive": archive, "password": password])
    }
}
