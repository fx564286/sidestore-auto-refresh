// All guest receipts share one anonymous App Group relay. This carries only
// bounded numbers and a PID; no labels, view names, app identifiers or images.
static void ScarletIslandPublish(NSDictionary *row) {
    NSUserDefaults *store = NSUserDefaults.lcSharedDefaults;
    if(!store || !row) return;
    NSString *key = @"Scarlet.Guest.IslandRootReceipt.V1";
    NSData *raw = [store dataForKey:key];
    NSMutableArray *items = [NSMutableArray array];
    if(raw) {
        id decoded = [NSJSONSerialization JSONObjectWithData:raw
            options:NSJSONReadingMutableContainers error:NULL];
        if([decoded isKindOfClass:NSArray.class]) [items addObjectsFromArray:decoded];
    }
    [items addObject:row];
    if(items.count > 18)
        [items removeObjectsInRange:NSMakeRange(0, items.count - 18)];
    NSData *bytes = [NSJSONSerialization dataWithJSONObject:items
        options:0 error:NULL];
    if(!bytes) return;
    [store setObject:bytes forKey:key];
    NSString *name = [NSString stringWithFormat:
        @"com.scarlet.guest.island.ack.%d", getpid()];
    CFNotificationCenterPostNotification(
        CFNotificationCenterGetDarwinNotifyCenter(),
        (__bridge CFStringRef)name, NULL, NULL, true);
}

static void ScarletIslandReceipt(NSString *result, CGFloat required,
                                 CGFloat before, CGFloat after) {
    ScarletIslandPublish(@{@"receipt":NSUUID.UUID.UUIDString,
        @"pid":@(getpid()), @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":result ?: @"unknown", @"target":@((int)round(required)),
        @"before":@((int)round(before)), @"after":@((int)round(after))});
}

// R6P18: root Safe Area was already 62pt on both iPhone LiveProcesses.
// Measure actual descendant interactive geometry, never reposition it.
// Only UIKit UIView descendants are enumerable; SwiftUI virtual elements
// can be invisible to this scanner. Zero candidates never proves no overlap.
typedef struct {
    NSUInteger visited;
    NSUInteger controls;
    NSUInteger topControls;
    NSUInteger unsafeControls;
    NSUInteger visibleUnsafeControls;
    NSUInteger clippedUnsafeControls;
    NSUInteger unsafeControlsInScroll;
    NSUInteger unsafeControlsOutsideScroll;
    NSUInteger unsafeControlsUnderZeroInsetChildOwner;
    NSUInteger topScrollWithoutInset;
    NSUInteger topTextNodes;
    NSUInteger unsafeTextNodes;
    NSUInteger topImageNodes;
    NSUInteger unsafeImageNodes;
    NSUInteger visibleUnsafeImageNodes;
    NSUInteger clippedUnsafeImageNodes;
    NSUInteger unsafeImageNodesInScroll;
    NSUInteger unsafeImageNodesOutsideScroll;
    NSUInteger maxDepthSeen;
    BOOL depthCapped;
    BOOL countCapped;
    NSInteger firstUnsafeTextY;
    NSInteger firstUnsafeImageY;
    NSInteger firstTopScrollY;
    NSInteger firstTopControlY;
    NSInteger firstUnsafeControlY;
    BOOL capped;
} ScarletIslandViewMetrics;

// R6P23: inspect ancestry, not another root Safe Area / scroll inset.
// A correct 62pt root can host top controls owned by an independent
// nested UIViewController, or by a scroll view that R6P22 intentionally
// refused to modify. Names, classes, URLs and labels are never serialized.
static void ScarletIslandMeasureUnsafeOwner(
    UIView *node, UIView *root, BOOL *hasScroll, BOOL *zeroInsetChildOwner) {
    *hasScroll = NO;
    *zeroInsetChildOwner = NO;
    NSUInteger budget = 48;
    for(UIView *ancestor = node; ancestor && budget--; ancestor = ancestor.superview) {
        if([ancestor isKindOfClass:UIScrollView.class]) *hasScroll = YES;
        UIResponder *responder = ancestor.nextResponder;
        if([responder isKindOfClass:UIViewController.class]) {
            UIViewController *owner = (UIViewController *)responder;
            if(owner.view != root && owner.isViewLoaded &&
               owner.view.safeAreaInsets.top < 2)
                *zeroInsetChildOwner = YES;
        }
        if(ancestor == root) break;
    }
}

// R6P25: a subview with window-space y<62 can be entirely clipped by an
// ancestor UIScrollView. Distinguish *visible* unsafe content from offscreen
// layout geometry before any new per-scroll repair is considered. READ ONLY.
static CGRect ScarletIslandVisibleFrameInWindow(UIView *node, UIWindow *window) {
    CGRect visible = [node convertRect:node.bounds toView:window];
    if(CGRectIsNull(visible) || CGRectIsEmpty(visible)) return CGRectNull;
    visible = CGRectIntersection(visible, window.bounds);
    NSUInteger hops = 0;
    for(UIView *parent=node.superview; parent && hops++ < 64;
        parent=parent.superview) {
        if(parent.clipsToBounds || [parent isKindOfClass:UIScrollView.class]) {
            CGRect clip = [parent convertRect:parent.bounds toView:window];
            visible = CGRectIntersection(visible, clip);
            if(CGRectIsNull(visible) || CGRectIsEmpty(visible)) return CGRectNull;
        }
        if(parent == window) break;
    }
    return hops >= 64 ? CGRectNull : visible;
}

// R6P27: attribute *visible* unsafe nodes to their closest real scroll
// ancestor, rather than selecting a global scroll by width/verticality.
// This is a read-only ownership probe. Never record view or app identities.
static void ScarletIslandRecordVisibleScrollOwner(
    UIView *node, UIView *root, NSMutableDictionary *owners, BOOL image) {
    UIScrollView *owner = nil;
    NSUInteger hops = 0;
    for(UIView *parent = node.superview; parent && hops++ < 64;
        parent = parent.superview) {
        if([parent isKindOfClass:UIScrollView.class]) {
            owner = (UIScrollView *)parent;
            break;
        }
        if(parent == root) break;
    }
    if(!owner) return;
    NSValue *key = [NSValue valueWithNonretainedObject:owner];
    NSMutableDictionary *counts = owners[key];
    if(!counts) {
        if(owners.count >= 16) return;
        counts = [@{@"controls":@0, @"images":@0} mutableCopy];
        owners[key] = counts;
    }
    NSString *field = image ? @"images" : @"controls";
    counts[field] = @([counts[field] unsignedIntegerValue] + 1);
    // Only retained for this synchronous probe; never serialize objects.
    if(!image) {
        NSMutableArray *nodes = counts[@"control_nodes"];
        if(!nodes) { nodes = [NSMutableArray array]; counts[@"control_nodes"] = nodes; }
        if(nodes.count < 8) [nodes addObject:node];
    }
}

static NSDictionary *ScarletIslandVisibleOwnerMetrics(
    NSDictionary<NSValue *, NSDictionary *> *owners, UIWindow *window) {
    UIScrollView *dominant = nil;
    NSUInteger best = 0, tied = 0, controls = 0, images = 0;
    for(NSValue *key in owners) {
        UIScrollView *scroll = (UIScrollView *)[key nonretainedObjectValue];
        if(!scroll || scroll.window != window) continue;
        NSDictionary *counts = owners[key];
        NSUInteger c = [counts[@"controls"] unsignedIntegerValue];
        NSUInteger i = [counts[@"images"] unsignedIntegerValue];
        NSUInteger score = c * 2 + i;
        if(score > best) {
            best = score;
            dominant = scroll;
            tied = 0;
            controls = c;
            images = i;
        } else if(score && score == best) {
            tied++;
        }
    }
    if(!dominant) {
        return @{@"owner_groups":@(owners.count), @"owner_dominant_controls":@0,
            @"owner_dominant_images":@0, @"owner_tie":@0,
            @"owner_frame_y":@(-999), @"owner_frame_width_pct":@(-999),
            @"owner_frame_height_pct":@(-999),
            @"owner_adjusted_inset_top":@(-999),
            @"owner_content_inset_top":@(-999), @"owner_offset_y":@(-999),
            @"owner_adjustment_behavior":@(-999),
            @"owner_nested_scrolls":@(-999),
            @"owner_scroll_enabled":@(-999),
            @"owner_content_height_pct":@(-999),
            @"owner_safe_top":@(-999)};
    }
    NSUInteger nested = 0, budget = 48;
    for(UIView *parent = dominant.superview; parent && budget--;
        parent = parent.superview)
        if([parent isKindOfClass:UIScrollView.class]) nested++;
    CGRect frame = [dominant convertRect:dominant.bounds toView:window];
    CGFloat winW = MAX(1, CGRectGetWidth(window.bounds));
    CGFloat winH = MAX(1, CGRectGetHeight(window.bounds));
    CGFloat contentH = MAX(1, CGRectGetHeight(dominant.bounds));
    return @{
        @"owner_groups":@(owners.count),
        @"owner_dominant_controls":@(controls),
        @"owner_dominant_images":@(images),
        @"owner_tie":@(tied > 0),
        @"owner_frame_y":@((int)lround(CGRectGetMinY(frame))),
        @"owner_frame_width_pct":@((int)lround(100 * CGRectGetWidth(frame)/winW)),
        @"owner_frame_height_pct":@((int)lround(100 * CGRectGetHeight(frame)/winH)),
        @"owner_adjusted_inset_top":@((int)lround(dominant.adjustedContentInset.top)),
        @"owner_content_inset_top":@((int)lround(dominant.contentInset.top)),
        @"owner_offset_y":@((int)lround(dominant.contentOffset.y)),
        @"owner_adjustment_behavior":@(dominant.contentInsetAdjustmentBehavior),
        @"owner_nested_scrolls":@(nested),
        @"owner_scroll_enabled":@(dominant.scrollEnabled),
        @"owner_content_height_pct":@((int)lround(100 * dominant.contentSize.height/contentH)),
        @"owner_safe_top":@((int)lround(dominant.safeAreaInsets.top))
    };
}

static void ScarletIslandWalkViews(UIView *view, UIWindow *window,
                                   CGFloat safeY, NSUInteger depth,
                                   ScarletIslandViewMetrics *metrics,
                                   NSMutableDictionary *ownerGroups) {
    if(!view || view.hidden || view.alpha < 0.05 || view.window != window)
        return;
    // R6P21: the R6P20 device shows depth=9 truncation after only 15/46
    // visible nodes. This remains a bounded READ-ONLY scan, but distinguishes
    // depth from breadth and includes deeply nested UIKit/SwiftUI host views.
    if(metrics->visited >= 1200) {
        metrics->countCapped = YES;
        metrics->capped = YES;
        return;
    }
    if(depth > 24) {
        metrics->depthCapped = YES;
        metrics->capped = YES;
        return;
    }
    metrics->visited++;
    metrics->maxDepthSeen = MAX(metrics->maxDepthSeen, depth);
    if(depth > 0) {
        CGRect frame = [view convertRect:view.bounds toView:window];
        CGFloat top = CGRectGetMinY(frame);
        CGFloat bottom = CGRectGetMaxY(frame);
        CGFloat height = CGRectGetHeight(frame);
        if(isfinite(top) && isfinite(bottom) && isfinite(height) &&
           height > 2 && height <= 160 && bottom > 0 &&
           top < CGRectGetHeight(window.bounds)) {
            BOOL clickable = [view isKindOfClass:UIControl.class] ||
                (view.isAccessibilityElement &&
                 (view.accessibilityTraits & UIAccessibilityTraitButton) != 0);
            if(clickable) {
                metrics->controls++;
                if(top < safeY + 160) {
                    metrics->topControls++;
                    metrics->firstTopControlY = MIN(
                        metrics->firstTopControlY, (NSInteger)lround(top));
                    if(top < safeY - 2 && bottom > 5) {
                        metrics->unsafeControls++;
                        CGRect shown = ScarletIslandVisibleFrameInWindow(view, window);
                        if(!CGRectIsNull(shown) && !CGRectIsEmpty(shown) &&
                           CGRectGetMinY(shown) < safeY - 2 &&
                           CGRectGetMaxY(shown) > 5)
                            {
                            metrics->visibleUnsafeControls++;
                            ScarletIslandRecordVisibleScrollOwner(
                                view, window.rootViewController.view,
                                ownerGroups, NO);
                        }
                        else metrics->clippedUnsafeControls++;
                        BOOL scrollAncestor = NO, childZero = NO;
                        ScarletIslandMeasureUnsafeOwner(
                            view, window.rootViewController.view,
                            &scrollAncestor, &childZero);
                        if(scrollAncestor) metrics->unsafeControlsInScroll++;
                        else metrics->unsafeControlsOutsideScroll++;
                        if(childZero)
                            metrics->unsafeControlsUnderZeroInsetChildOwner++;
                        metrics->firstUnsafeControlY = MIN(
                            metrics->firstUnsafeControlY, (NSInteger)lround(top));
                    }
                }
            }
        }
        // Text and imagery may be the actual visual navigation bar in
        // SwiftUI-bridged guests, without ever subclassing UIControl.
        // Never inspect/record labels, image bytes or accessibility content.
        if(isfinite(top) && isfinite(bottom) && isfinite(height) &&
           height > 2 && height <= 160 && bottom > 0 &&
           top < safeY + 160) {
            BOOL textNode = [view isKindOfClass:UILabel.class] ||
                [view isKindOfClass:UITextField.class] ||
                [view isKindOfClass:UITextView.class];
            BOOL imageNode = [view isKindOfClass:UIImageView.class];
            if(textNode) {
                metrics->topTextNodes++;
                if(top < safeY - 2 && bottom > 5) {
                    metrics->unsafeTextNodes++;
                    metrics->firstUnsafeTextY = MIN(
                        metrics->firstUnsafeTextY, (NSInteger)lround(top));
                }
            }
            if(imageNode) {
                metrics->topImageNodes++;
                if(top < safeY - 2 && bottom > 5) {
                    metrics->unsafeImageNodes++;
                    CGRect shown = ScarletIslandVisibleFrameInWindow(view, window);
                    if(!CGRectIsNull(shown) && !CGRectIsEmpty(shown) &&
                       CGRectGetMinY(shown) < safeY - 2 &&
                       CGRectGetMaxY(shown) > 5)
                        {
                        metrics->visibleUnsafeImageNodes++;
                        ScarletIslandRecordVisibleScrollOwner(
                            view, window.rootViewController.view,
                            ownerGroups, YES);
                    }
                    else metrics->clippedUnsafeImageNodes++;
                    BOOL scrollAncestor = NO, childZero = NO;
                    ScarletIslandMeasureUnsafeOwner(
                        view, window.rootViewController.view,
                        &scrollAncestor, &childZero);
                    if(scrollAncestor) metrics->unsafeImageNodesInScroll++;
                    else metrics->unsafeImageNodesOutsideScroll++;
                    metrics->firstUnsafeImageY = MIN(
                        metrics->firstUnsafeImageY, (NSInteger)lround(top));
                }
            }
        }
        if([view isKindOfClass:UIScrollView.class]) {
            UIScrollView *scroll = (UIScrollView *)view;
            if(isfinite(top) && isfinite(bottom) && top < safeY &&
               bottom > safeY && scroll.adjustedContentInset.top < 2) {
                metrics->topScrollWithoutInset++;
                metrics->firstTopScrollY = MIN(
                    metrics->firstTopScrollY, (NSInteger)lround(top));
            }
        }
    }
    for(UIView *child in view.subviews) {
        if(metrics->visited >= 1200) {
            metrics->capped = YES;
            metrics->countCapped = YES;
            break;
        }
        ScarletIslandWalkViews(child, window, safeY, depth + 1,
                              metrics, ownerGroups);
    }
}

// R6P34: the signed P33 owner was foreground, inset62, offset/bounds-62,
// but its unsafe count stayed3. Quarantine that experiment and inspect
// actual constraints of the observed controls. This helper is READ ONLY.
static NSInteger ScarletIslandAnonymousSlot(UIView *view) {
    static NSMapTable<UIView *, NSNumber *> *slots;
    static NSInteger next = 0;
    if(!slots) slots = [NSMapTable weakToStrongObjectsMapTable];
    NSNumber *slot = [slots objectForKey:view];
    if(slot) return slot.integerValue;
    if(next >= 64) return 0;
    slot = @(++next);
    [slots setObject:slot forKey:view];
    return slot.integerValue;
}

static BOOL ScarletIslandVerticalPosition(NSLayoutAttribute attribute) {
    switch(attribute) {
        case NSLayoutAttributeTop: case NSLayoutAttributeBottom:
        case NSLayoutAttributeCenterY: case NSLayoutAttributeBaseline:
        case NSLayoutAttributeFirstBaseline: case NSLayoutAttributeTopMargin:
        case NSLayoutAttributeBottomMargin: case NSLayoutAttributeCenterYWithinMargins:
            return YES;
        default: return NO;
    }
}

static UIView *ScarletIslandConstraintItemView(id item) {
    if([item isKindOfClass:UIView.class]) return item;
    if([item isKindOfClass:UILayoutGuide.class]) return [(UILayoutGuide *)item owningView];
    return nil;
}

static NSInteger ScarletIslandMetric(CGFloat value) {
    return isfinite(value) ? (NSInteger)lround(value) : -999;
}

static void ScarletIslandProbeConstraints(
    UIWindow *window, UIView *root, CGFloat safeTop, NSUInteger sample,
    NSDictionary<NSValue *, NSDictionary *> *groups) {
    NSMutableArray *rows = [NSMutableArray array];
    BOOL rowsCapped = NO;
    for(NSValue *key in groups) {
        UIScrollView *owner = [key nonretainedObjectValue];
        if(!owner || owner.window != window) continue;
        NSArray *nodes = groups[key][@"control_nodes"];
        for(UIView *node in nodes) {
            if(rows.count >= 12) { rowsCapped = YES; break; }
            if(node.window != window || ![node isDescendantOfView:owner]) continue;
            CGRect visible = ScarletIslandVisibleFrameInWindow(node, window);
            if(CGRectIsNull(visible) || CGRectIsEmpty(visible) ||
               CGRectGetMinY(visible) >= safeTop - 2) continue;
            CGRect frame = [node convertRect:node.bounds toView:window];
            CGRect relative = [node convertRect:node.bounds toView:owner];
            NSUInteger inspected=0, total=0, frameRefs=0, contentRefs=0;
            NSUInteger safeRefs=0, externalRefs=0, centerRefs=0, otherRefs=0;
            NSUInteger levels=0;
            BOOL capped=NO, inNavbar=NO;
            CGFloat controllerSafe=-999, firstConstant=-999;
            NSInteger firstAttribute=-999, firstOtherAttribute=-999;
            NSInteger firstRelation=-999, firstPriority=-999;
            NSMutableSet *seen = [NSMutableSet set];
            UIView *cursor=node;
            while(cursor && cursor != owner && levels++ < 8) {
                if([cursor isKindOfClass:UINavigationBar.class]) inNavbar=YES;
                if(controllerSafe == -999 &&
                   [cursor.nextResponder isKindOfClass:UIViewController.class]) {
                    UIViewController *controller=(UIViewController *)cursor.nextResponder;
                    if(controller.isViewLoaded) controllerSafe=controller.view.safeAreaInsets.top;
                }
                for(NSLayoutConstraint *constraint in
                    [cursor constraintsAffectingLayoutForAxis:UILayoutConstraintAxisVertical]) {
                    if(inspected++ >= 96) { capped=YES; break; }
                    if(!constraint.active || [seen containsObject:constraint]) continue;
                    [seen addObject:constraint];
                    BOOL first=ScarletIslandConstraintItemView(constraint.firstItem)==cursor;
                    BOOL second=ScarletIslandConstraintItemView(constraint.secondItem)==cursor;
                    if(!first && !second) continue;
                    NSLayoutAttribute attr=first ? constraint.firstAttribute : constraint.secondAttribute;
                    NSLayoutAttribute otherAttr=first ? constraint.secondAttribute : constraint.firstAttribute;
                    if(!ScarletIslandVerticalPosition(attr)) continue;
                    id other=first ? constraint.secondItem : constraint.firstItem;
                    UIView *otherView=ScarletIslandConstraintItemView(other);
                    if(!otherView) continue;
                    total++;
                    BOOL framePin=NO;
                    if([other isKindOfClass:UILayoutGuide.class] &&
                       [otherView isKindOfClass:UIScrollView.class] &&
                       other == ((UIScrollView *)otherView).frameLayoutGuide) {
                        frameRefs++; framePin=YES;
                    } else if([other isKindOfClass:UILayoutGuide.class] &&
                              [otherView isKindOfClass:UIScrollView.class] &&
                              other == ((UIScrollView *)otherView).contentLayoutGuide) {
                        contentRefs++;
                    } else if([other isKindOfClass:UILayoutGuide.class] &&
                              other == otherView.safeAreaLayoutGuide) {
                        safeRefs++;
                    } else if(otherView != owner && ![otherView isDescendantOfView:owner]) {
                        externalRefs++; framePin=YES;
                    } else if(other == owner &&
                              (otherAttr == NSLayoutAttributeCenterY ||
                               otherAttr == NSLayoutAttributeCenterYWithinMargins)) {
                        centerRefs++; framePin=YES;
                    } else {
                        otherRefs++;
                    }
                    if(framePin && firstAttribute == -999) {
                        firstAttribute=attr; firstOtherAttribute=otherAttr;
                        firstRelation=constraint.relation;
                        firstConstant=constraint.constant;
                        firstPriority=(NSInteger)lround(constraint.priority);
                    }
                }
                if(capped) break;
                cursor=cursor.superview;
            }
            if(cursor && cursor != owner) capped=YES;
            [rows addObject:@{
                @"node_slot":@(ScarletIslandAnonymousSlot(node)),
                @"scroll_slot":@(ScarletIslandAnonymousSlot(owner)),
                @"node_window_x":@(ScarletIslandMetric(CGRectGetMinX(frame))),
                @"node_window_width":@(ScarletIslandMetric(CGRectGetWidth(frame))),
                @"node_window_height":@(ScarletIslandMetric(CGRectGetHeight(frame))),
                @"node_window_y":@(ScarletIslandMetric(CGRectGetMinY(frame))),
                @"node_window_bottom":@(ScarletIslandMetric(CGRectGetMaxY(frame))),
                @"node_owner_y":@(ScarletIslandMetric(CGRectGetMinY(relative))),
                @"node_local_y":@(ScarletIslandMetric(CGRectGetMinY(node.frame))),
                @"node_autoresizing":@(node.translatesAutoresizingMaskIntoConstraints),
                @"node_ambiguous":@(node.hasAmbiguousLayout),
                @"node_in_navbar":@(inNavbar),
                @"controller_safe_top":@(ScarletIslandMetric(controllerSafe)),
                @"owner_offset_y":@(ScarletIslandMetric(owner.contentOffset.y)),
                @"owner_bounds_y":@(ScarletIslandMetric(owner.bounds.origin.y)),
                @"constraint_count":@(total), @"constraint_inspected":@(MIN(inspected,96)),
                @"constraint_frame_refs":@(frameRefs),
                @"constraint_content_refs":@(contentRefs),
                @"constraint_safe_refs":@(safeRefs),
                @"constraint_external_refs":@(externalRefs),
                @"constraint_center_refs":@(centerRefs),
                @"constraint_other_refs":@(otherRefs),
                @"constraint_first_attribute":@(firstAttribute),
                @"constraint_other_attribute":@(firstOtherAttribute),
                @"constraint_first_relation":@(firstRelation),
                @"constraint_first_constant":@(ScarletIslandMetric(firstConstant)),
                @"constraint_first_priority":@(firstPriority),
                @"constraint_scan_capped":@(capped)
            }];
        }
        if(rowsCapped) break;
    }
    if(rows.count == 0) return;
    // One bounded relay packet avoids overflowing the shared 18-packet ring.
    ScarletIslandPublish(@{
        @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
        @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":@"guest_constraint_owner_probe", @"target":@(ScarletIslandMetric(safeTop)),
        @"sample_index":@(sample), @"scene_activation_state":@(window.windowScene.activationState),
        @"probe_nodes_capped":@(rowsCapped), @"nodes":rows
    });
}

// R6P35: P34 signed iPhone PID2894 found 3 severely unsafe header
// controls at window Y3/13 even though the root and their owner both
// report top safe62. P34's eight-hop ancestor constraint inspection
// was capped for precisely those 3 controls (inspected2/capped1).
// Do NOT infer an unconstrained/manual layout from that incomplete scan.
// Trace up to 24 real ancestor hops through the owner to the root, and
// capture only nonidentifying numeric layout facts. READ ONLY.
static void ScarletIslandProbeUnsafeAncestorChain(
    UIWindow *window, UIView *root, CGFloat safeTop, NSUInteger sample,
    NSDictionary<NSValue *, NSDictionary *> *groups) {
    if(sample != 2600 || !window || !root || safeTop < 20) return;
    NSMutableArray *rows = [NSMutableArray array];
    NSUInteger selected = 0, foundOwner = 0, foundRoot = 0;
    BOOL capped = NO;
    for(NSValue *key in groups) {
        UIScrollView *owner = [key nonretainedObjectValue];
        if(!owner || owner.window != window) continue;
        for(UIView *node in groups[key][@"control_nodes"]) {
            if(selected >= 2) break;
            if(node.window != window || ![node isDescendantOfView:owner])
                continue;
            CGRect shown = ScarletIslandVisibleFrameInWindow(node, window);
            if(CGRectIsNull(shown) || CGRectIsEmpty(shown) ||
               CGRectGetMinY(shown) >= safeTop - 20 ||
               CGRectGetMaxY(shown) <= 5) continue;
            selected++;
            BOOL ownerReached = NO, rootReached = NO;
            NSUInteger hops = 0;
            NSInteger nodeSlot = ScarletIslandAnonymousSlot(node);
            NSInteger ownerSlot = ScarletIslandAnonymousSlot(owner);
            for(UIView *cursor=node; cursor && hops < 24;
                cursor=cursor.superview, hops++) {
                if(rows.count >= 48) { capped = YES; break; }
                if(cursor.window != window) break;
                CGRect rect = [cursor convertRect:cursor.bounds toView:window];
                CGRect guide = [cursor convertRect:
                    cursor.safeAreaLayoutGuide.layoutFrame toView:window];
                CGFloat parentY = cursor.superview ?
                    CGRectGetMinY([cursor.superview convertRect:
                        cursor.superview.bounds toView:window]) : -999;
                NSUInteger vertical = 0;
                for(NSLayoutConstraint *constraint in cursor.constraints) {
                    if(constraint.active &&
                       (ScarletIslandVerticalPosition(constraint.firstAttribute) ||
                        ScarletIslandVerticalPosition(constraint.secondAttribute)))
                        vertical++;
                }
                BOOL controllerRoot = NO;
                NSInteger controllerSafe = -999;
                UIResponder *responder = cursor.nextResponder;
                if([responder isKindOfClass:UIViewController.class]) {
                    UIViewController *vc = (UIViewController *)responder;
                    if(vc.isViewLoaded && vc.view == cursor) {
                        controllerRoot = YES;
                        controllerSafe = ScarletIslandMetric(vc.view.safeAreaInsets.top);
                    }
                }
                if(cursor == owner) ownerReached = YES;
                if(cursor == root) rootReached = YES;
                [rows addObject:@{
                    @"node_slot":@(nodeSlot), @"scroll_slot":@(ownerSlot),
                    @"ancestor_slot":@(ScarletIslandAnonymousSlot(cursor)),
                    @"ancestor_depth":@(hops),
                    @"ancestor_window_y":@(ScarletIslandMetric(CGRectGetMinY(rect))),
                    @"ancestor_parent_window_y":@(ScarletIslandMetric(parentY)),
                    @"ancestor_local_y":@(ScarletIslandMetric(CGRectGetMinY(cursor.frame))),
                    @"ancestor_guide_y":@(ScarletIslandMetric(CGRectGetMinY(guide))),
                    @"ancestor_safe_top":@(ScarletIslandMetric(cursor.safeAreaInsets.top)),
                    @"ancestor_bounds_y":@(ScarletIslandMetric(CGRectGetMinY(cursor.bounds))),

                    @"ancestor_autoresize_mask":@((NSUInteger)cursor.autoresizingMask),
                    @"ancestor_translates":@(cursor.translatesAutoresizingMaskIntoConstraints),
                    @"ancestor_is_controller_root":@(controllerRoot),
                    @"ancestor_controller_safe_top":@(controllerSafe),
                    @"ancestor_is_scroll":@([cursor isKindOfClass:UIScrollView.class]),
                    @"ancestor_is_control":@([cursor isKindOfClass:UIControl.class]),
                    @"ancestor_clips":@(cursor.clipsToBounds),
                    @"ancestor_has_transform":@(!CGAffineTransformIsIdentity(cursor.transform)),
                    @"ancestor_is_owner":@(cursor == owner),
                    @"ancestor_is_root":@(cursor == root)
                }];
                if(cursor == root) break;
            }
            foundOwner += ownerReached;
            foundRoot += rootReached;
            if(!rootReached) capped = YES;
        }
        if(selected >= 2) break;
    }
    if(rows.count == 0) return;
    ScarletIslandPublish(@{
        @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
        @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":@"guest_ancestor_layout_probe",
        @"target":@(ScarletIslandMetric(safeTop)),
        @"sample_index":@(sample),
        @"selected_nodes":@(selected),
        @"owner_reached_nodes":@(foundOwner),
        @"root_reached_nodes":@(foundRoot),
        @"ancestor_rows_capped":@(capped),
        @"nodes":rows
    });
}

// P37 iPhone: header frame62 was restored to0 by native layout. Stop that
// one-off write. Inspect the exact header contract without changing UIKit.
static NSInteger ScarletIslandImplementationOrigin(id object, SEL selector) {
    if(!object) return 0;
    Method method=class_getInstanceMethod([object class],selector);
    if(!method) return 0;
    Dl_info info={0}, own={0};
    if(!dladdr((const void *)method_getImplementation(method),&info) ||
       !info.dli_fname) return 0;
    dladdr((const void *)&ScarletIslandImplementationOrigin,&own);
    if(info.dli_fbase==own.dli_fbase) return 4; // Injected guard image.
    if(strstr(info.dli_fname,"UIKit")) return 1;
    if(strstr(info.dli_fname,"/System/Library/") ||
       strstr(info.dli_fname,"/usr/lib/")) return 2;
    return 3; // Guest/third-party image; never export its path or symbols.
}

static NSInteger ScarletIslandHeaderItemKind(
    id item, UIView *header, UIView *parent, UIView *root) {
    if(!item) return 0;
    if(item==header) return 1;
    if(item==parent) return 2;
    if(item==root) return 3;
    if(item==header.safeAreaLayoutGuide) return 4;
    if(item==parent.safeAreaLayoutGuide) return 5;
    if(item==root.safeAreaLayoutGuide) return 6;
    return [item isKindOfClass:UILayoutGuide.class] ? 8 : 7;
}

static void ScarletIslandObserveDeepHeaderLayoutOwner(
    UIWindow *window, UIView *root, CGFloat safeTop, NSUInteger sample,
    NSDictionary<NSValue *, NSDictionary *> *groups,
    NSUInteger beforeControls, NSUInteger beforeImages) {
    if((sample!=650 && sample!=2600) || !window || !root ||
       safeTop<40 || safeTop>130 || window.hidden ||
       window.windowLevel!=UIWindowLevelNormal ||
       window.rootViewController.view!=root ||
       window.windowScene.activationState!=UISceneActivationStateForegroundActive ||
       !UIInterfaceOrientationIsPortrait(window.windowScene.interfaceOrientation))
        return;
    NSUInteger bit=sample==650 ? 1 : 2;
    NSUInteger seen=[objc_getAssociatedObject(root,&scarletIslandHeaderOwnerProbeKey)
        unsignedIntegerValue];

                             OBJC_ASSOCIATION_RETAIN_NONATOMIC);

                                 @YES,OBJC_ASSOCIATION_RETAIN_NONATOMIC);

    }
    UIView *header=nil, *parent=nil;
    NSUInteger severe=0;
    BOOL conflicting=NO;
    __weak UIView *firstWeak=nil, *secondWeak=nil;
    for(NSValue *key in groups) {
        UIScrollView *scroll=[key nonretainedObjectValue];
        if(!scroll || scroll.window!=window) continue;
        CGRect scrollRect=[scroll convertRect:scroll.bounds toView:window];
        if(fabs(CGRectGetMinY(scrollRect))>3 ||
           CGRectGetHeight(scrollRect)<CGRectGetHeight(window.bounds)*0.85)
            continue;
        for(UIView *node in groups[key][@"control_nodes"]) {
            if(!node || node.window!=window) continue;
            CGRect shown=ScarletIslandVisibleFrameInWindow(node,window);
            if(CGRectIsNull(shown) || CGRectIsEmpty(shown) ||
               CGRectGetMinY(shown)>=safeTop-20 ||
               CGRectGetMaxY(shown)<=5) continue;
            UIView *candidate=nil, *candidateParent=nil;
            NSUInteger depth=0;
            for(UIView *cursor=node; cursor && cursor!=root && depth++<24;
                cursor=cursor.superview) {
                UIResponder *responder=cursor.nextResponder;
                if(![responder isKindOfClass:UIViewController.class] ||
                   ((UIViewController *)responder).view!=cursor)
                    continue;
                UIView *p=cursor.superview;
                if(!p || p.window!=window ||
                   CGRectGetHeight(cursor.bounds)<34 ||
                   CGRectGetHeight(cursor.bounds)>72 ||
                   CGRectGetHeight(p.bounds)<CGRectGetHeight(window.bounds)*0.85 ||

                       toView:window]))>3 ||
                   fabs(CGRectGetMinY([p convertRect:
                       p.safeAreaLayoutGuide.layoutFrame toView:window])-safeTop)>3 ||
                   cursor.safeAreaInsets.top<safeTop-2 ||
                   !cursor.translatesAutoresizingMaskIntoConstraints ||
                   !CGAffineTransformIsIdentity(cursor.transform))
                    continue;
                CGRect guide=[cursor convertRect:
                    cursor.safeAreaLayoutGuide.layoutFrame toView:window];
                CGRect frame=[cursor convertRect:cursor.bounds toView:window];
                if(fabs(CGRectGetMinY(guide)-CGRectGetMaxY(frame))>3)
                    continue; // Safe guide is collapsed by 50pt header.
                candidate=cursor; candidateParent=p;
                break;
            }
            if(!candidate) continue;
            if(header && header!=candidate) { conflicting=YES; break; }
            header=candidate; parent=candidateParent;
            severe++;
            if(severe==1) firstWeak=node;
            if(severe==2) secondWeak=node;
        }
        if(conflicting) break;

        firstWeak && secondWeak;
    NSMutableDictionary *fields=[@{
        @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
        @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":@"guest_header_layout_owner_probe", @"target":@(ScarletIslandMetric(safeTop)),

        @"header_severe_nodes":@(severe), @"header_controls_before":@(beforeControls),
        @"header_images_after":@(beforeImages)

        [parent.nextResponder isKindOfClass:UIViewController.class]
        ? (UIViewController *)parent.nextResponder : nil;
    UIStatusBarManager *manager=window.windowScene.statusBarManager;
    CGRect hrect=[header convertRect:header.bounds toView:window];
    CGRect guide=[parent convertRect:parent.safeAreaLayoutGuide.layoutFrame toView:window];
    Method method=class_getInstanceMethod([parent class],@selector(layoutSubviews));
    Method base=class_getInstanceMethod(UIView.class,@selector(layoutSubviews));
    [fields addEntriesFromDictionary:@{
        @"header_slot":@(ScarletIslandAnonymousSlot(header)),
        @"header_parent_slot":@(ScarletIslandAnonymousSlot(parent)),
        @"header_window_y":@(ScarletIslandMetric(CGRectGetMinY(hrect))),
        @"header_height":@(ScarletIslandMetric(CGRectGetHeight(hrect))),
        @"header_local_y":@(ScarletIslandMetric(header.frame.origin.y)),
        @"header_parent_guide_y":@(ScarletIslandMetric(CGRectGetMinY(guide))),
        @"header_translates":@(header.translatesAutoresizingMaskIntoConstraints),
        @"header_autoresize_mask":@(header.autoresizingMask),
        @"header_parent_autoresizes":@(parent.autoresizesSubviews),
        @"header_ambiguous":@(header.hasAmbiguousLayout),
        @"header_controller_parent_match":@(controller.parentViewController==parentController && parentController!=nil),
        @"header_edges":@((NSInteger)(controller.edgesForExtendedLayout & 15)),
        @"header_parent_edges":@(parentController ? (NSInteger)(parentController.edgesForExtendedLayout & 15) : -1),
        @"header_prefers_status_hidden":@(controller.prefersStatusBarHidden),
        @"header_parent_prefers_status_hidden":@(parentController ? parentController.prefersStatusBarHidden : -1),
        @"header_scene_status_hidden":@(manager ? manager.isStatusBarHidden : -1),

        @"header_parent_layout_impl_origin":@(ScarletIslandImplementationOrigin(parent,@selector(layoutSubviews))),
        @"header_parent_layout_override":@(method && base &&
            method_getImplementation(method)!=method_getImplementation(base)),
        @"header_controller_layout_impl_origin":@(ScarletIslandImplementationOrigin(controller,@selector(viewDidLayoutSubviews))),
        @"header_parent_controller_layout_impl_origin":@(ScarletIslandImplementationOrigin(parentController,@selector(viewDidLayoutSubviews))),
        @"header_setframe_impl_origin":@(ScarletIslandImplementationOrigin(header,@selector(setFrame:))),
        @"header_setcenter_impl_origin":@(ScarletIslandImplementationOrigin(header,@selector(setCenter:))),
        @"header_setbounds_impl_origin":@(ScarletIslandImplementationOrigin(header,@selector(setBounds:)))
    }];
    // These are implementation owners, NOT evidence that a method executed.
    // Direct constraint endpoints reveal whether reported constraints actually
    // position the short header or merely describe an unrelated ancestor.
    NSMutableArray *rows=[NSMutableArray array];
    NSMutableSet *unique=[NSMutableSet set];
    NSArray *lists=@[header.constraints,
        [header constraintsAffectingLayoutForAxis:UILayoutConstraintAxisVertical],
        parent.constraints];
    NSUInteger inspected=0, relevant=0, source=0;
    BOOL capped=NO;
    Class maskClass=NSClassFromString(@"NSAutoresizingMaskLayoutConstraint");
    for(NSArray *list in lists) {
        source++;
        for(NSLayoutConstraint *constraint in list) {
            if(inspected++>=96) { capped=YES; break; }
            if(!constraint.active || [unique containsObject:constraint]) continue;
            [unique addObject:constraint];
            UIView *a=ScarletIslandConstraintItemView(constraint.firstItem);
            UIView *b=ScarletIslandConstraintItemView(constraint.secondItem);
            if(a!=header && b!=header) continue;
            if(!ScarletIslandVerticalPosition(constraint.firstAttribute) &&
               !ScarletIslandVerticalPosition(constraint.secondAttribute) &&
               constraint.firstAttribute!=NSLayoutAttributeHeight &&
               constraint.secondAttribute!=NSLayoutAttributeHeight) continue;
            relevant++;
            if(rows.count>=16) { capped=YES; continue; }
            NSInteger holder=0;
            if([header.constraints containsObject:constraint]) holder=1;
            else if([parent.constraints containsObject:constraint]) holder=2;
            else if([root.constraints containsObject:constraint]) holder=3;
            [rows addObject:@{
                @"header_constraint_source":@(source), @"header_constraint_holder":@(holder),
                @"header_constraint_first_kind":@(ScarletIslandHeaderItemKind(constraint.firstItem,header,parent,root)),
                @"header_constraint_second_kind":@(ScarletIslandHeaderItemKind(constraint.secondItem,header,parent,root)),
                @"header_constraint_first_attr":@(constraint.firstAttribute),
                @"header_constraint_second_attr":@(constraint.secondAttribute),
                @"header_constraint_relation":@(constraint.relation),
                @"header_constraint_constant":@(ScarletIslandMetric(constraint.constant)),
                @"header_constraint_multiplier_milli":@(ScarletIslandMetric(constraint.multiplier*1000)),
                @"header_constraint_priority":@(ScarletIslandMetric(constraint.priority)),
                @"header_constraint_mask_known":@(maskClass!=Nil),
                @"header_constraint_autoresizing":@(maskClass && [constraint isKindOfClass:maskClass])
            }];
        }
        if(inspected>=96) break;
    }
    fields[@"header_constraints_inspected"]=@(MIN(inspected,96));
    fields[@"header_constraints_relevant"]=@(relevant);
    fields[@"header_constraints_capped"]=@(capped);
    fields[@"header_constraints"]=rows;
    ScarletIslandPublish(fields);
    ScarletIslandTryMeasuredHeaderConstraintBridge(
        window, root, header, parent, safeTop, sample,
        beforeControls, beforeImages, firstWeak, secondWeak);
}

            if(severe==1) firstWeak=node;
            if(severe==2) secondWeak=node;
        }
        if(conflicting) break;
    }
    if(!header || !parent || conflicting || severe<2 ||
       !firstWeak || !secondWeak)
        return;
    CGRect original=header.frame;
    CGRect winBounds=window.bounds;
    CGPoint parentTarget=[parent convertPoint:CGPointMake(0,safeTop)
        fromView:window];
    CGFloat targetY=parentTarget.y;
    if(!isfinite(targetY) ||
       targetY-original.origin.y<safeTop-5 ||
       targetY-original.origin.y>safeTop+5 ||
       !ScarletIslandDeepHeaderOwnerReady(header,parent,root,window,
                                        safeTop,0))
        return;
    objc_setAssociatedObject(root,&scarletIslandDeepHeaderAnchorAttemptedKey,
                             @YES,OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    __weak UIView *weakHeader=header, *weakParent=parent, *weakRoot=root;
    __weak UIWindow *weakWindow=window;
    CGRect moved=original;
    moved.origin.y=targetY; // ONLY the 50pt child header view's origin.
    header.frame=moved; // Preserve original x, width, height and touch geometry.
    ScarletIslandDeepHeaderTrialReceipt(
        @"guest_short_header_trial_staged", @"parent_safe_guide_anchor",
        safeTop, original.origin.y, targetY, beforeControls,
        beforeControls, beforeImages, NO);
    // No re-entrant layout setter: verify after UIKit settles naturally.
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
        (int64_t)180*NSEC_PER_MSEC),dispatch_get_main_queue(), ^{
        UIView *h=weakHeader, *p=weakParent, *rt=weakRoot;
        UIWindow *w=weakWindow;
        BOOL attached=ScarletIslandDeepHeaderOwnerReady(
            h,p,rt,w,safeTop,safeTop);
        BOOL originalSize=w &&
            fabs(CGRectGetWidth(w.bounds)-CGRectGetWidth(winBounds))<2 &&
            fabs(CGRectGetHeight(w.bounds)-CGRectGetHeight(winBounds))<2;
        NSUInteger images=beforeImages;
        NSUInteger controls=(attached && originalSize) ?
            ScarletIslandDeepHeaderCountUnsafe(rt,w,safeTop,&images) :
            beforeControls;
        CGRect first=firstWeak && w ?
            ScarletIslandVisibleFrameInWindow(firstWeak,w) : CGRectNull;
        CGRect second=secondWeak && w ?
            ScarletIslandVisibleFrameInWindow(secondWeak,w) : CGRectNull;
        BOOL improved=attached && originalSize &&
            !CGRectIsNull(first) && !CGRectIsNull(second) &&
            !CGRectIsEmpty(first) && !CGRectIsEmpty(second) &&
            CGRectGetMinY(first)>=safeTop-2 &&
            CGRectGetMinY(second)>=safeTop-2 &&
            controls+2<=beforeControls &&
            images<=beforeImages+1;
        if(!improved) {
            if(h && p && w && h.superview==p &&
               fabs(h.frame.origin.y-targetY)<=3)
                h.frame=original;
            ScarletIslandDeepHeaderTrialReceipt(
                @"guest_short_header_trial_reverted",
                attached ? @"no_immediate_reduction" : @"native_layout_overrode",
                safeTop, original.origin.y,
                h ? h.frame.origin.y : -999, beforeControls,
                controls, images, NO);
            return;
        }
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
            (int64_t)1100*NSEC_PER_MSEC),dispatch_get_main_queue(), ^{
            UIView *lateHeader=weakHeader,*lateParent=weakParent,*lateRoot=weakRoot;
            UIWindow *lateWindow=weakWindow;
            BOOL stable=ScarletIslandDeepHeaderOwnerReady(
                lateHeader,lateParent,lateRoot,lateWindow,safeTop,safeTop);
            BOOL windowSame=lateWindow &&
                fabs(CGRectGetWidth(lateWindow.bounds)-CGRectGetWidth(winBounds))<2 &&
                fabs(CGRectGetHeight(lateWindow.bounds)-CGRectGetHeight(winBounds))<2;
            NSUInteger lateImages=beforeImages;
            NSUInteger lateControls=(stable && windowSame) ?
                ScarletIslandDeepHeaderCountUnsafe(
                    lateRoot,lateWindow,safeTop,&lateImages) : beforeControls;
            CGRect one=firstWeak && lateWindow ?
                ScarletIslandVisibleFrameInWindow(firstWeak,lateWindow) : CGRectNull;
            CGRect two=secondWeak && lateWindow ?
                ScarletIslandVisibleFrameInWindow(secondWeak,lateWindow) : CGRectNull;
            BOOL retained=stable && windowSame &&
                !CGRectIsNull(one) && !CGRectIsNull(two) &&
                !CGRectIsEmpty(one) && !CGRectIsEmpty(two) &&
                CGRectGetMinY(one)>=safeTop-2 &&
                CGRectGetMinY(two)>=safeTop-2 &&
                lateControls+2<=beforeControls &&
                lateImages<=beforeImages+1;
            if(!retained && lateHeader && lateParent && lateWindow &&
               lateHeader.superview==lateParent &&
               fabs(lateHeader.frame.origin.y-targetY)<=3)
                lateHeader.frame=original;
            ScarletIslandDeepHeaderTrialReceipt(
                retained ? @"guest_short_header_trial_retained"
                         : @"guest_short_header_trial_reverted",
                retained ? @"late_safe_guide_confirmed"
                         : @"late_layout_not_persistent",
                safeTop, original.origin.y,
                lateHeader ? lateHeader.frame.origin.y : -999,
                beforeControls, lateControls, lateImages, retained);
        });
    });
}

// R6P36: P35 iPhone PID3252 receives a valid root-safe62 receipt but
// emits no 650/2600 geometry packet, while PID3244 does. The original
// scheduled reader silently returns on a lost/replaced/hidden root.
// Record exact exclusion reason. Read a *replacement* attached root once
// at the already scheduled 2600ms sample; never alter UIKit geometry.
static void ScarletIslandProbeStatus(NSString *reason, UIWindow *window,
    UIViewController *root, CGFloat expectedTop, NSUInteger sample,
    BOOL replacement) {
    UIWindowScene *scene=window.windowScene;
    UIView *view=(root && root.isViewLoaded) ? root.view : nil;
    CGFloat safe=view ? view.safeAreaInsets.top : -999;
    ScarletIslandPublish(@{
        @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
        @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":@"guest_geometry_probe_status",
        @"reason":reason ?: @"unknown", @"target":@(ScarletIslandMetric(expectedTop)),
        @"sample_index":@(sample),
        @"probe_window_present":@(window != nil),
        @"probe_window_visible":@(window && !window.hidden),
        @"probe_window_normal":@(window && window.windowLevel==UIWindowLevelNormal),
        @"probe_scene_present":@(scene != nil),
        @"probe_scene_state":@(scene ? scene.activationState : -1),
        @"probe_portrait":@(scene && UIInterfaceOrientationIsPortrait(scene.interfaceOrientation)),
        @"probe_root_present":@(root != nil),
        @"probe_root_match":@(window && root && window.rootViewController==root),
        @"probe_root_attached":@(window && view && view.window==window),
        @"probe_root_safe_top":@(ScarletIslandMetric(safe)),
        @"probe_replacement":@(replacement)
    });
}
static void ScarletIslandScheduleContentProbe(UIWindow *window,
                                               UIViewController *root,
                                               CGFloat expectedTop) {
    if(objc_getAssociatedObject(root, &scarletIslandContentProbeKey)) return;
    objc_setAssociatedObject(root, &scarletIslandContentProbeKey,
        @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    __weak UIWindow *weakWindow = window;
    __weak UIViewController *weakRoot = root;
    ScarletIslandProbeStatus(@"scheduled", window, root, expectedTop, 0, NO);
    // Preserve existing P35 650/2600ms read times: this is NOT a timer
    // retry nor another contentInset/frame/safe-area adjustment.
    for(NSNumber *delay in @[@650, @2600]) {
        NSUInteger sample = delay.unsignedIntegerValue;
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
            (int64_t)sample * NSEC_PER_MSEC), dispatch_get_main_queue(), ^{
            UIWindow *liveWindow=weakWindow;
            UIViewController *liveRoot=weakRoot;
            if(!liveWindow) {
                ScarletIslandProbeStatus(@"window_released", nil, liveRoot,
                    expectedTop, sample, NO);
                return;
            }
            BOOL replacement=NO;
            if(!liveRoot || liveWindow.rootViewController != liveRoot) {
                // At the single late measurement only, follow the actual
                // attached root rather than silently dropping the packet.
                if(sample==2600 && liveWindow.rootViewController) {
                    liveRoot=liveWindow.rootViewController;
                    replacement=YES;
                    ScarletIslandProbeStatus(@"root_replaced", liveWindow,
                        liveRoot, expectedTop, sample, YES);
                } else {
                    ScarletIslandProbeStatus(@"root_stale", liveWindow,
                        liveRoot, expectedTop, sample, NO);
                    return;
                }
            }
            if(liveWindow.hidden || liveWindow.windowLevel!=UIWindowLevelNormal) {
                ScarletIslandProbeStatus(@"window_ineligible", liveWindow,
                    liveRoot, expectedTop, sample, replacement);
                return;
            }
            UIWindowScene *scene=liveWindow.windowScene;
            if(!scene || !UIInterfaceOrientationIsPortrait(scene.interfaceOrientation)) {
                ScarletIslandProbeStatus(@"scene_ineligible", liveWindow,
                    liveRoot, expectedTop, sample, replacement);
                return;
            }
            UIView *view=liveRoot.isViewLoaded ? liveRoot.view : nil;
            if(!view || view.window!=liveWindow) {
                ScarletIslandProbeStatus(@"root_detached", liveWindow,
                    liveRoot, expectedTop, sample, replacement);
                return;
            }
            CGFloat top=view.safeAreaInsets.top;
            if(!isfinite(top) || top<20 || !isfinite(expectedTop)) {
                ScarletIslandProbeStatus(@"safe_unready", liveWindow,
                    liveRoot, expectedTop, sample, replacement);
                return;
            }
            CGRect frame=[view convertRect:view.bounds toView:liveWindow];
            CGRect guide=[view convertRect:
                view.safeAreaLayoutGuide.layoutFrame toView:liveWindow];
            if(!isfinite(CGRectGetMinY(frame)) ||
               !isfinite(CGRectGetMinY(guide))) {
                ScarletIslandProbeStatus(@"frame_invalid", liveWindow,
                    liveRoot, expectedTop, sample, replacement);
                return;
            }
            ScarletIslandContentProbe(liveWindow, liveRoot, expectedTop, sample);
        });
    }
}

static void ScarletIslandInspect(UIWindow *window, NSUInteger attempt) {
    if(!window || window.hidden || window.windowLevel != UIWindowLevelNormal ||
       UIDevice.currentDevice.userInterfaceIdiom != UIUserInterfaceIdiomPhone)
        return;
    UIWindowScene *scene = window.windowScene;
    if(!scene || !UIInterfaceOrientationIsPortrait(scene.interfaceOrientation))
        return;
    UIViewController *root = window.rootViewController;
    UIView *content = root.isViewLoaded ? root.view : nil;
    if(!root || !content || content.window != window ||
       CGRectGetWidth(window.bounds) < 250 || attempt < 2) {
        // Avoid a false root 0pt while UIWindow is visible but its initial
        // UIKit safe area has not completed first layout; one bounded settle.
        if(attempt < 7) {
            __weak UIWindow *weakWindow = window;
            dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
                (int64_t)(attempt + 1) * 150 * NSEC_PER_MSEC),
                dispatch_get_main_queue(), ^{
                    ScarletIslandInspect(weakWindow, attempt + 1);
                });
        }
        return;
    }
    // Host safe-top fallback is anonymous normal UIWindow geometry; the
    // correction itself is applied inside the real guest UI root.
    CGFloat sceneTop = CGRectGetMaxY(scene.statusBarManager.statusBarFrame);
    CGFloat hostTop = [NSUserDefaults.lcSharedDefaults
        doubleForKey:@"Scarlet.Guest.PortraitSafeTop.V1"];
    CGFloat target = MAX(sceneTop, hostTop);
    [content layoutIfNeeded];
    CGFloat before = content.safeAreaInsets.top;
    // R6P40: original-LC-style transparent host chrome reserves the actual
    // 62pt BEFORE remote FBScene content. This is NOT a separate guest inset.
    // Never guess from a process-global flag alone: require the guest's own
    // UIWindow to have the expected native screen-height-minus-slot height.
    CGFloat chrome = [NSUserDefaults.lcSharedDefaults
        doubleForKey:@"Scarlet.Guest.TransparentChromeTop.V1"];
    CGFloat screen = CGRectGetHeight(UIScreen.mainScreen.bounds);
    if(isfinite(chrome) && chrome >= 20 && chrome <= 130 &&
       isfinite(screen) && screen >= 500 &&
       fabs(CGRectGetHeight(window.bounds) + chrome - screen) < 4) {
        ScarletIslandReceipt(@"guest_transparent_chrome_slot_unpadded",
                             chrome, before, before);
        return;
    }
    if(!isfinite(target) || target < 20 || target > 130) {
        ScarletIslandReceipt(@"target_unavailable", 0, before, before);
        return;
    }
    // Two UIWindow notifications can arrive in the same layout pass. Own
    // exactly one additive delta per guest root and prevent double padding.
    NSNumber *owned = objc_getAssociatedObject(
        root, &scarletIslandRootAddedInsetKey);
    CGFloat ourDelta = MAX(0, owned.doubleValue);
    if(ourDelta > 0 && before > target + 3) {
        // If UIKit's natural safe area became valid after an earlier 0pt
        // readback, remove ONLY our excess contribution, not app-owned inset.
        CGFloat trim = MIN(ourDelta, before - target);
        UIEdgeInsets extra = root.additionalSafeAreaInsets;
        extra.top -= trim;
        objc_setAssociatedObject(root, &scarletIslandRootAddedInsetKey,
            @(ourDelta - trim), OBJC_ASSOCIATION_RETAIN_NONATOMIC);
        root.additionalSafeAreaInsets = extra;
        [content setNeedsLayout];
        [content layoutIfNeeded];
        ScarletIslandReceipt(@"guest_root_excess_trimmed", target, before,
                             content.safeAreaInsets.top);
        return;
    }
    if(before >= target - 2) {
        ScarletIslandReceipt(@"guest_root_already_safe", target, before, before);
        ScarletIslandScheduleContentProbe(window, root, target);
        // R6P22: one separate, bounded descendant-scroll remedy. Never add
        // another inset to this already-correct guest root or its UIWindow.
        ScarletIslandScheduleScrollContentRepair(window, root, target);
        // R6P30 iPhone evidence cc5bafc85b66 PID1059: L071's manual
        // local-inset write read back 3->12 immediately but was reset to
        // 3pt after layout; owner frame jumped y50->23 and remaining unsafe
        // controls grew 8->9 and images 6->8. QUARANTINE the P28 mutator,
        // do not keep re-applying it during the next guest session. L072's
        // separately guarded automatic owner trial remains eligible.
        // Keep L071 implementation in immutable source history for audit.
        if(!objc_getAssociatedObject(root, &scarletIslandP28QuarantineReceiptKey)) {
            objc_setAssociatedObject(root, &scarletIslandP28QuarantineReceiptKey,
                @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
            ScarletIslandReceipt(@"guest_local_owner_quarantined",
                                 target, 0, 0);
        }
        // R6P29: one independent UIKit automatic-inset experiment for a
        // different, full-viewport residual owner AFTER the R6P28 local fix.
        // R6P31: verified P30 iPhone P29 .automatic trial read back 0pt;
        // use the distinct ALWAYS policy for non-vertical content, with
        // immediate AND late content visibility checks and safe rollback.
        // Signed P33 proves the ALWAYS anchor was actually applied at -62
        // in foreground and still failed. Do not repeat its enum/offset writes.
        if(!objc_getAssociatedObject(root, &scarletIslandAlwaysScheduleKey)) {
            objc_setAssociatedObject(root, &scarletIslandAlwaysScheduleKey,
                @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
            ScarletIslandReceipt(@"guest_always_policy_quarantined", target, 0, 0);
        }
        return;
    }
    if(ourDelta > 0) {
        // A pending UIKit layout readback cannot authorize a second write.
        ScarletIslandReceipt(@"guest_root_update_unconfirmed", target,
                             before, before);
        return;
    }
    UIEdgeInsets extra = root.additionalSafeAreaInsets;
    CGFloat deficit = MIN(130, MAX(0, target - before));
    objc_setAssociatedObject(root, &scarletIslandRootAddedInsetKey,
        @(deficit), OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    extra.top += deficit; // no guest window/frame/scale changes
    root.additionalSafeAreaInsets = extra;
    [content setNeedsLayout];
    [content layoutIfNeeded];
    CGFloat after = content.safeAreaInsets.top;
    ScarletIslandReceipt(after >= target - 2
                             ? @"guest_root_corrected"
                             : @"guest_root_update_unconfirmed",
                         target, before, after);
}

// R6P22: unlike the L050 root correction and the L061 read-only scanner,
// repair a single, demonstrably unsafe, full-width vertical scroll's content.
// Do not change any UIWindow, root safe area, display scale or guest bundle.
// A deliberate edge-to-edge layout without unsafe interactive descendants is
// left unchanged. Multiple candidate scrollers are left unchanged rather than
// guessing which one owns the guest's navigation controls.
static BOOL ScarletIslandHasUnsafeScrollControl(
    UIView *view, UIWindow *window, CGFloat safeY,
    NSUInteger depth, NSUInteger *budget) {
    if(!view || !*budget || depth > 32 ||
       view.hidden || view.alpha < 0.05 || view.window != window)
        return NO;
    --*budget;
    if(depth > 0) {
        CGRect rect = [view convertRect:view.bounds toView:window];
        CGFloat top = CGRectGetMinY(rect), bottom = CGRectGetMaxY(rect);
        BOOL control = [view isKindOfClass:UIControl.class] ||
            (view.isAccessibilityElement &&
             (view.accessibilityTraits & UIAccessibilityTraitButton) != 0);
        if(control && isfinite(top) && isfinite(bottom) &&
           top < safeY - 2 && bottom > 5 &&
           CGRectGetHeight(rect) > 2 && CGRectGetHeight(rect) < 160)
            return YES;
    }
    for(UIView *child in view.subviews)
        if(ScarletIslandHasUnsafeScrollControl(
               child, window, safeY, depth + 1, budget))
            return YES;
    return NO;
}

// R6P23 classifier: each exclusion counter is a real predicate readback.
// Counters overlap by design; no scroll receives an inset from this probe.
typedef struct {
    NSUInteger seen, nested, notTopFullWidth, notVertical,
        existingInset, notAtTop, busy, missingUnsafeDescendant, eligible;
} ScarletIslandScrollClass;

static void ScarletIslandCollectUnsafeScrolls(
    UIView *view, UIView *root, UIWindow *window, CGFloat safeY,
    NSUInteger depth, NSUInteger *budget,
    NSMutableArray<UIScrollView *> *candidates,
    ScarletIslandScrollClass *classification) {
    if(!view || !*budget || depth > 32 ||
       view.hidden || view.alpha < 0.05 || view.window != window)
        return;
    --*budget;
    if([view isKindOfClass:UIScrollView.class]) {
        UIScrollView *scroll = (UIScrollView *)view;
        BOOL nested = NO;
        for(UIView *parent = scroll.superview; parent && parent != root;
            parent = parent.superview)
            if([parent isKindOfClass:UIScrollView.class]) {
                nested = YES;
                break;
            }
        CGRect frame = [scroll convertRect:scroll.bounds toView:window];
        CGFloat top = CGRectGetMinY(frame);
        classification->seen++;
        BOOL topFullWidth = isfinite(top) && top >= -5 && top <= 12 &&
            CGRectGetWidth(frame) >= CGRectGetWidth(window.bounds) * 0.7 &&
            CGRectGetHeight(frame) >= CGRectGetHeight(window.bounds) * 0.4;
        BOOL vertical = scroll.scrollEnabled &&
            scroll.contentSize.height > scroll.bounds.size.height + 12;
        BOOL unmodified = !objc_getAssociatedObject(
            scroll, &scarletIslandScrollInsetKey) &&
            scroll.adjustedContentInset.top < 2;
        BOOL noGesture = !scroll.dragging && !scroll.tracking &&
            !scroll.decelerating;
        BOOL nearTop = scroll.contentOffset.y >= -2 &&
            scroll.contentOffset.y <= 8;
        NSUInteger childBudget = 900;
        BOOL unsafe = ScarletIslandHasUnsafeScrollControl(
            scroll, window, safeY, 0, &childBudget);
        if(nested) classification->nested++;
        if(!topFullWidth) classification->notTopFullWidth++;
        if(!vertical) classification->notVertical++;
        if(!unmodified) classification->existingInset++;
        if(!nearTop) classification->notAtTop++;
        if(!noGesture) classification->busy++;
        if(!unsafe) classification->missingUnsafeDescendant++;
        if(!nested && topFullWidth && vertical && unmodified &&
           nearTop && noGesture && unsafe) {
            classification->eligible++;
            [candidates addObject:scroll];
        }
    }
    for(UIView *child in view.subviews)
        ScarletIslandCollectUnsafeScrolls(
            child, root, window, safeY, depth + 1, budget,
            candidates, classification);
}

static void ScarletIslandRepairTopScrollableContent(
    UIWindow *window, UIViewController *root, CGFloat expectedTop) {
    if(!window || window.hidden || window.rootViewController != root ||
       !UIInterfaceOrientationIsPortrait(window.windowScene.interfaceOrientation))
        return;
    UIView *rootView = root.isViewLoaded ? root.view : nil;
    if(!rootView || rootView.window != window ||
       rootView.safeAreaInsets.top < expectedTop - 2 ||
       expectedTop < 20 || expectedTop > 130)
        return;
    NSMutableArray<UIScrollView *> *candidates = [NSMutableArray array];
    ScarletIslandScrollClass classification = {0};
    NSUInteger budget = 1200;
    ScarletIslandCollectUnsafeScrolls(rootView, rootView, window,
                                      expectedTop, 0, &budget, candidates,
                                      &classification);
    // Distinguish R6P22 '0 eligible' from 'multiple eligible', and expose
    // exactly why. The bounded probe never serializes any guest identity.
    ScarletIslandPublish(@{
        @"receipt":NSUUID.UUID.UUIDString,
        @"pid":@(getpid()), @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":@"guest_scroll_target_classifier",
        @"target":@((int)lround(expectedTop)),
        @"seen":@(classification.seen),
        @"nested":@(classification.nested),
        @"not_top_full_width":@(classification.notTopFullWidth),
        @"not_vertical":@(classification.notVertical),
        @"existing_inset":@(classification.existingInset),
        @"not_at_top":@(classification.notAtTop),
        @"busy":@(classification.busy),
        @"missing_unsafe_descendant":@(classification.missingUnsafeDescendant),
        @"eligible":@(classification.eligible),
        @"scan_budget_exhausted":@(budget == 0)
    });
    if(candidates.count != 1) {
        ScarletIslandReceipt(@"guest_scroll_no_unique_unsafe_target",
                             expectedTop, candidates.count, candidates.count);
        return;
    }
    UIScrollView *scroll = candidates.firstObject;
    CGRect frame = [scroll convertRect:scroll.bounds toView:window];
    CGFloat deficit = ceil(expectedTop - CGRectGetMinY(frame));
    if(!isfinite(deficit) || deficit < 16 || deficit > 90) return;
    // One write per actual guest scroller; never modify a scroller with an
    // existing top inset, active gesture, or a non-top content offset.
    if(scroll.adjustedContentInset.top >= 2 || scroll.tracking ||
       scroll.dragging || scroll.decelerating ||
       scroll.contentOffset.y < -2 || scroll.contentOffset.y > 8)
        return;
    CGFloat before = scroll.adjustedContentInset.top;
    UIEdgeInsets inset = scroll.contentInset;
    inset.top += deficit;
    objc_setAssociatedObject(scroll, &scarletIslandScrollInsetKey,
                             @(deficit), OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    scroll.contentInset = inset;
    CGPoint offset = scroll.contentOffset;
    offset.y = -scroll.adjustedContentInset.top;
    scroll.contentOffset = offset;
    ScarletIslandReceipt(@"guest_scroll_content_inset_applied",
                         expectedTop, before, scroll.adjustedContentInset.top);
}

static void ScarletIslandScheduleScrollContentRepair(
    UIWindow *window, UIViewController *root, CGFloat expectedTop) {
    if(objc_getAssociatedObject(root, &scarletIslandScrollScheduleKey))
        return;
    objc_setAssociatedObject(root, &scarletIslandScrollScheduleKey,
                             @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    __weak UIWindow *weakWindow = window;
    __weak UIViewController *weakRoot = root;
    // 650ms probe captures original layout; 900ms single guarded correction;
    // 2600ms existing probe observes the result without masking the evidence.
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
        (int64_t)900 * NSEC_PER_MSEC), dispatch_get_main_queue(), ^{
        ScarletIslandRepairTopScrollableContent(
            weakWindow, weakRoot, expectedTop);
    });
}

// R6P28: unique visible control owner in the original guest's nested scroll
// hierarchy. L063's global vertical/full-height scroll gate could not select
// any of five scrollers, despite 8 visible controls at Y3 and root safeTop62.
// R6P27 iPhone: dominant short header scroll starts at window Y50, owns 5
// visible unsafe controls and 5 images, local safeTop12 but top inset3,
// contentOffset -3 and adjustmentBehavior NEVER. Only add the verified LOCAL
// missing 9pt to this one native owner. Never resize/translate the guest,
// alter an app's adjustment behavior, or change other/parent scrollers.
static void ScarletIslandLocalOwnerReceipt(
    NSString *result, NSString *reason, CGFloat target,
    CGFloat before, CGFloat after, UIScrollView *owner, UIWindow *window,
    NSUInteger groupCount, NSUInteger controls, NSUInteger images) {
    CGRect frame = owner
        ? [owner convertRect:owner.bounds toView:window] : CGRectNull;
    ScarletIslandPublish(@{
        @"receipt":NSUUID.UUID.UUIDString,
        @"pid":@(getpid()),
        @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":result ?: @"guest_local_owner_inset_skipped",
        @"reason":reason ?: @"none",
        @"target":@((int)lround(target)),
        @"before":@((int)lround(before)),
        @"after":@((int)lround(after)),
        @"owner_groups":@(groupCount),
        @"owner_dominant_controls":@(controls),
        @"owner_dominant_images":@(images),
        @"owner_frame_y":@(owner ? (int)lround(CGRectGetMinY(frame)) : -999),
        @"owner_safe_top":@(owner
            ? (int)lround(owner.safeAreaInsets.top) : -999)
    });
}

static void ScarletIslandRepairVisibleLocalScrollOwner(
    UIWindow *window, UIViewController *root, CGFloat expectedTop) {
    if(!window || window.hidden || window.rootViewController != root ||
       !window.windowScene ||
       !UIInterfaceOrientationIsPortrait(window.windowScene.interfaceOrientation))
        return;
    UIView *rootView = root.isViewLoaded ? root.view : nil;
    if(!rootView || rootView.window != window || expectedTop < 20 ||
       expectedTop > 130 ||
       fabs(rootView.safeAreaInsets.top - expectedTop) > 2)
        return;

    // Reuse the existing R6P27 actual-visibility classifier, including the
    // complete ancestor clipping chain, not the old raw window-space Y.
    ScarletIslandViewMetrics metrics = {0};
    NSMutableDictionary *groups = [NSMutableDictionary dictionary];
    ScarletIslandWalkViews(rootView, window, expectedTop, 0,
                           &metrics, groups);
    if(metrics.visibleUnsafeControls < 2 || groups.count == 0) return;

    UIScrollView *owner = nil;
    NSUInteger bestScore = 0, selectedControls = 0, selectedImages = 0;
    BOOL ambiguous = NO;
    for(NSValue *key in groups) {
        UIScrollView *candidate = (UIScrollView *)[key nonretainedObjectValue];
        if(!candidate || candidate.window != window) continue;
        NSDictionary *count = groups[key];
        NSUInteger controls = [count[@"controls"] unsignedIntegerValue];
        NSUInteger images = [count[@"images"] unsignedIntegerValue];
        NSUInteger score = controls * 2 + images;
        if(score > bestScore) {
            owner = candidate;
            bestScore = score;
            selectedControls = controls;
            selectedImages = images;
            ambiguous = NO;
        } else if(score > 0 && score == bestScore) {
            ambiguous = YES;
        }
    }
    if(!owner || selectedControls < 2 || ambiguous) {
        ScarletIslandLocalOwnerReceipt(
            @"guest_local_owner_inset_skipped", @"ambiguous_visible_owner",
            expectedTop, 0, 0, owner, window, groups.count,
            selectedControls, selectedImages);
        return;
    }
    // Never overwrite an existing R6P22 inset or run twice on the same view.
    if(objc_getAssociatedObject(owner, &scarletIslandScrollInsetKey) ||
       objc_getAssociatedObject(owner, &scarletIslandLocalOwnerAppliedKey))
        return;

    CGRect frame = [owner convertRect:owner.bounds toView:window];
    CGFloat y = CGRectGetMinY(frame);
    CGFloat width = CGRectGetWidth(frame);
    CGFloat height = CGRectGetHeight(frame);
    CGFloat windowWidth = CGRectGetWidth(window.bounds);
    CGFloat windowHeight = CGRectGetHeight(window.bounds);
    CGFloat localSafeTop = owner.safeAreaInsets.top;
    CGFloat before = owner.adjustedContentInset.top;
    CGFloat deficit = ceil(localSafeTop - before);
    // An observed local safe-area gap, NOT another root 62pt inset. Require
    // a short full-width owner crossing the actual guest safe-top, at rest,
    // with UIKit automatic inset adjustment explicitly disabled. A deliberate
    // edge-to-edge view without visible unsafe descendants is never touched.
    BOOL trust = isfinite(y) && isfinite(width) && isfinite(height) &&
        isfinite(localSafeTop) && isfinite(before) &&
        y >= 0 && y < expectedTop - 2 &&
        CGRectGetMaxY(frame) > expectedTop + 2 &&
        width >= windowWidth * 0.90 &&
        height >= windowHeight * 0.05 &&
        height <= windowHeight * 0.30 &&
        fabs(y + localSafeTop - expectedTop) <= 3 &&
        deficit >= 4 && deficit <= 24 &&
        before >= 0 && before <= 20 &&
        UIScrollViewContentInsetAdjustmentNever ==
            owner.contentInsetAdjustmentBehavior &&
        owner.scrollEnabled &&
        owner.contentSize.height <= owner.bounds.size.height + 24 &&
        !owner.dragging && !owner.tracking && !owner.decelerating &&
        fabs(owner.contentOffset.y + before) <= 2;
    if(!trust) {
        ScarletIslandLocalOwnerReceipt(
            @"guest_local_owner_inset_skipped", @"owner_guard_not_satisfied",
            localSafeTop, before, before, owner, window, groups.count,
            selectedControls, selectedImages);
        return;
    }

    // Single bounded local write. Preserve existing 3pt, add at most 24pt;
    // keep a top-anchored scroll at the new negative inset offset.
    UIEdgeInsets original = owner.contentInset;
    CGPoint originalOffset = owner.contentOffset;
    UIEdgeInsets inset = original;
    inset.top += deficit;
    owner.contentInset = inset;
    CGPoint offset = owner.contentOffset;
    offset.y = -owner.adjustedContentInset.top;
    owner.contentOffset = offset;
    CGFloat after = owner.adjustedContentInset.top;
    if(!isfinite(after) || after < localSafeTop - 2 ||
       after > localSafeTop + 3) {
        // If UIKit refused/recomputed the requested inset, restore exactly
        // the original guest-owned values rather than leaving two offsets.
        owner.contentInset = original;
        owner.contentOffset = originalOffset;
        ScarletIslandLocalOwnerReceipt(
            @"guest_local_owner_inset_skipped", @"write_readback_rollback",
            localSafeTop, before, owner.adjustedContentInset.top,
            owner, window, groups.count, selectedControls, selectedImages);
        return;
    }
    objc_setAssociatedObject(owner, &scarletIslandLocalOwnerAppliedKey,
                             @(deficit), OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    ScarletIslandLocalOwnerReceipt(
        @"guest_local_owner_inset_applied", @"local_safe_gap_closed",
        localSafeTop, before, after, owner, window, groups.count,
        selectedControls, selectedImages);
}

static void ScarletIslandScheduleVisibleLocalOwnerRepair(
    UIWindow *window, UIViewController *root, CGFloat expectedTop) {
    if(objc_getAssociatedObject(root, &scarletIslandLocalOwnerScheduleKey))
        return;
    objc_setAssociatedObject(root, &scarletIslandLocalOwnerScheduleKey,
                             @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    __weak UIWindow *weakWindow = window;
    __weak UIViewController *weakRoot = root;
    // 650ms: untouched actual visibility. 900ms: old L063 guard.
    // 1100ms: this owner-specific repair only if the old guard did nothing.
    // 2600ms: existing diagnostic readback measures the remaining overlap.
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
        (int64_t)1100 * NSEC_PER_MSEC), dispatch_get_main_queue(), ^{
        ScarletIslandRepairVisibleLocalScrollOwner(
            weakWindow, weakRoot, expectedTop);
    });
}

// R6P29: R6P28 actual signed iPhone (PID698) proved 5/8 controls moved
// out of the island overlap by the *local* 9pt header inset. At 2600ms the
// remaining 3 controls + 2 images belonged to another, FULL-SIZE scroll
// owner (y0, height100%, width103%, adjustedInset0, safeTop62,
// adjustmentBehavior NEVER). L063 explicitly required a tall VERTICAL
// scroller and could never select this non-scrolling viewport owner.
// Different causal lever: ask UIKit to calculate its automatic safe-area
// inset rather than manually writing another global/root/local inset.
// Roll back immediately unless UIKit both creates the expected inset and
// reduces actually visible unsafe controls for this exact owner.
static void ScarletIslandResidualAutoReceipt(
    NSString *result, NSString *reason, CGFloat safeTop,
    CGFloat before, CGFloat after, NSUInteger controls,
    NSUInteger remaining) {
    ScarletIslandPublish(@{
        @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
        @"epoch":@([[NSDate date] timeIntervalSince1970]),
        @"result":result ?: @"guest_residual_auto_skipped",
        @"reason":reason ?: @"none",
        @"target":@((int)lround(safeTop)),
        @"before":@((int)lround(before)), @"after":@((int)lround(after)),
        @"owner_dominant_controls":@(controls),
        @"visible_unsafe_controls":@(remaining)
    });
}

// R6P33_ANCHOR_DECISION_BEGIN
// Shared with the executable C regression test. A late inset is a new
// readiness transition, not permission to retry a guest-rejected offset.
static int ScarletIslandShouldAnchorAfterInset(
    double initialInset, double observedInset, double target,
    double originalY, double currentY, int attempted, int stationary) {
    return !attempted && stationary &&
        isfinite(initialInset) && isfinite(observedInset) &&
        isfinite(target) && isfinite(originalY) && isfinite(currentY) &&
        target >= 20 && target <= 130 &&
        !(initialInset >= target - 3 && initialInset <= target + 12) &&
        observedInset >= target - 3 && observedInset <= target + 12 &&
        fabs(originalY) <= 2 && fabs(currentY - originalY) <= 2;
}
// R6P33_ANCHOR_DECISION_END

static void ScarletIslandTryResidualAutomaticInset(
    UIWindow *window, UIViewController *root, CGFloat expectedTop,
    BOOL always, BOOL finalPass) {
    if(!window || window.hidden || window.rootViewController != root ||
       !window.windowScene ||
       !UIInterfaceOrientationIsPortrait(window.windowScene.interfaceOrientation))
        return;
    UIView *rootView = root.isViewLoaded ? root.view : nil;
    if(!rootView || rootView.window != window ||
       expectedTop < 20 || expectedTop > 130 ||
       fabs(rootView.safeAreaInsets.top - expectedTop) > 2)
        return;
    // The P28 local-owner fix ran at 1100ms. Reclassify REAL remaining
    // visible controls rather than rediscovering the already-fixed header.
    ScarletIslandViewMetrics metrics = {0};
    NSMutableDictionary *groups = [NSMutableDictionary dictionary];
    ScarletIslandWalkViews(rootView, window, expectedTop, 0,
                           &metrics, groups);
    if(metrics.visibleUnsafeControls < 2 || groups.count == 0) return;

    UIScrollView *selected = nil;
    NSUInteger selectedControls = 0;
    // R6P30: when P29 skips, preserve exactly which geometry or native
    // scroll-state predicates excluded the remaining visible owners.
    NSUInteger withControls = 0, frameMatched = 0, safeMatched = 0;
    NSUInteger insetMatched = 0, contentMatched = 0, stateMatched = 0;
    NSUInteger disabledScrollOwners = 0;
    CGFloat winW = CGRectGetWidth(window.bounds);
    CGFloat winH = CGRectGetHeight(window.bounds);
    for(NSValue *key in groups) {
        UIScrollView *candidate = (UIScrollView *)[key nonretainedObjectValue];
        if(!candidate || candidate.window != window) continue;
        NSUInteger n = [groups[key][@"controls"] unsignedIntegerValue];
        CGRect rect = [candidate convertRect:candidate.bounds toView:window];
        CGFloat localSafe = candidate.safeAreaInsets.top;
        CGFloat adjusted = candidate.adjustedContentInset.top;
        CGFloat explicitInset = candidate.contentInset.top;
        CGFloat offset = candidate.contentOffset.y;
        CGFloat contentRatio = candidate.contentSize.height /
            MAX(1, candidate.bounds.size.height);
        BOOL matchedFrame = fabs(CGRectGetMinY(rect)) <= 3 &&
            CGRectGetWidth(rect) >= winW * 0.97 &&
            CGRectGetWidth(rect) <= winW * 1.10 &&
            CGRectGetHeight(rect) >= winH * 0.85 &&
            CGRectGetHeight(rect) <= winH * 1.12;
        BOOL matchedSafe = fabs(localSafe - expectedTop) <= 2;
        BOOL matchedInset = fabs(adjusted) <= 1 &&
            fabs(explicitInset) <= 1;
        BOOL matchedContent = contentRatio >= 0.85 &&
            contentRatio <= 1.12;
        // R6P32: P30 signed iPhone PID1604 shows the full-viewport
        // owner has scrollEnabled=0 once stable. This is a stationary
        // UIScrollView whose child controls still intrude into the island.
        // UIKit ALWAYS is specifically allowed to manage this owner;
        // never alter scrollEnabled or enable gestures. Preserve the legacy
        // AUTOMATIC pathway's stricter scrollEnabled guard.
        BOOL matchedState = fabs(offset) <= 2 &&
            (always || candidate.scrollEnabled) &&
            candidate.contentInsetAdjustmentBehavior ==
                UIScrollViewContentInsetAdjustmentNever &&
            !candidate.dragging && !candidate.tracking &&
            !candidate.decelerating;
        if(n >= 2) {
            withControls++;
            if(!candidate.scrollEnabled) disabledScrollOwners++;
            if(matchedFrame) frameMatched++;
            if(matchedSafe) safeMatched++;
            if(matchedInset) insetMatched++;
            if(matchedContent) contentMatched++;
            if(matchedState) stateMatched++;
        }
        BOOL exactResidualOwner = n >= 2 &&
            !objc_getAssociatedObject(candidate, &scarletIslandLocalOwnerAppliedKey) &&
            !objc_getAssociatedObject(candidate, &scarletIslandResidualAutoAppliedKey) &&
            !objc_getAssociatedObject(candidate, &scarletIslandAlwaysAppliedKey) &&
            fabs(CGRectGetMinY(rect)) <= 3 &&
            CGRectGetWidth(rect) >= winW * 0.97 &&
            CGRectGetWidth(rect) <= winW * 1.10 &&
            CGRectGetHeight(rect) >= winH * 0.85 &&
            CGRectGetHeight(rect) <= winH * 1.12 &&
            fabs(localSafe - expectedTop) <= 2 &&
            fabs(adjusted) <= 1 && fabs(explicitInset) <= 1 &&
            fabs(offset) <= 2 &&
            contentRatio >= 0.85 && contentRatio <= 1.12 &&
            candidate.contentInsetAdjustmentBehavior ==
                UIScrollViewContentInsetAdjustmentNever &&
            (always || candidate.scrollEnabled) &&
            !candidate.dragging && !candidate.tracking &&
            !candidate.decelerating;
        if(!exactResidualOwner) continue;
        if(selected) {
            ScarletIslandResidualAutoReceipt(
                @"guest_residual_auto_skipped", @"multiple_matching_owners",
                expectedTop, 0, 0, n, metrics.visibleUnsafeControls);
            return;
        }
        selected = candidate;
        selectedControls = n;
    }
    if(!selected) {
        ScarletIslandPublish(@{
            @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
            @"epoch":@([[NSDate date] timeIntervalSince1970]),
            @"result":@"guest_residual_auto_classifier",
            @"owner_groups":@(groups.count),
            @"candidate_with_controls":@(withControls),
            @"candidate_frame_matched":@(frameMatched),
            @"candidate_safe_matched":@(safeMatched),
            @"candidate_inset_matched":@(insetMatched),
            @"candidate_content_matched":@(contentMatched),
            @"candidate_state_matched":@(stateMatched),
            @"candidate_disabled_scroll":@(disabledScrollOwners),
            @"owner_gate_pass":@(finalPass ? 2 : 1)
        });
        if(always && !finalPass) {
            // P30 PID1604: at 1850ms one full-window owner matched frame,
            // safeTop and height, but its inset and state gates differed
            // from the stable 2600ms readback (adjustedInset0,
            // scrollEnabled0). Retry ONCE after native layout settles.
            // Keep every frame/inset/safe-area/uniqueness predicate strict.
            ScarletIslandResidualAutoReceipt(
                @"guest_always_owner_retry_scheduled",
                @"owner_metrics_not_stable",
                expectedTop, 0, 0, 0, metrics.visibleUnsafeControls);
            __weak UIWindow *retryWindow = window;
            __weak UIViewController *retryRoot = root;
            dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
                (int64_t)900 * NSEC_PER_MSEC),
                dispatch_get_main_queue(), ^{
                ScarletIslandTryResidualAutomaticInset(
                    retryWindow, retryRoot, expectedTop, YES, YES);
            });
            return;
        }
        ScarletIslandResidualAutoReceipt(
            always ? @"guest_always_owner_retry_exhausted"
                   : @"guest_residual_auto_skipped",
            always ? @"strict_owner_unavailable_after_retry"
                   : @"no_matching_fullscreen_owner",
            expectedTop, 0, 0, 0, metrics.visibleUnsafeControls);
        return;
    }
    // No frame, transform, additionalSafeAreaInsets, custom contentInset,
    // appearance, display scale, or gesture state is modified.
    CGFloat before = selected.adjustedContentInset.top;
    CGPoint originalOffset = selected.contentOffset;
    __weak UIWindow *weakWindow = window;
    __weak UIViewController *weakRoot = root;
    __weak UIScrollView *weakOwner = selected;
    // P30 iPhone proved `.automatic` is a no-op on this nonvertical owner.
    // Apple's `.always` includes safe-area insets even without scrollable
    // content; no manual contentInset/root/window/frame/scale modification.
    selected.contentInsetAdjustmentBehavior = always
        ? UIScrollViewContentInsetAdjustmentAlways
        : UIScrollViewContentInsetAdjustmentAutomatic;
    [selected setNeedsLayout];
    [selected layoutIfNeeded];
    CGFloat initialAdjustmentTop = selected.adjustedContentInset.top;
    BOOL initialAnchorAttempted = NO;
    if(always && isfinite(selected.adjustedContentInset.top) &&
       selected.adjustedContentInset.top >= expectedTop - 3 &&
       selected.adjustedContentInset.top <= expectedTop + 12) {
        // The guest's owner begins at Y0 and was contentOffset0. Under
        // UIKit-adjusted top inset, anchor content at the new safe top.
        initialAnchorAttempted = YES;
        CGPoint anchored = selected.contentOffset;
        anchored.y = -selected.adjustedContentInset.top;
        selected.contentOffset = anchored;
        [selected layoutIfNeeded];
    }
    CGFloat initialAnchorOffset = selected.contentOffset.y;
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
        (int64_t)180 * NSEC_PER_MSEC), dispatch_get_main_queue(), ^{
        UIScrollView *owner = weakOwner;
        UIWindow *liveWindow = weakWindow;
        UIViewController *liveRoot = weakRoot;
        if(!owner || !liveWindow || !liveRoot ||
           liveWindow.rootViewController != liveRoot ||
           owner.window != liveWindow ||
           !UIInterfaceOrientationIsPortrait(
               liveWindow.windowScene.interfaceOrientation)) {
            if(owner) {
                owner.contentInsetAdjustmentBehavior =
                    UIScrollViewContentInsetAdjustmentNever;
                owner.contentOffset = originalOffset;
            }
            return;
        }
        // R6P33: P32 read back adjustedInset=62 but controls3->3.
        // The synchronous check above can observe 0 and skip the only
        // anchor. When the SAME owner's inset first becomes valid, consume
        // that pending anchor once, only while its guest-owned offset and
        // adjustment policy remain untouched. Never retry a rejected write.
        CGFloat observedInset = owner.adjustedContentInset.top;
        CGFloat observedOffset = owner.contentOffset.y;
        BOOL anchoredLate = always && ScarletIslandShouldAnchorAfterInset(
            initialAdjustmentTop, observedInset, expectedTop,
            originalOffset.y, observedOffset, initialAnchorAttempted,
            !owner.dragging && !owner.tracking && !owner.decelerating) &&
            owner.contentInsetAdjustmentBehavior ==
                UIScrollViewContentInsetAdjustmentAlways &&
            fabs(owner.contentInset.top) <= 1 &&
            fabs(owner.contentOffset.x - originalOffset.x) <= 2 &&
            fabs(liveRoot.view.safeAreaInsets.top - expectedTop) <= 2 &&
            liveWindow.windowScene.activationState ==
                UISceneActivationStateForegroundActive;
        if(anchoredLate) {
            CGPoint anchored = owner.contentOffset;
            anchored.y = -observedInset;
            owner.contentOffset = anchored;
            [owner setNeedsLayout];
            [owner layoutIfNeeded];
        }
        if(always) {
            // Capture BEFORE rollback, so a guest-reset offset cannot be
            // mistaken for a successful -adjustedInset anchor.
            NSString *anchorResult = anchoredLate ? @"late_anchor_attempted"
                : (initialAnchorAttempted ? @"initial_anchor_attempted"
                                         : @"anchor_not_attempted");
            ScarletIslandPublish(@{
                @"receipt":NSUUID.UUID.UUIDString, @"pid":@(getpid()),
                @"epoch":@([[NSDate date] timeIntervalSince1970]),
                @"result":@"guest_always_anchor_readback",
                @"reason":anchorResult,
                @"target":@((int)lround(expectedTop)),
                @"anchor_initial_inset_pt":@((int)lround(initialAdjustmentTop)),
                @"anchor_initial_offset_y":@((int)lround(initialAnchorOffset)),
                @"anchor_observed_inset_pt":@((int)lround(observedInset)),
                @"anchor_observed_offset_y":@((int)lround(observedOffset)),
                @"anchor_final_offset_y":@((int)lround(owner.contentOffset.y)),
                @"anchor_attempts":@((int)initialAnchorAttempted + (int)anchoredLate),
                @"owner_bounds_y":@((int)lround(owner.bounds.origin.y)),
                @"scene_activation_state":@(liveWindow.windowScene.activationState)
            });
        }
        ScarletIslandViewMetrics resultMetrics = {0};
        NSMutableDictionary *remainingGroups = [NSMutableDictionary dictionary];
        ScarletIslandWalkViews(liveRoot.view, liveWindow, expectedTop, 0,
                               &resultMetrics, remainingGroups);
        NSNumber *afterCount = remainingGroups[
            [NSValue valueWithNonretainedObject:owner]][@"controls"];
        NSUInteger remaining = afterCount.unsignedIntegerValue;
        CGFloat after = owner.adjustedContentInset.top;
        BOOL proved = isfinite(after) &&
            after >= expectedTop - 3 && after <= expectedTop + 12 &&
            remaining < selectedControls &&
            (!always || fabs(owner.contentOffset.y + after) <= 3) &&
            !owner.dragging && !owner.tracking && !owner.decelerating;
        if(!proved) {
            owner.contentInsetAdjustmentBehavior =
                UIScrollViewContentInsetAdjustmentNever;
            owner.contentOffset = originalOffset;
            [owner setNeedsLayout];
            [owner layoutIfNeeded];
            ScarletIslandResidualAutoReceipt(
                always ? @"guest_always_trial_reverted"
                       : @"guest_residual_auto_reverted",
                always ? @"no_immediate_geometry_improvement"
                       : @"no_measured_overlap_reduction",
                expectedTop, before, after, selectedControls, remaining);
            return;
        }
        if(always) {
            // A one-frame success is insufficient: L071's manual value was
            // overwritten by guest relayout after it initially read back.
            // Recheck beyond the existing 2600ms root geometry sample.
            ScarletIslandResidualAutoReceipt(
                @"guest_always_trial_staged", @"native_inset_tentative",
                expectedTop, before, after, selectedControls, remaining);
            __weak UIScrollView *stageOwner = owner;
            __weak UIWindow *stageWindow = liveWindow;
            __weak UIViewController *stageRoot = liveRoot;
            dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
                (int64_t)1100 * NSEC_PER_MSEC),
                dispatch_get_main_queue(), ^{
                UIScrollView *stableOwner = stageOwner;
                UIWindow *stableWindow = stageWindow;
                UIViewController *stableRoot = stageRoot;
                BOOL attached = stableOwner && stableWindow && stableRoot &&
                    stableOwner.window == stableWindow &&
                    stableWindow.rootViewController == stableRoot &&
                    stableWindow.windowScene &&
                    UIInterfaceOrientationIsPortrait(
                        stableWindow.windowScene.interfaceOrientation) &&
                    fabs(stableRoot.view.safeAreaInsets.top - expectedTop) <= 2;
                NSUInteger lateControls = selectedControls;
                CGFloat lateInset = 0;
                if(attached) {
                    ScarletIslandViewMetrics lateMetrics = {0};
                    NSMutableDictionary *lateGroups =
                        [NSMutableDictionary dictionary];
                    ScarletIslandWalkViews(stableRoot.view, stableWindow,
                                           expectedTop, 0, &lateMetrics,
                                           lateGroups);
                    NSNumber *count = lateGroups[
                        [NSValue valueWithNonretainedObject:stableOwner]][@"controls"];
                    lateControls = count.unsignedIntegerValue;
                    lateInset = stableOwner.adjustedContentInset.top;
                }
                BOOL persisted = attached && isfinite(lateInset) &&
                    lateInset >= expectedTop - 3 &&
                    lateInset <= expectedTop + 12 &&
                    lateControls < selectedControls &&
                    stableOwner.contentInsetAdjustmentBehavior ==
                        UIScrollViewContentInsetAdjustmentAlways &&
                    fabs(stableOwner.contentOffset.y + lateInset) <= 3 &&
                    !stableOwner.dragging && !stableOwner.tracking &&
                    !stableOwner.decelerating;
                if(!persisted) {
                    if(stableOwner) {
                        stableOwner.contentInsetAdjustmentBehavior =
                            UIScrollViewContentInsetAdjustmentNever;
                        stableOwner.contentOffset = originalOffset;
                        [stableOwner setNeedsLayout];
                        [stableOwner layoutIfNeeded];
                    }
                    ScarletIslandResidualAutoReceipt(
                        @"guest_always_trial_reverted",
                        @"late_geometry_not_persistent",
                        expectedTop, before, lateInset,
                        selectedControls, lateControls);
                    return;
                }
                objc_setAssociatedObject(stableOwner,
                    &scarletIslandAlwaysAppliedKey, @YES,
                    OBJC_ASSOCIATION_RETAIN_NONATOMIC);
                ScarletIslandResidualAutoReceipt(
                    @"guest_always_trial_retained", @"late_native_inset_confirmed",
                    expectedTop, before, lateInset,
                    selectedControls, lateControls);
            });
            return;
        }
        objc_setAssociatedObject(owner, &scarletIslandResidualAutoAppliedKey,
                                 @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
        ScarletIslandResidualAutoReceipt(
            @"guest_residual_auto_retained", @"automatic_inset_proved",
            expectedTop, before, after, selectedControls, remaining);
    });
}

// R6P31: UIKit ALWAYS, not P29 AUTOMATIC. The iPhone P30 payload proves
// P29 accepted a unique fullscreen nonvertical owner (3 controls), but
// UIKit AUTOMATIC stayed at 0pt and no content moved. Apple's ALWAYS policy
// unconditionally includes safe-area insets. Every mutation is reversible,
// and any failure at either 180ms or ~3130ms restores the guest's own state.
static void ScarletIslandScheduleNonverticalAlwaysInset(
    UIWindow *window, UIViewController *root, CGFloat expectedTop) {

        return;

                             @YES, OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    __weak UIWindow *weakWindow = window;
    __weak UIViewController *weakRoot = root;
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
        (int64_t)1850 * NSEC_PER_MSEC), dispatch_get_main_queue(), ^{
        ScarletIslandTryResidualAutomaticInset(
            weakWindow, weakRoot, expectedTop, YES, NO);
    });
}

// Signed P38 session 4e059ca3 (PID1631) measured one 50pt child Y0,
// parent.safeAreaLayoutGuide Y62, four relevant vertical constraints:
// own height=50, own guide positions and generated autoresize height=50,
// but NO explicit constraint attaching the child top to the parent's guide.
// Unlike P37's overwritten one-shot frame, this uses three owned Auto Layout
// constraints, retaining all existing guest/root/scroll sizes. It is a
// strictly unique, reversible experiment, NOT device-proven visual success.
static void ScarletIslandTryMeasuredHeaderConstraintBridge(
    UIWindow *window, UIView *root, UIView *header, UIView *parent,
    CGFloat safeTop, NSUInteger sample, NSUInteger beforeControls,
    NSUInteger beforeImages, UIView *firstNode, UIView *secondNode) {
    if(sample!=2600 || !window || !root || !header || !parent ||
       !firstNode || !secondNode || beforeControls<2 ||
       objc_getAssociatedObject(root,&scarletIslandHeaderConstraintTrialKey))
        return;
    if(header.superview!=parent || header.window!=window ||
       ![header isDescendantOfView:root] ||
       !ScarletIslandDeepHeaderOwnerReady(header,parent,root,window,
                                         safeTop,0) ||
       !header.translatesAutoresizingMaskIntoConstraints ||
       header.autoresizingMask!=UIViewAutoresizingNone ||
       header.hasAmbiguousLayout ||
       !parent.autoresizesSubviews ||
       !CGAffineTransformIsIdentity(parent.transform) ||
       ![header.nextResponder isKindOfClass:UIViewController.class] ||
       ![parent.nextResponder isKindOfClass:UIViewController.class])
        return;
    UIViewController *child=(UIViewController *)header.nextResponder;
    UIViewController *owner=(UIViewController *)parent.nextResponder;
    if(child.parentViewController!=owner) return;
    // Never activate alongside an existing positional relationship. A
    // generated NSAutoresizingMask height is the only permitted parent
    // constraint on this measured header, and will be removed by turning
    // off translation. The header already owns an explicit 50pt height.
    Class mask=NSClassFromString(@"NSAutoresizingMaskLayoutConstraint");
    BOOL hasOwnHeight=NO;
    for(NSLayoutConstraint *constraint in header.constraints) {
        if(constraint.active && constraint.firstItem==header &&
           constraint.secondItem==nil &&
           constraint.firstAttribute==NSLayoutAttributeHeight &&
           constraint.relation==NSLayoutRelationEqual &&
           fabs(constraint.constant-CGRectGetHeight(header.bounds))<1)
            hasOwnHeight=YES;
    }
    if(!hasOwnHeight) return;
    for(NSLayoutConstraint *constraint in parent.constraints) {
        if(!constraint.active || (mask && [constraint isKindOfClass:mask]))
            continue;
        UIView *one=ScarletIslandConstraintItemView(constraint.firstItem);
        UIView *two=ScarletIslandConstraintItemView(constraint.secondItem);
        if(one==header || two==header) return;
    }
    CGRect original=header.frame;
    CGRect hostBounds=window.bounds;
    CGFloat parentSafeY=CGRectGetMinY(parent.safeAreaLayoutGuide.layoutFrame);
    if(!isfinite(parentSafeY) || !isfinite(original.origin.x) ||
       !isfinite(original.size.width) ||
       fabs(parentSafeY-original.origin.y-safeTop)>5 ||
       fabs(original.size.height-50)>4 ||
       fabs(original.size.width-CGRectGetWidth(parent.bounds))>4)
        return;
    objc_setAssociatedObject(root,&scarletIslandHeaderConstraintTrialKey,
                             @YES,OBJC_ASSOCIATION_RETAIN_NONATOMIC);
    __weak UIView *weakHeader=header,*weakParent=parent,*weakRoot=root;
    __weak UIView *weakFirst=firstNode,*weakSecond=secondNode;
    __weak UIWindow *weakWindow=window;
    // Set only the child header's Auto Layout ownership. Do not set a frame,
    // trigger a forced layout pass or change a global/inset/offset/scale.
    header.translatesAutoresizingMaskIntoConstraints=NO;
    NSArray<NSLayoutConstraint *> *owned=@[
        [header.topAnchor constraintEqualToAnchor:parent.safeAreaLayoutGuide.topAnchor],
        [header.leadingAnchor constraintEqualToAnchor:parent.leadingAnchor
                                           constant:original.origin.x],
        [header.widthAnchor constraintEqualToConstant:original.size.width]
    ];
    [NSLayoutConstraint activateConstraints:owned];
    ScarletIslandDeepHeaderTrialReceipt(
        @"guest_header_constraint_bridge_staged", @"parent_safe_guide_constraint",
        safeTop, original.origin.y, parentSafeY,
        beforeControls, beforeControls, beforeImages, NO);

    // Revert only our constraints and original translation flag; any frame
    // reset is rollback of a failed experiment, never the treatment.
    void (^undo)(UIView *)=^(UIView *h) {
        [NSLayoutConstraint deactivateConstraints:owned];
        if(h && h.superview==weakParent) {
            h.translatesAutoresizingMaskIntoConstraints=YES;
            if(fabs(h.frame.origin.y-parentSafeY)<=3)
                h.frame=original; // Restore original frame after failed trial only.
        }
    };
    // Main queue readbacks only after UIKit naturally resolves its layout.
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
        (int64_t)240*NSEC_PER_MSEC),dispatch_get_main_queue(), ^{
        UIView *h=weakHeader,*p=weakParent,*rt=weakRoot;
        UIWindow *w=weakWindow;
        BOOL attached=ScarletIslandDeepHeaderOwnerReady(
            h,p,rt,w,safeTop,safeTop);
        BOOL sameSize=w &&
            fabs(CGRectGetWidth(w.bounds)-CGRectGetWidth(hostBounds))<2 &&
            fabs(CGRectGetHeight(w.bounds)-CGRectGetHeight(hostBounds))<2 &&
            h && fabs(CGRectGetWidth(h.bounds)-original.size.width)<2 &&
            fabs(CGRectGetHeight(h.bounds)-original.size.height)<2;
        NSUInteger images=beforeImages;
        NSUInteger controls=(attached && sameSize) ?
            ScarletIslandDeepHeaderCountUnsafe(rt,w,safeTop,&images) :
            beforeControls;
        CGRect a=weakFirst && w ?
            ScarletIslandVisibleFrameInWindow(weakFirst,w):CGRectNull;
        CGRect b=weakSecond && w ?
            ScarletIslandVisibleFrameInWindow(weakSecond,w):CGRectNull;
        BOOL pass=attached && sameSize &&
            !CGRectIsNull(a) && !CGRectIsNull(b) &&
            !CGRectIsEmpty(a) && !CGRectIsEmpty(b) &&
            CGRectGetMinY(a)>=safeTop-2 &&
            CGRectGetMinY(b)>=safeTop-2 &&
            controls+2<=beforeControls &&
            images<=beforeImages+1;
        if(!pass) {
            undo(h);
            ScarletIslandDeepHeaderTrialReceipt(
                @"guest_header_constraint_bridge_reverted",
                attached ? @"no_immediate_reduction" : @"native_layout_overrode",
                safeTop,original.origin.y,h ? h.frame.origin.y:-999,
                beforeControls,controls,images,NO);
            return;
        }
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW,
            (int64_t)1200*NSEC_PER_MSEC),dispatch_get_main_queue(), ^{
            UIView *late=weakHeader,*lp=weakParent,*lr=weakRoot;
            UIWindow *lw=weakWindow;
            BOOL stable=ScarletIslandDeepHeaderOwnerReady(
                late,lp,lr,lw,safeTop,safeTop);
            BOOL same=lw &&
                fabs(CGRectGetWidth(lw.bounds)-CGRectGetWidth(hostBounds))<2 &&
                fabs(CGRectGetHeight(lw.bounds)-CGRectGetHeight(hostBounds))<2 &&
                late && fabs(CGRectGetWidth(late.bounds)-original.size.width)<2 &&
                fabs(CGRectGetHeight(late.bounds)-original.size.height)<2;
            NSUInteger lateImages=beforeImages;
            NSUInteger lateControls=(stable && same) ?
                ScarletIslandDeepHeaderCountUnsafe(lr,lw,safeTop,&lateImages):
                beforeControls;
            CGRect one=weakFirst && lw ?
                ScarletIslandVisibleFrameInWindow(weakFirst,lw):CGRectNull;
            CGRect two=weakSecond && lw ?
                ScarletIslandVisibleFrameInWindow(weakSecond,lw):CGRectNull;
            BOOL retained=stable && same &&
                !CGRectIsNull(one) && !CGRectIsNull(two) &&
                !CGRectIsEmpty(one) && !CGRectIsEmpty(two) &&
                CGRectGetMinY(one)>=safeTop-2 &&
                CGRectGetMinY(two)>=safeTop-2 &&
                lateControls+2<=beforeControls &&
                lateImages<=beforeImages+1;
            if(!retained) undo(late);
            ScarletIslandDeepHeaderTrialReceipt(
                retained ? @"guest_header_constraint_bridge_retained"
                         : @"guest_header_constraint_bridge_reverted",
                retained ? @"late_native_anchor_confirmed"
                         : @"late_layout_not_persistent",
                safeTop,original.origin.y,
                late ? late.frame.origin.y:-999,
                beforeControls,lateControls,lateImages,retained);
        });
    });
}

void initScarletGuestIslandContentGuard(void) {
    // The resident opt-in bit is supplied before LC guest bootstrap; all
    // classic LC, embedded SideStore, and unrelated extensions remain inert.
    // R6P17: LiveProcessMain saved this opt-in in the ORIGINAL LC defaults.
    // NUDGuestHooksInit swaps standardUserDefaults to the guest's own domain
    // before this initializer runs. Never gate on that replaced instance.
    if(!NSUserDefaults.isLiveProcess ||
       ![NSUserDefaults.lcUserDefaults
           boolForKey:@"Scarlet.ResidentSDKContract.V1"])
        return;
    static dispatch_once_t once;
    dispatch_once(&once, ^{
        // The very first independent guest-process receipt proves the guard
        // was actually armed; host geometry alone cannot prove this.
        ScarletIslandReceipt(@"guest_root_guard_armed", 0, 0, 0);
        NSNotificationCenter *center = NSNotificationCenter.defaultCenter;
        [center addObserverForName:UIWindowDidBecomeVisibleNotification
            object:nil queue:NSOperationQueue.mainQueue
            usingBlock:^(NSNotification *note) {
                UIWindow *window = [note.object isKindOfClass:UIWindow.class]
                    ? (UIWindow *)note.object : nil;
                if(!window) return;
                scarletIslandLastWindow = window;
                ScarletIslandInspect(window, 0);
            }];
        [center addObserverForName:UIWindowDidBecomeKeyNotification
            object:nil queue:NSOperationQueue.mainQueue
            usingBlock:^(NSNotification *note) {
                UIWindow *window = [note.object isKindOfClass:UIWindow.class]
                    ? (UIWindow *)note.object : nil;
                if(!window) return;
                scarletIslandLastWindow = window;
                ScarletIslandInspect(window, 0);
            }];
        [center addObserverForName:
            @"UIApplicationDidBecomeActiveNotification"
            object:nil queue:NSOperationQueue.mainQueue
            usingBlock:^(__unused NSNotification *note) {
                ScarletIslandInspect(scarletIslandLastWindow, 0);
            }];
    });
}
